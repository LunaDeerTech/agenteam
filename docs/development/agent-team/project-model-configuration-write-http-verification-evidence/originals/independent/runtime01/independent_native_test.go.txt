package model

import (
	"bufio"
	"io"
	"net/http"
	"strings"
	"sync/atomic"
	"testing"
	"time"
)

// A short independent mixed-method sequence. The author's natural 30s body
// deadline remains separate evidence; this does not repeat it.
func TestIndependentConfigurationNativeMixedMethods(t *testing.T) {
	requireProjectConfigurationNative(t)
	h, _, s := configurationTestHandler(t)
	var eof, closes atomic.Int32
	address, terminals, connections := summaryNativeListener(t, http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		r.Body = summaryNativeBody{r.Body, &eof, &closes, false}
		h.ServeHTTP(w, r)
	}), false)
	conn := summaryNativeDial(t, address, false)
	reader := bufio.NewReader(conn)
	for i, tc := range []struct {
		method, path, body string
		status             int
	}{
		{"HEAD", configurationTestLookup, "", 405},
		{"POST", configurationTestProviders, `{"input":` + configurationTestProviderJSON + `}`, 200},
		{"POST", configurationTestLookup, `{"command":"provider.create"}`, 200},
	} {
		if _, err := io.WriteString(conn, configurationNativeRequest(tc.method, tc.path, address, tc.body)); err != nil {
			t.Fatal("native request write", err)
		}
		response, err := http.ReadResponse(reader, &http.Request{Method: tc.method})
		if err != nil {
			t.Fatal("native response read", err)
		}
		data, readErr := io.ReadAll(response.Body)
		closeErr := response.Body.Close()
		if readErr != nil || closeErr != nil || response.StatusCode != tc.status || response.Header.Get("X-Request-ID") == "" || response.Header.Get("Cache-Control") != "no-store" {
			t.Fatal("native complete safe response")
		}
		if i == 0 {
			if len(data) != 0 || response.Header.Get("Allow") != "POST" {
				t.Fatal("HEAD lookup method boundary")
			}
		} else if len(data) == 0 || response.ContentLength != int64(len(data)) || strings.Contains(string(data), "provider-private") {
			t.Fatal("unsafe/incomplete result")
		}
		if summaryNativeTerminal(t, terminals, time.Second).aborted {
			t.Fatal("completed request aborted")
		}
		if i == 0 {
			timer := time.NewTimer(2100 * time.Millisecond)
			<-timer.C
		}
	}
	if connections.Load() != 1 || closes.Load() != 3 || s.writes.Load() != 1 || s.lookups.Load() != 1 {
		t.Fatal("HEAD retirement or subsequent write/lookup reset changed")
	}
	t.Log("one real TCP: HEAD405 Flush/Close -> idle over 2s -> write200 -> lookup200; exactly one write and one lookup")
}
