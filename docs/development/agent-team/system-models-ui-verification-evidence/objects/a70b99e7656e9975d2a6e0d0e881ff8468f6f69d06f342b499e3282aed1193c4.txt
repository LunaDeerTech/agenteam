const command = (kind: ModelCommand['kind'] = 'model.create'): ModelCommand =>

const receipt = (kind: ModelCommand['kind'] = 'model.create') => ({
  kind,

  version: kind === 'model.create' ? '1' : '2',
  affected_references: kind === 'model.delete' ? '10001' : '0',

const json = (value: unknown, status = 200) =>
  new Response(JSON.stringify(value), {
    status,
    headers: {
      'Content-Type': status >= 400 ? 'application/problem+json' : 'application/json',
      'X-Request-ID': id(9),
    },
  })
const problemValue = (code: string, status = 503, commit_state: CommitState = 'not_started') => ({
  type: 'urn:agenteam:problem:test',
  title: 'Failure',
  status,
  code,
  detail: '',
  instance: '/api/v1/system/models',
  request_id: id(9),
  commit_state,
})
const problem = (code: string, status = 503, state: CommitState = 'not_started') =>
  json(problemValue(code, status, state), status)
function barrier<T>() {
  let resolve!: (value: T) => void
  const promise = new Promise<T>((done) => {
    resolve = done
  })
  return { promise, resolve }
}
const owners: SessionController[] = []
async function fixture() {
  let current = view()
  let session: Fetch = async () => json(current)
  let perform: Fetch = async (path, init) => {
    if (path.endsWith('/lookup'))
      return json({ found: true, receipt: receipt(JSON.parse(init.body as string).command) })
    if (path.endsWith('/deletion-impact')) return json(impact())
    if (init.method === 'GET')
      return json(path.includes('?') ? { items: [model()], next_cursor: null } : model())
    return json(
      receipt(
        init.method === 'PUT'
          ? 'model.update'
          : init.method === 'DELETE'
            ? 'model.delete'
            : 'model.create',
      ),
    )
  }
  let other: Fetch = async (path, init) => {
    if (path === '/api/v1/me') return json({ user: current.user, avatar: null })
    if (path === '/api/v1/me/preferences') return json({ theme: 'dark', version: '2' })
    if (path.endsWith('/model-credentials'))
      return json({ credential_id: id(11), purpose: 'model', version: '1', deleted: false })
    if (path.endsWith('/model-providers') && init.method === 'POST')
      return json({
        kind: 'provider.create',
        resource_id: context.provider_id,
        version: '1',
        affected_references: '0',
      })
    if (path.includes('/model-providers'))
      return json(path.includes('?') ? { items: [provider()], next_cursor: null } : provider())
    if (path.endsWith('/invitations') && init.method === 'POST')
      return json({ id: id(30), job_id: id(31), version: '1' }, 201)
    if (path === '/api/v1/password-resets/request') return json({ accepted: true }, 202)
    return json({ items: [] })
  }
  const fetch = vi.fn<Fetch>(async (path, init) => {
    if (path === '/api/v1/session') return session(path, init)
    if (path.endsWith('/bootstrap'))
      return json({
        csrf_token: 'A'.repeat(43),
        challenge_modes: ['rotate'],
        delivery_channel: 'backend_log',
      })
    if (path.endsWith('/logout')) return new Response(null, { status: 204 })
    if (
      path.startsWith('/api/v1/system/models') ||
      (path.endsWith('/model-commands/lookup') &&
        JSON.parse(init.body as string).command.startsWith('model.'))
    )
      return perform(path, init)
    return other(path, init)
  })
  const modelAPI = createSystemModelAPI(fetch)
  const auth = createSessionController(
    createAccountAPI(fetch),
    createSystemAccountAPI(fetch),
    createSystemInvitationAPI(fetch),
    createSystemProviderAPI(fetch),
    modelAPI,
  )
  owners.push(auth)
  await auth.restore()
  return {
    auth,
    fetch,
    modelAPI,
    setPerform(value: Fetch) {
      perform = value
    },
    setOther(value: Fetch) {
      other = value
    },
    setSession(value: SessionView) {
      current = value
    },
    setSessionRequest(value: Fetch) {
      session = value
    },
    writes() {
      return fetch.mock.calls.filter(
        ([path, init]) => path.startsWith('/api/v1/system/models') && init.method !== 'GET',
      )
    },
  }
}

    expect(f.auth.system.models.progress?.contextValid).toBe(true)
    f.setSessionRequest(async () => json(view(id(3))))

    expect(f.auth.system.models.progress).toBeNull()
    expect(f.fetch.mock.calls.slice(count).map(([path]) => path)).toEqual(['/api/v1/session'])
  })
  it.each(['write', 'lookup'] as const)(
    'invalidates exact identity on current Model %s CSRF_FAILED',
    async (operation) => {
      const f = await fixture()
      if (operation === 'lookup') {

  it.each(['read', 'write', 'lookup'] as const)(
    'rejects current revoked Session during Model %s and clears its private state',
    async (operation) => {
      const f = await fixture()
      if (operation === 'lookup') {
        f.setPerform(async () => {
          throw new Error('lost')
        })
        await f.auth.system.models.start(command()).catch(() => undefined)
      }
      f.setPerform(async () => problem('SESSION_REVOKED', 401))
      await (

  it.each(['FORBIDDEN', 'SESSION_REVOKED', 'CSRF_FAILED'] as const)(
    'does not apply old-generation %s after global leave; still blocks the new identity until join',
    async (code) => {
      const f = await fixture()
      // This barrier models an already parsed API failure arriving after abandon;
      // transport EOF/cancel behavior is separately exercised above.
      let reject!: (error: unknown) => void
      const failure = new Promise<never>((_resolve, fail) => {
        reject = fail
      })
      f.modelAPI.getModel = () => failure
      const old = f.auth.system.models.get(target).catch((e: unknown) => e)
      await flushPromises()
      f.auth.leave()
      expect(await old).toMatchObject({ kind: 'cancelled' })
      const count = f.fetch.mock.calls.length
      f.setSession(view(id(3)))
      await f.auth.restore()
      expect(f.fetch.mock.calls).toHaveLength(count)
      expect(f.auth.state.busy).toBe(true)
      reject(
        new AccountFailure('problem', problemValue(code, code === 'SESSION_REVOKED' ? 401 : 403)),
      )
      await flushPromises()
      expect(f.auth.system.denied).toBe(false)
      expect(f.auth.state.phase).toBe('checking')
      expect(f.auth.state.busy).toBe(false)
      await f.auth.restore()
      expect(f.auth.personalContext.identity?.sessionID).toBe(id(3))
      expect(f.auth.system.models.progress).toBeNull()
    },
  )
  it('does not erase a successor started synchronously from confirmation', async () => {
    const f = await fixture(),
      held = barrier<Response>()
    let next: Promise<unknown> | undefined,
      once = false
    const stop = watch(
      () => f.auth.system.models.progress?.phase,
      (phase) => {
        if (phase !== 'confirmed' || once) return
        once = true
        f.auth.system.models.abandon()
        f.setPerform(() => held.promise)
        next = f.auth.system.models.start(command())
      },
      { flush: 'sync' },
    )
    await expect(f.auth.system.models.start(command())).rejects.toMatchObject({ kind: 'cancelled' })
    await flushPromises()
    expect(f.auth.system.models.progress?.phase).toBe('submitting')
    expect(f.auth.state.busy).toBe(true)
    held.resolve(json(receipt()))
    await next
    stop()
    expect(f.auth.system.models.progress?.phase).toBe('confirmed')
    expect(f.writes()).toHaveLength(2)
  })
