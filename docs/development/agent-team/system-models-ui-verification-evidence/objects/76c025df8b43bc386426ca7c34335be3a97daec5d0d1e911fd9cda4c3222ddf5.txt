function gate<T>() { let resolve!: (value: T) => void; const promise = new Promise<T>((done) => { resolve = done }); return { promise, resolve } }
function nativeBody(value: unknown) {
  const cancelled = gate<void>(), cancelTail = gate<void>()
  let controller!: ReadableStreamDefaultController<Uint8Array>
  const response = new Response(new ReadableStream<Uint8Array>({
    start(c) { controller = c; c.enqueue(new TextEncoder().encode(JSON.stringify(value))) },
    cancel() { cancelled.resolve(); return cancelTail.promise },
  }), { headers: { 'Content-Type': 'application/json' } })
  return { response, eof: () => controller.close(), cancelled, cancelTail }
}

async function fixture(app = false) {
  const sources = new Map([[id(100), provider()], [id(110), provider(110)]])
  const records = new Map([[id(200), model()], [id(300), model(300, 110)]])
  let current = session(), memory = false, conflict = false, failRefresh = false
  let sessionReply: Fetch = async () => json(current)
  let providersReply: Fetch = async () => json({ items: [...sources.values()].sort((a, b) => b.id.localeCompare(a.id)), next_cursor: null })
  let modelsReply: Fetch = async (path) => json({ items: [...records.values()].filter((m) => m.provider_id === new URL(path, 'http://fixture.invalid').searchParams.get('provider_id')).sort((a, b) => b.id.localeCompare(a.id)), next_cursor: null })
  let detailReply: Fetch = async (path) => {
    if (failRefresh) { failRefresh = false; return problem() }
    return records.has(path.split('/').at(-1)!) ? json(records.get(path.split('/').at(-1)!)) : problem('NOT_FOUND', 404)
  }
  const calls: { path: string; init: RequestInit }[] = []
  const fetcher: Fetch = async (path, init) => {
    calls.push({ path, init })
    if (path === '/api/v1/session') return sessionReply(path, init)
    if (path.startsWith('/api/v1/system/model-providers?')) return providersReply(path, init)
    if (path.startsWith('/api/v1/system/model-providers/') && init.method === 'GET') return json(sources.get(path.split('/').at(-1)!))
    if (path.startsWith('/api/v1/system/models?')) return modelsReply(path, init)
    if (path.endsWith('/deletion-impact')) {
      const value: ModelDeletionImpact = { model_id: id(200), version: records.get(id(200))!.version, reference_count: memory ? '1' : '0', reference_groups: memory ? [{ owner_kind: 'platform_selector', role: 'memory', count: '1' }] : [], replacement_requirement: memory ? 'required' : 'none', delete_blocker: null }
      return json(value)
    }
    if (path.startsWith('/api/v1/system/models/') && init.method === 'GET') return detailReply(path, init)
    if (path.startsWith('/api/v1/system/models') && init.method !== 'GET') {
      if (conflict) { conflict = false; records.set(id(200), { ...records.get(id(200))!, version: '2' }); return problem('VERSION_CONFLICT', 409) }
      const body = JSON.parse(String(init.body)), creating = init.method === 'POST', target = creating ? id(201) : path.split('/').at(-1)!, version = creating ? '1' : String(BigInt(body.expected_version) + 1n)
      if (init.method === 'DELETE') records.delete(target)
      else records.set(target, { id: target, provider_id: creating ? body.provider_id : records.get(target)!.provider_id, input: body.input, version, created_at: time, updated_at: time })
      return json({ kind: creating ? 'model.create' : init.method === 'DELETE' ? 'model.delete' : 'model.update', resource_id: target, version, affected_references: init.method === 'DELETE' ? '1' : '0' })
    }
    if (path.startsWith('/api/v1/system/users?') || path.startsWith('/api/v1/system/invitations?')) return json({ items: [] })
    if (path.endsWith('/bootstrap')) return json({ csrf_token: 'A'.repeat(43), challenge_modes: ['rotate'], delivery_channel: 'backend_log' })
    if (path.endsWith('/logout')) return new Response(null, { status: 204 })
    throw new Error('independent unexpected route')
  }
  const auth = createSessionController(createAccountAPI(fetcher), createSystemAccountAPI(fetcher), createSystemInvitationAPI(fetcher), createSystemProviderAPI(fetcher), createSystemModelAPI(fetcher))
  selected.auth = auth
  let page: SystemModelsController | undefined
  const router = createRouter({ history: createMemoryHistory(), routes: [
    { path: '/', component: HomeView, meta: { authentication: true, protected: true } },
    { path: '/login', name: 'login', component: LoginView, meta: { authentication: true } },
    { path: '/system', component: SystemSettingsView, redirect: '/system/users', meta: { authentication: true, protected: true, systemAdmin: true, navigation: { label: '系统设置', order: 20 } }, children: [ { path: 'users', component: SystemUsersView }, { path: 'invitations', component: SystemInvitationsView }, { path: 'providers', component: SystemProvidersView }, { path: 'models', component: SystemModelsView } ] },
  ] })
  if (app) {
    installAuthentication(router, auth)
    await router.push('/system/models'); await router.isReady()
    const host = document.createElement('div'); host.id = 'app'; document.body.append(host)
    mounted = mount(App, { attachTo: host, global: { plugins: [router] } })
  } else {
    await auth.restore()
    page = createSystemModels(auth); controllers.push(page); page.attach()
  }
  await flushPromises()
  return { auth, page: page!, router, sources, records, calls,
    memory: () => { memory = true }, conflict: () => { conflict = true }, failRefresh: () => { failRefresh = true },
    setSession: (reply: Fetch) => { sessionReply = reply }, resetSession: (n = 2) => { current = session(n); sessionReply = async () => json(current) },
    setProviders: (reply: Fetch) => { providersReply = reply }, setModels: (reply: Fetch) => { modelsReply = reply }, setDetail: (reply: Fetch) => { detailReply = reply },
    resetDetail: () => { detailReply = async (p) => records.has(p.split('/').at(-1)!) ? json(records.get(p.split('/').at(-1)!)) : problem('NOT_FOUND', 404) },
    writes: () => calls.filter((c) => c.path.startsWith('/api/v1/system/models') && c.init.method !== 'GET'),
  }
}

function label(button: HTMLButtonElement) { return button.getAttribute('aria-label') ?? button.querySelector('.button-label[aria-hidden="false"]')?.textContent?.trim() ?? button.textContent?.trim() }

const dialogs = () => [...document.querySelectorAll<HTMLElement>('[role="dialog"]')]
