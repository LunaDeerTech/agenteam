const parseImpact = (value: unknown) =>

function changed(value: unknown, path: string, replacement?: unknown, remove = false) {
  const result = JSON.parse(JSON.stringify(value)) as Record<string, unknown>
  const keys = path.split('.')
  let parent = result
  for (const key of keys.slice(0, -1)) parent = parent[key] as Record<string, unknown>
  if (remove) delete parent[keys.at(-1)!]
  else parent[keys.at(-1)!] = replacement
  return result
}

describe('Exact registered Model deletion impact', () => {
  const all = () => ({
    ...emptyImpact(),
    reference_count: '10000',
    replacement_requirement: 'required',
    delete_blocker: 'reference_adapter_unbound',
    reference_groups: [
      { owner_kind: 'agent', role: 'agent_model', count: '9994' },
      { owner_kind: 'agent', role: 'approval_model', count: '1' },
      { owner_kind: 'platform_selector', role: 'embedding', count: '1' },
      { owner_kind: 'platform_selector', role: 'image', count: '1' },
      { owner_kind: 'platform_selector', role: 'memory', count: '1' },
      { owner_kind: 'platform_selector', role: 'reranker', count: '1' },
      { owner_kind: 'project_summary', role: 'meeting_summary', count: '1' },
    ],
  })
  it('accepts zero, exact 10000 and every closed group with the correct requirement/blocker', async () => {
    expect(await parseImpact(emptyImpact())).toEqual(emptyImpact())
    expect(await parseImpact(all())).toEqual(all())
    for (const group of all().reference_groups) {
      const value = {
        ...emptyImpact(),
        reference_count: group.count,
        reference_groups: [group],
        replacement_requirement: ['image', 'reranker'].includes(group.role)
          ? 'optional'
          : 'required',
        delete_blocker:
          group.owner_kind === 'platform_selector' ? null : 'reference_adapter_unbound',
      }
      const result = await parseImpact(value)
      expect(result).toEqual(value)
      expect(Object.isFrozen(result.reference_groups[0])).toBe(true)
    }
  })
  it.each(Object.keys(emptyImpact()))('rejects missing required Impact %s', async (field) => {
    await expect(parseImpact(changed(emptyImpact(), field, undefined, true))).rejects.toMatchObject(
      { kind: 'invalid-response' },
    )
  })
  it.each([
    ['model_id', id(101)],
    ['version', '0'],
    ['reference_count', '10001'],
    ['reference_count', 0],
    ['reference_count', '00'],
    ['reference_groups', null],
    ['reference_groups', [{}]],
    ['replacement_requirement', 'optional'],
    ['replacement_requirement', null],
    ['delete_blocker', 'reference_adapter_unbound'],
    ['delete_blocker', 'unknown'],
    ['owner_id', id(200)],
  ])('rejects invalid zero Impact %s=%s', async (field, value) => {
    await expect(parseImpact(changed(emptyImpact(), field as string, value))).rejects.toMatchObject(
      { kind: 'invalid-response' },
    )
  })
  it.each([
    ['reference_count', '9999'],
    ['reference_count', '10001'],
    ['reference_groups.0.count', '0'],
    ['reference_groups.0.count', '9995'],
    ['reference_groups.0.count', '01'],
    ['reference_groups.0.name', 'private'],
    ['reference_groups.0.owner_kind', 'project'],
    ['reference_groups.0.role', 'embedding'],
    ['reference_groups.0.count', null],
    ['replacement_requirement', 'optional'],
    ['delete_blocker', null],
  ])('rejects invalid grouped Impact %s=%s', async (field, value) => {
    await expect(parseImpact(changed(all(), field as string, value))).rejects.toMatchObject({
      kind: 'invalid-response',
    })
  })
  it('rejects order changes, duplicate groups and eight groups rather than publishing a partial candidate', async () => {
    const value = all()
    for (const groups of [
      value.reference_groups.toReversed(),
      [...value.reference_groups, value.reference_groups[0]!],
      [value.reference_groups[0]!, value.reference_groups[0]!],
    ])
      await expect(parseImpact({ ...value, reference_groups: groups })).rejects.toMatchObject({

      })
  })
})

describe('Model transport retains inherited page and stream limits', () => {
  it('checks full microseconds, duplicate IDs, Provider identity and complete pages', async () => {
    const rows = Array.from({ length: 25 }, (_, index) => model(100 - index))
    const api = (value: unknown) =>
      createSystemModelAPI(async () => json(value)).listModels(context, signal())
    expect((await api({ items: rows, next_cursor: 'next' })).items).toHaveLength(25)
    for (const value of [
      { items: rows.slice(1), next_cursor: 'next' },
      { items: rows, next_cursor: '' },
      { items: [model(), model()], next_cursor: null },
      { items: rows.toReversed(), next_cursor: null },
      { items: [{ ...model(), provider_id: id(2) }], next_cursor: null },
      { items: rows, next_cursor: 'x'.repeat(8193) },
      { items: rows },
      {
        items: [{ ...model(101), created_at: '2026-10-06T12:34:56.123455Z' }, model(100)],
        next_cursor: null,
      },
    ])
      await expect(api(value)).rejects.toMatchObject({ kind: 'invalid-response' })
  })
  it('keeps the 2MiB Provider success exception, 600000B Model/error cap and 16KiB write cap', async () => {
    const large = JSON.stringify({ items: [provider()], next_cursor: null }).padEnd(600001, ' ')
    const fetch: Fetch = async () =>
      new Response(large, { headers: { 'Content-Type': 'application/json' } })
    expect((await createSystemModelAPI(fetch).listProviders({}, signal())).items).toHaveLength(1)
    await expect(
      createSystemModelAPI(fetch).getDeletionImpact(target.id, signal()),
    ).rejects.toMatchObject({ kind: 'invalid-response' })
    const encoded = JSON.stringify(model())
    const long = encoded + ' '.repeat(600000 - new TextEncoder().encode(encoded).byteLength)
    expect(
      await createSystemModelAPI(
        async () => new Response(long, { headers: { 'Content-Type': 'application/json' } }),
      ).getModel(target, signal()),
    ).toEqual(model())
    await expect(
      createSystemModelAPI(
        async () => new Response(long + ' ', { headers: { 'Content-Type': 'application/json' } }),
      ).getModel(target, signal()),
    ).rejects.toMatchObject({ kind: 'invalid-response' })
    const noFetch = vi.fn<Fetch>(),
      request = accountTransport(noFetch)
    await expect(
      request('createModel', (v) => v, {
        signal: signal(),
        csrf: 'S'.repeat(43),
        key: 'model',
        body: { value: 'x'.repeat(16 * 1024) },
      }),
    ).rejects.toMatchObject({ kind: 'invalid-input' })
    await expect(
      request('getModelDeletionImpact', (v) => v, {
        signal: signal(),
        target: target.id,
        cursor: 'extra',
      } as never),
    ).rejects.toMatchObject({ kind: 'invalid-input' })
    expect(noFetch).not.toHaveBeenCalled()
  })
})
