import { test, expect } from '@playwright/test'
import {
  material, result, ipc, top, button, title, facts, seq, writes, canonical,
  factAfter, session, setup, selectProvider, selectModel, removePreview,
  candidateProvider, choose, perform, confirmed, dbFacts, focusContained,
} from './fixed-helpers'

test('[independent-replay] stale none preview, current optional reference and original deleted-object replay', async ({ page }) => {
  await setup(page)
  const ids = material().ids
  await selectProvider(page, ids.target_provider)
  const target = await selectModel(page, ids.target)
  const impact = await removePreview(page)
  expect(impact).toMatchObject({ model_id: target.id, version: target.version, reference_count: '0', replacement_requirement: 'none', delete_blocker: null })
  expect(impact.reference_groups).toEqual([])
  expect((await writes(page)).length).toBe(0)
  await ipc('independent-check', { stage: 'preview' })

  // Formal selector transaction after the actual zero preview. Do not refresh UI.
  await ipc('selection', { stage: 'reranker', id: target.id })
  const linked = await dbFacts(target.id)
  expect(linked.version).toBe(target.version)
  expect(linked.target_references).toBe(1)
  expect(linked.selection.reranker).toBe(target.id)
  await ipc('independent-check', { stage: 'referenced' })
  await ipc('arm-drop', { stage: 'delete', id: target.id })
  await perform(page, '确认删除 Model')
  await expect(title(page, '请求结果未确认')).toBeVisible()
  const lost = await ipc('drop-facts')
  expect(lost.dropped === 1 && lost.replayed === 0 && lost.target === target.id && !lost.armed).toBe(true)
  const accepted = await dbFacts(target.id)
  expect(accepted.exists).toBe(false)
  expect(accepted.commands).toBe(linked.commands + 1)
  expect(accepted.audits).toBe(linked.audits + 1)
  expect(accepted.events).toBe(linked.events + 1)
  expect(accepted.target_references).toBe(0)
  expect(accepted.selection.reranker).toBeNull()
  expect(accepted.selection.version).toBe(String(BigInt(linked.selection.version) + 1n))
  await ipc('independent-check', { stage: 'accepted' })

  const identity = await session(page), before = await seq(page)
  await top(page).getByRole('button', { name: '检查原请求', exact: true }).click()
  await factAfter(page, before, f => f.path === '/api/v1/session' && f.status === 200 && f.eof === true && f.session?.user_id === identity.user_id && f.session.session_id === identity.session_id && f.session.role === identity.role)
  await factAfter(page, before, f => f.path === '/api/v1/system/model-commands/lookup' && f.status === 200 && f.eof === true && f.found === true)
  await factAfter(page, before, f => f.path === '/api/v1/system/models/' + target.id && f.status === 404 && f.eof === true && f.code === 'NOT_FOUND')
  await expect(top(page)).toContainText('已观察到历史回执')
  await expect(title(page, '删除影响尚未确认')).toBeVisible()
  expect((await writes(page)).length).toBe(1)
  expect((await facts(page)).some(f => f.receipt?.kind === 'model.delete')).toBe(false)
  expect((await ipc('drop-facts')).replayed).toBe(0)
  expect(canonical(await dbFacts(target.id))).toBe(canonical(accepted))

  const retry = await perform(page, '重试原请求')
  const receipt = await confirmed(page, retry, 'model.delete')
  expect(receipt).toEqual({ kind: 'model.delete', resource_id: target.id, version: String(BigInt(target.version) + 1n), affected_references: '1' })
  await factAfter(page, retry, f => f.path === '/api/v1/system/models/' + target.id && f.method === 'DELETE' && f.eof === true && f.receipt?.affected_references === '1')
  expect((await writes(page)).length).toBe(2)
  expect(canonical(await dbFacts(target.id))).toBe(canonical(accepted))
  const final = await ipc('independent-check', { stage: 'replayed' })
  expect(final.same_key && final.same_body && final.same_csrf && final.same_path && final.complete_requests && final.unique_facts).toBe(true)
  result({ replay: true, zero_preview: true, optional_reference: true, current_404: true, reads_not_receipt: true, exact_replay: true, affected_one: true, unique_facts: true })
})

test('[independent-replacement] current candidate rejection, explicit review and pending confirmation recovery', async ({ page }) => {
  await setup(page)
  const ids = material().ids
  await selectProvider(page, ids.target_provider)
  const target = await selectModel(page, ids.target)
  const impact = await removePreview(page)
  expect(impact).toMatchObject({ model_id: target.id, version: target.version, reference_count: '1', replacement_requirement: 'required', delete_blocker: null })
  expect(impact.reference_groups).toEqual([{ owner_kind: 'platform_selector', role: 'memory', count: '1' }])
  expect(ids.replacement_provider !== ids.target_provider).toBe(true)
  await candidateProvider(page, ids.replacement_provider)
  await choose(page, ids.replacement)
  await ipc('disable', { id: ids.replacement })
  const disabled = await dbFacts(target.id)
  await ipc('independent-check', { stage: 'candidate-disabled' })
  const rejected = await perform(page, '确认删除 Model')
  await factAfter(page, rejected, f => f.path === '/api/v1/system/models/' + target.id && f.method === 'DELETE' && f.status === 400 && f.eof === true && f.code === 'INVALID_ARGUMENT' && f.commit_state === 'not_started')
  await expect(top(page)).toContainText('请检查原字段长度、Unicode、启用状态及能力声明。输入不会自动改写。')
  expect((await writes(page)).length).toBe(1)
  expect(canonical(await dbFacts(target.id))).toBe(canonical(disabled))
  await ipc('independent-check', { stage: 'rejected' })

  await ipc('enable', { id: ids.replacement })
  await top(page).getByRole('button', { name: '重新读取并核对删除预览', exact: true }).click()
  await expect(top(page).getByRole('button', { name: '确认删除 Model', exact: true })).toBeDisabled()
  await candidateProvider(page, ids.replacement_provider)
  await choose(page, ids.replacement)
  expect((await writes(page)).length).toBe(1)
  await ipc('independent-check', { stage: 'candidate-restored' })
  await top(page).getByRole('button', { name: '取消', exact: true }).click()
  await expect(top(page)).toHaveAccessibleName('放弃未保存修改？')
  const identity = await session(page), before = await seq(page)
  await ipc('arm-session-fail')
  const checking = await page.evaluate(async () => {
    let observer: MutationObserver | undefined
    try {
      return await new Promise<boolean>(resolve => {
        observer = new MutationObserver(() => {
          if (document.querySelector('.session-check')) resolve(document.querySelectorAll('[role="dialog"]').length === 0 && document.body.style.overflow !== 'hidden')
        })
        observer.observe(document.body, { childList: true, subtree: true })
        window.dispatchEvent(new Event('pageshow'))
      })
    } finally { observer?.disconnect() }
  })
  expect(checking).toBe(true)
  // text/plain 503 follows Content-Type rejection and cancel, without reader EOF.
  await factAfter(page, before, f => f.path === '/api/v1/session' && f.status === 503)
  await expect(title(page, '会话尚未确认')).toBeVisible()
  await expect(page.locator('[role="dialog"]')).toHaveCount(0)
  const restore = button(page, '检查当前会话')
  await expect(restore).toBeEnabled()
  expect(await restore.evaluate(n => n.closest('[inert]') === null)).toBe(true)
  expect((await ipc('failure-facts')).session_failures).toBe(1)
  const restoreAfter = await seq(page)
  await restore.click()
  await factAfter(page, restoreAfter, f => f.path === '/api/v1/session' && f.status === 200 && f.eof === true && f.session?.user_id === identity.user_id && f.session.session_id === identity.session_id && f.session.role === identity.role)
  await expect(page.locator('[role="dialog"]')).toHaveCount(2)
  await expect(page.getByRole('dialog')).toHaveCount(1)
  expect(await page.locator('[role="dialog"]').evaluateAll(nodes => nodes.map(n => ({ inert: (n as HTMLElement).inert, hidden: n.getAttribute('aria-hidden') === 'true' })))).toEqual([{ inert: true, hidden: true }, { inert: false, hidden: false }])
  await expect(top(page)).toHaveAccessibleName('放弃未保存修改？')
  await top(page).getByRole('button', { name: '继续编辑', exact: true }).click()
  await expect(page.locator('[role="dialog"]')).toHaveCount(1)
  await expect(top(page)).toHaveAccessibleName('删除 Model')
  await expect.poll(() => top(page).evaluate(n => n.contains(document.activeElement))).toBe(true)
  await focusContained(page)
  await expect(top(page)).toContainText('已选择替代')
  expect((await writes(page)).length).toBe(1)
  const completed = await perform(page, '确认删除 Model')
  const receipt = await confirmed(page, completed, 'model.delete')
  expect(receipt).toEqual({ kind: 'model.delete', resource_id: target.id, version: String(BigInt(target.version) + 1n), affected_references: '1' })
  await factAfter(page, completed, f => f.path === '/api/v1/system/models/' + target.id && f.method === 'DELETE' && f.eof === true && f.receipt?.affected_references === '1')
  expect((await writes(page)).length).toBe(2)
  const final = await ipc('independent-check', { stage: 'confirmed' })
  expect(final.new_key && final.same_csrf && final.complete_requests && final.unique_facts && final.index_consistent).toBe(true)
  result({ rejection: true, atomic_failure: true, explicit_review: true, checking: true, same_identity: true, focus: true, cross_provider: true, unique_facts: true })
})
