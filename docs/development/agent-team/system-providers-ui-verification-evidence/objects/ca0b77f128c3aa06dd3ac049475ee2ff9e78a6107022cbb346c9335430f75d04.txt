import { describe,it,expect } from 'vitest'
import { captureProviderCommand,createSystemProviderAPI } from '../api/system-providers'
import cases from './extra-url-oracle.json'
const target='01900000-0000-7000-8000-000000000001', instant='2026-10-06T12:34:56.123456Z'
describe('Independent new URL branches',()=>{
  it.each(cases)('matches the exact fixed Go branch for $case',async(c)=>{
    const input={name:'Provider',protocol:'openai-chat-completions' as const,base_url:c.url,enabled:false,credential_ref:null,options:{}}
    let capture=false,parse=false,exact=false
    try{const value=captureProviderCommand({kind:'provider.create',input});capture=value.kind==='provider.create'&&value.input.base_url===c.url}catch{}
    try{const result=await createSystemProviderAPI(async()=>new Response(JSON.stringify({id:target,input,version:'1',created_at:instant,updated_at:instant}),{headers:{'Content-Type':'application/json'}})).getProvider(target,new AbortController().signal);parse=true;exact=result.input.base_url===c.url}catch{}
    expect({capture,parse,exact}).toEqual({capture:c.go_accepted,parse:c.go_accepted,exact:c.go_accepted})
  })
})
