// Reduced static DOM with exact frozen CSS; layout diagnosis only, not App or
// business evidence. No HTTP server, transport fixture, credentials, or API.
import { readFile, writeFile, readdir } from 'node:fs/promises';
import { resolve } from 'node:path';
import { createRequire } from 'node:module';
import assert from 'node:assert/strict';
const require = createRequire(resolve('workspace/tests/account-captcha-web/package.json'));
const { chromium } = require('playwright');
const run = process.env.PROBE_RUN, assets = resolve('input06/web/dist/assets');
const styles = await Promise.all((await readdir(assets)).filter(name => name.endsWith('.css')).map(name => readFile(resolve(assets, name), 'utf8')));
const css = styles.join('\n');
const providerScope = css.match(/\.provider-table\[([^\]]+)\]/)[1];
const shellScope = css.match(/\.settings-shell\[([^\]]+)\]/)[1];
const scoped = (html, scope) => html.replace(/<([a-z][a-z0-9]*)(?=[\s>])/g, '<$1 ' + scope);
const endpoint = 'https://provider.invalid/' + 'long-path/'.repeat(100);
const provider = scoped(`<div class="system-providers"><h1>Providers</h1>
  <div class="provider-table"><table><caption class="sr-only">Providers配置</caption>
  <thead><tr><th>名称</th><th>协议</th><th>Base URL</th><th>状态</th><th>操作</th></tr></thead>
  <tbody><tr><td data-label="名称">Owned ${'长名称'.repeat(35)}</td><td data-label="协议">openai-chat-completions</td>
  <td data-label="Base URL">${endpoint}</td><td data-label="状态">启用</td><td data-label="操作"><button>查看</button></td></tr></tbody></table></div>
  <section class="provider-detail"><h2>Provider详情</h2><dl class="provider-facts"><dt>Base URL</dt><dd>${endpoint}</dd></dl>
  <section><h3>Models</h3><div class="provider-table"><table><caption class="sr-only">当前Provider的Models</caption>
  <thead><tr><th>名称</th><th>原生模型ID</th><th>类型</th><th>状态</th></tr></thead>
  <tbody><tr><td>Model</td><td>native-99</td><td>chat</td><td>启用</td></tr></tbody></table></div></section></section></div>`, providerScope);
const settings = scoped(`<div class="settings-shell"><aside class="settings-sidebar">系统设置</aside><section class="settings-content">${provider}</section></div>`, shellScope);
const html = `<html data-theme="light"><head><style>${css}</style></head><body><div class="app-shell"><header class="system-nav">Layout diagnosis</header><main class="app-content">${settings}</main></div></body></html>`;
await writeFile(resolve(run, 'reduced-dom.html'), html);
let browser, timedOut = false;
const timeout = setTimeout(() => { timedOut = true; void browser?.close(); }, 45_000);
try {
  browser = await chromium.launch({ executablePath: '/usr/bin/chromium', headless: true, timeout: 10_000, args: ['--no-sandbox', '--disable-dev-shm-usage'] });
  const page = await browser.newPage({ viewport: { width: 1440, height: 900 }, reducedMotion: 'reduce' });
  await page.setContent(html);
  const metrics = () => page.evaluate(() => {
    const rect = node => { const r = node.getBoundingClientRect(); return { x:r.x, y:r.y, width:r.width, height:r.height }; };
    const main = document.querySelector('main.app-content');
    return { innerHeight, documentHeight:document.documentElement.scrollHeight, bodyHeight:document.body.scrollHeight,
      main:{ clientHeight:main.clientHeight, scrollHeight:main.scrollHeight, scrollTop:main.scrollTop },
      captions:Array.from(document.querySelectorAll('.provider-table caption'), node => ({
        offsetParent:node.offsetParent?.tagName??null, offsetParentClass:node.offsetParent?.className??null,
        anchored:node.offsetParent===node.closest('.provider-table'), rect:rect(node),
        wrapperRect:rect(node.closest('.provider-table')), clip:getComputedStyle(node).clipPath,
      })) };
  });
  const settle = () => page.evaluate(() => new Promise(done => requestAnimationFrame(() => requestAnimationFrame(done))));
  const results = [];
  for (const width of [1440,390]) {
    await page.setViewportSize({ width,height:900 });await settle();
    const before = await metrics();
    const style = await page.addStyleTag({content:'.provider-table { position: relative !important; }'});await settle();
    const after = await metrics();results.push({width,before,after});
    await style.evaluate(node=>node.remove());await settle();
  }
  await writeFile(resolve(run,'metrics.json'),JSON.stringify({evidence_kind:'reduced static DOM/exact input06 CSS diagnosis; not App/backend/business evidence',results},null,2)+'\n');
  for(const row of results) {
    assert.equal(row.before.captions.length,2);assert.equal(row.before.captions.every(c=>!c.anchored),true);
    assert.ok(row.before.documentHeight>row.before.innerHeight);
    assert.equal(row.after.captions.every(c=>c.anchored),true);assert.ok(row.after.documentHeight<=row.after.innerHeight+1);
  }
  assert.equal(timedOut,false);
  console.log(JSON.stringify({diagnosis:'PASS',viewports:results.map(row=>({width:row.width,old_document_height:row.before.documentHeight,new_document_height:row.after.documentHeight,old_parents:row.before.captions.map(c=>c.offsetParent),new_anchors:row.after.captions.map(c=>c.anchored)})),backend_requests:0}));
} finally {
  clearTimeout(timeout);if(browser)await browser.close();
  await writeFile(resolve(run,'closed.json'),JSON.stringify({closedBrowser:true,server_started:false,timedOut})+'\n');
}
