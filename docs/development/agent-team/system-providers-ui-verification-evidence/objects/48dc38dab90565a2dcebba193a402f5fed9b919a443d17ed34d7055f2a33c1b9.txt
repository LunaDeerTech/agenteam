// Only this owned same-origin server controls a completed response. Production
// authentication, commands, transactions, origin and CSRF remain untouched.
type providersResponseLost struct {
	header http.Header
	prefix []byte
	length int
}

func (*providersResponseLost) Error() string {
	return "owned Provider response intentionally truncated"
}

type independentBodyKey struct{}

type independentBodyCapture struct {
	mu       sync.Mutex
	source   io.ReadCloser
	bytes    []byte
	overflow bool
}

func (c *independentBodyCapture) Read(p []byte) (int, error) {
	n, err := c.source.Read(p)
	c.mu.Lock()
	defer c.mu.Unlock()
	if n > 0 && !c.overflow {
		if len(c.bytes)+n > 524288 {
			clear(c.bytes)
			c.bytes = nil
			c.overflow = true
		} else {
			c.bytes = append(c.bytes, p[:n]...)
		}
	}
	return n, err
}
func (c *independentBodyCapture) Close() error { return c.source.Close() }
func (c *independentBodyCapture) snapshot() ([]byte, bool) {
	c.mu.Lock()
	defer c.mu.Unlock()
	return append([]byte(nil), c.bytes...), c.overflow
}
func (c *independentBodyCapture) erase() {
	c.mu.Lock()
	defer c.mu.Unlock()
	clear(c.bytes)
	c.bytes = nil
}

func newProvidersWebRoot(t *testing.T, ctx context.Context, own *providersWebFixture) *authenticationWebFixture {
	t.Helper()

	if err != nil {
		t.Fatal("owned frontend listener unavailable")
	}
	t.Cleanup(func() { _ = listener.Close() })
	origin := "http://" + listener.Addr().String()
	inputs := accountenv.New(t)
	objects, err := objectfixture.Environment(ctx, db.Name)
	if err != nil {
		t.Fatal("owned object fixture unavailable")
	}
	values := inputs.Values()
	for _, v := range objects {
		k, value, _ := strings.Cut(v, "=")
		values[k] = value
	}
	for k, v := range map[string]string{
		"DATABASE_URL": db.Fixture.URL(db.Name), "DATABASE_CA_FILE": db.Fixture.CAFile,
		"DATABASE_STARTUP_TIMEOUT": "15s", "HTTP_ADDR": "127.0.0.1:0", "PUBLIC_ORIGIN": origin, "SHUTDOWN_TIMEOUT": "2s",
		"CURSOR_KEYRING": `{"format":1,"current_kid":"test","keys":[{"kid":"test","key_b64":"AAECAwQFBgcICQoLDA0ODxAREhMUFRYXGBkaGxwdHh8="}]}`,

	} {
		values[config.Prefix+k] = v
	}
	var environment []string
	for k, v := range values {
		environment = append(environment, k+"="+v)
	}
	cfg, err := config.Load(func(k string) (string, bool) { v, ok := values[k]; return v, ok }, environment)
	if err != nil {
		t.Fatal("owned root configuration invalid")
	}
	output := &httpFixtureLog{listening: make(chan string, 1)}
	logger, err := logging.New(logging.Central, slog.LevelInfo, output)
	if err != nil {
		t.Fatal(err)
	}
	rootCtx, cancel := context.WithCancel(ctx)
	rootDone := make(chan error, 1)
	go func() { rootDone <- app.Run(rootCtx, cfg, logger, nil) }()
	t.Cleanup(func() {
		cancel()
		select {
		case err := <-rootDone:
			if err != nil {
				t.Error("full Account/Secret/Model root shutdown failed", err)
			}
		case <-time.After(5 * time.Second):
			t.Error("full root did not join within its original shutdown budget")
		}
	})
	var address string
	select {
	case address = <-output.listening:
	case err := <-rootDone:
		rootDone <- err
		t.Fatal("full root initialization failed", err)
	case <-ctx.Done():
		t.Fatal("shared browser budget ended during root startup")
	}
	backend, err := url.Parse("http://" + address)
	if err != nil {
		t.Fatal(err)
	}
	transport := &http.Transport{Proxy: nil}
	proxy := httputil.NewSingleHostReverseProxy(backend)
	proxy.Transport = transport // NewSingleHostReverseProxy preserves the original Host.
	proxy.ModifyResponse = own.controlResponse
	proxy.ErrorHandler = func(w http.ResponseWriter, _ *http.Request, err error) {
		var lost *providersResponseLost
		if !errors.As(err, &lost) {
			http.Error(w, "owned API unavailable", 502)
			return
		}
		defer clear(lost.prefix)
		for name, values := range lost.header {
			w.Header()[name] = append([]string(nil), values...)
		}
		w.Header().Del("Transfer-Encoding")
		w.Header().Set("Content-Length", strconv.Itoa(lost.length))
		w.Header().Set("Connection", "close")
		w.WriteHeader(200)
		_, _ = w.Write(lost.prefix)
		_ = http.NewResponseController(w).Flush()
		if hijacker, ok := w.(http.Hijacker); ok {
			if conn, _, e := hijacker.Hijack(); e == nil {
				_ = conn.Close()
			}
		}
	}
	root, err := filepath.Abs("../../web/dist")
	if err != nil {
		t.Fatal(err)
	}
	if _, err = os.Stat(filepath.Join(root, "index.html")); err != nil {
		t.Fatal("formal production dist must be built before browser execution")
	}
	assets := http.FileServer(http.Dir(root))
	server := &http.Server{ReadHeaderTimeout: 5 * time.Second, Handler: http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if r.Method == "GET" && r.URL.Path == "/api/v1/session" && own.failSession.CompareAndSwap(true, false) {
			own.sessionFailures.Add(1)
			http.Error(w, "owned bounded session read unavailable", 503)
			return
		}
		if r.Method == "GET" && strings.HasPrefix(r.URL.Path, "/api/v1/system/model-providers/") && own.failRead.CompareAndSwap(true, false) {
			own.readFailures.Add(1)
			http.Error(w, "owned bounded configuration read unavailable", 503)
			return
		}

			// Private independent mode only: observe exactly the bytes the
			// reverse proxy actually reads, without prereading or logging them.
			own.mu.Lock()
			capture := own.mode == "independent-replay" && r.URL.Path == own.dropPath && (r.Method == "POST" || r.Method == "PUT")

				observed := &independentBodyCapture{source: r.Body}

				r = r.WithContext(context.WithValue(r.Context(), independentBodyKey{}, observed))
				defer observed.erase()
			}
			proxy.ServeHTTP(w, r)
			return
		}
		if r.Method != http.MethodGet && r.Method != http.MethodHead {
			http.Error(w, "method unavailable", 405)
			return
		}
		clean := filepath.Clean("/" + r.URL.Path)
		if info, e := os.Stat(filepath.Join(root, clean)); e == nil && !info.IsDir() {
			assets.ServeHTTP(w, r)
			return
		}
		if strings.HasPrefix(clean, "/assets/") || filepath.Ext(clean) != "" {
			http.NotFound(w, r)
			return
		}
		w.Header().Set("Cache-Control", "no-store")
		http.ServeFile(w, r, filepath.Join(root, "index.html"))
	})}
	serverDone := make(chan error, 1)
	go func() { serverDone <- server.Serve(listener) }()
	t.Cleanup(func() {
		shutdown, stop := context.WithTimeout(context.Background(), 2*time.Second)
		defer stop()
		if err := server.Shutdown(shutdown); err != nil {
			t.Error("frontend server did not drain")
			_ = server.Close()
		}
		if err := <-serverDone; !errors.Is(err, http.ErrServerClosed) {
			t.Error("owned frontend server failed", err)
		}
		transport.CloseIdleConnections()
	})
	runtimeRoot := os.Getenv("AGENTEAM_AUTH_WEB_RUNTIME")
	if !filepath.IsAbs(runtimeRoot) || len(runtimeRoot) > 45 {
		t.Fatal("explicit short task-owned browser runtime is required")
	}
	directory, err := os.MkdirTemp(runtimeRoot, "run-")
	if err != nil {
		t.Fatal("private browser runtime creation failed")
	}
	if err := os.Chmod(directory, 0700); err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() {
		if err := os.RemoveAll(directory); err != nil {
			t.Error("private browser runtime cleanup failed")
		}
		if _, err := os.Stat(directory); !errors.Is(err, os.ErrNotExist) {
			t.Error("private browser runtime still exists")
		}
	})
	old := &httpFixture{t: t, db: db, config: cfg, address: address, log: output}
	entry := old.record("bootstrap", "admin@mail.com", "")
	material, err := json.Marshal(map[string]string{"email": entry.Email, "password": entry.Password})
	if err != nil {
		t.Fatal("private bootstrap encoding failed")
	}
	defer clear(material)
	if err := os.WriteFile(filepath.Join(directory, "credentials.json"), material, 0600); err != nil {
		t.Fatal("private bootstrap transfer failed")
	}
	f := &authenticationWebFixture{t: t, db: db, origin: origin, directory: directory, webRoot: root, entry: entry, log: output, record: old.record, recoveryLogPath: cfg.AccountRecoveryLog()}
	t.Cleanup(func() {
		if output.contains(entry.Password) {
			t.Error("bootstrap material escaped the restricted recovery log")
		}
	})
	conn, closeConnection := invitationsWebConnect(t, ctx, db)
	defer closeConnection()
	var configured bool
	if err := conn.QueryRow(ctx, `SELECT configured FROM agenteam_model.platform_selection WHERE singleton`).Scan(&configured); err != nil || configured {
		t.Fatal("real Model Initialize technical singleton missing", err)
	}
	return f
}

func providerWebInput(name, endpoint string, ref any) map[string]any {
	return map[string]any{"name": name, "protocol": "openai-chat-completions", "base_url": endpoint, "enabled": true, "credential_ref": ref, "options": map[string]any{}}
}
func (f *providersWebFixture) request(ctx context.Context, method, path string, body any, status int) map[string]any {
	f.t.Helper()
	raw, err := json.Marshal(body)
	if err != nil {
		f.t.Fatal("owned Provider setup encoding failed")
	}
	defer clear(raw)
	var reader io.Reader
	if body != nil {
		reader = bytes.NewReader(raw)
	}
	r, err := http.NewRequestWithContext(ctx, method, f.origin+path, reader)
	if err != nil {
		f.t.Fatal("owned Provider setup request invalid")
	}
	r.Header.Set("Origin", f.origin)
	r.Header.Set("Sec-Fetch-Site", "same-origin")
	if body != nil {
		r.Header.Set("Content-Type", "application/json")
		r.Header.Set("Idempotency-Key", id[struct{}](f.t).String())
		r.Header.Set("X-CSRF-Token", f.csrf)
	}
	response, err := f.adminClient.Do(r)
	if err != nil {
		f.t.Fatal("formal Provider setup HTTP failed", err)
	}
	defer response.Body.Close()
	if response.StatusCode != status {
		f.t.Fatalf("formal Provider setup %s status=%d want=%d", method, response.StatusCode, status)
	}
	value := map[string]any{}
	if json.NewDecoder(io.LimitReader(response.Body, 2097153)).Decode(&value) != nil {
		f.t.Fatal("formal Provider setup response invalid")
	}
	return value
}

func (f *providersWebFixture) controlResponse(response *http.Response) error {
	r := response.Request
	if r == nil || response.StatusCode != 200 || (r.Method != "POST" && r.Method != "PUT" && r.Method != "DELETE") || !strings.HasPrefix(r.URL.Path, "/api/v1/system/model-") || strings.HasSuffix(r.URL.Path, "/lookup") {
		return nil
	}
	f.mu.Lock()
	defer f.mu.Unlock()
	if f.lostKey != "" && r.Header.Get("Idempotency-Key") == f.lostKey {
		f.replayed++
		if f.mode == "independent-replay" {
			observed, ok := r.Context().Value(independentBodyKey{}).(*independentBodyCapture)
			if ok {
				captured, overflow := observed.snapshot()
				f.independentSameBody = !overflow && f.independentCaptured && bytes.Equal(f.independentOriginalBody, captured)
				clear(captured)
			}
			f.independentSameCSRF = f.independentOriginalCSRF != "" && r.Header.Get("X-CSRF-Token") == f.independentOriginalCSRF
		}
	}
	if f.mode == "independent-replay" && f.lostKey != "" && f.dropPath == "/api/v1/system/model-credentials" && r.URL.Path == "/api/v1/system/model-providers" && r.Method == "POST" {
		f.independentSeparateKeys = r.Header.Get("Idempotency-Key") != "" && r.Header.Get("Idempotency-Key") != f.lostKey
	}
	if !f.dropNext || r.URL.Path != f.dropPath {
		return nil
	}
	f.dropNext = false
	raw, err := io.ReadAll(io.LimitReader(response.Body, 600001))
	_ = response.Body.Close()
	defer clear(raw)
	var receipt map[string]any
	if err != nil || len(raw) > 600000 || json.Unmarshal(raw, &receipt) != nil {
		return errors.New("owned response control requires complete formal receipt")
	}
	target, ok := receipt["resource_id"].(string)
	if !ok {
		target, ok = receipt["credential_id"].(string)
	}
	if !ok || target == "" {
		return errors.New("owned response control has no formal target")
	}
	f.lostID = target
	f.lostKey = r.Header.Get("Idempotency-Key")
	if f.mode == "independent-replay" {
		observed, ok := r.Context().Value(independentBodyKey{}).(*independentBodyCapture)
		if !ok {
			return errors.New("independent accepted request capture unavailable")
		}
		captured, overflow := observed.snapshot()
		defer clear(captured)
		if overflow || len(captured) == 0 || f.lostKey == "" || r.Header.Get("X-CSRF-Token") == "" {
			return errors.New("independent accepted request capture incomplete")
		}
		clear(f.independentOriginalBody)
		f.independentOriginalBody = append([]byte(nil), captured...)
		f.independentOriginalCSRF = r.Header.Get("X-CSRF-Token")
		f.independentCaptured = true
	}
	f.dropped++
	return &providersResponseLost{header: response.Header.Clone(), prefix: append([]byte(nil), raw[:len(raw)/2]...), length: len(raw)}
}
func (f *providersWebFixture) snapshot(ctx context.Context) []map[string]any {
	f.t.Helper()
	conn, closeConnection := invitationsWebConnect(f.t, ctx, f.db)
	defer closeConnection()
	rows, err := conn.Query(ctx, `SELECT id::text,name,protocol,base_url,enabled,credential_id::text,version::text,created_at,updated_at FROM agenteam_model.providers WHERE scope='system' ORDER BY created_at DESC,id DESC`)
	if err != nil {
		f.t.Fatal(err)
	}
	defer rows.Close()
	out := []map[string]any{}
	for rows.Next() {
		var target, name, protocol, endpoint, version string
		var enabled bool
		var ref *string
		var created, updated time.Time
		if err = rows.Scan(&target, &name, &protocol, &endpoint, &enabled, &ref, &version, &created, &updated); err != nil {
			f.t.Fatal(err)
		}
		out = append(out, map[string]any{"id": target, "input": map[string]any{"name": name, "protocol": protocol, "base_url": endpoint, "enabled": enabled, "credential_ref": ref, "options": map[string]any{}}, "version": version, "created_at": created.UTC().Format("2006-01-02T15:04:05.000000Z"), "updated_at": updated.UTC().Format("2006-01-02T15:04:05.000000Z")})
	}
	if err = rows.Err(); err != nil {
		f.t.Fatal(err)
	}
	return out
}

func (f *providersWebFixture) ipc(ctx context.Context, r providersWebIPC) map[string]any {
	f.t.Helper()
	out := map[string]any{"ok": true, "sequence": r.Sequence}
	if r.ID != "" {
		if _, err := foundation.ParseID[struct{}](r.ID); err != nil {
			f.t.Fatal("owned Provider target invalid")
		}
	}
	conn, closeConnection := invitationsWebConnect(f.t, ctx, f.db)
	defer closeConnection()
	switch r.Action {
	case "snapshot":
		out["items"] = f.snapshot(ctx)
	case "provider-missing":
		f.request(ctx, "GET", "/api/v1/system/model-providers/"+r.ID, nil, 404)
		out["missing"] = true
	case "facts":
