import { test, expect, type Page } from "@playwright/test";
import { readFileSync, writeFileSync, renameSync, existsSync } from "node:fs";
import { join } from "node:path";
const directory = process.env.AGENTEAM_AUTH_WEB_PRIVATE!;
type Credential = { email: string; password: string; user_id: string };
type Provider = {
  id: string;
  input: {
    name: string;
    protocol: string;
    base_url: string;
    enabled: boolean;
    credential_ref: string | null;
    options: object;
  };
  version: string;
  created_at: string;
  updated_at: string;
};
type Receipt = {
  kind: string;
  resource_id: string;
  version: string;
  affected_references: string;
};
type Fact = {
  path: string;
  method: string;
  status: number;
  bytes: number;
  ended: boolean;
  code?: string;
  receipt?: Receipt;
  credential?: { credential_id: string; version: string };
  page?: { items: Provider[]; next_cursor: string | null };
  provider?: Provider;
  session?: { user_id: string; session_id: string; role: string };
  found?: boolean;
};
type ObservedWindow = Window & { __providerFacts?: Fact[] };
const material = (): {
  admin: Credential;
  member: Credential;
  value: string;
  target_id: string;
  shared_id: string;
  old_ref: string;
  expected: Provider[];
} =>
  JSON.parse(readFileSync(join(directory, "providers-material.json"), "utf8"));
function result(value: Record<string, unknown>) {
  writeFileSync(
    join(directory, "providers-result.json"),
    JSON.stringify({ completed: true, ...value }),
    { mode: 0o600 },
  );
}
let sequence = 0;
async function ipc(
  action: string,
  fields: Record<string, string> = {},
): Promise<Record<string, any>> {
  const next = ++sequence,
    request = join(directory, "providers-ipc.json"),
    ack = join(directory, `providers-ack-${next}.json`);
  writeFileSync(
    request + ".tmp",
    JSON.stringify({ sequence: next, action, ...fields }),
    { mode: 0o600 },
  );
  renameSync(request + ".tmp", request);
  await expect.poll(() => existsSync(ack), { timeout: 14_000 }).toBe(true);
  const value = JSON.parse(readFileSync(ack, "utf8"));
  expect(value.ok === true && value.sequence === next).toBe(true);
  return value;
}
// Observe bounded safe projections from the application's original native
// reader. No body clone/tee, no request body/header collection, no abort delay.
// Actual joining is covered by the transport/owner barrier tests, not this hook.
function observeNative() {
  const scope = window as ObservedWindow;
  scope.__providerFacts = [];
  const fetch = window.fetch.bind(window);
  window.fetch = (input, init) => {
    const path =
      typeof input === "string"
        ? input
        : input instanceof URL
          ? input.pathname + input.search
          : new URL(input.url).pathname;
    const watched =
      path.startsWith("/api/v1/system/model") || path === "/api/v1/session";
    return fetch(input, init).then((response) => {
      if (!watched) return response;
      const fact: Fact = {
        path,
        method: init?.method ?? "GET",
        status: response.status,
        bytes: 0,
        ended: false,
      };
      scope.__providerFacts!.push(fact);
      const body = response.body;
      if (!body) {
        fact.ended = true;
        return response;
      }
      const get = body.getReader.bind(body);
      let text = "";
      const decoder = new TextDecoder();
      Object.defineProperty(body, "getReader", {
        value: () => {
          const reader = get(),
            read = reader.read.bind(reader);
          Object.defineProperty(reader, "read", {
            value: () => {
              const original = read();
              void original.then(
                ({ done, value }) => {
                  if (value) {
                    fact.bytes += value.byteLength;
                    if (fact.bytes <= 2097152)
                      text += decoder.decode(value, { stream: true });
                    else text = "";
                  }
                  if (!done) return;
                  fact.ended = true;
                  try {
                    const value = JSON.parse(text + decoder.decode());
                    if (path === "/api/v1/session") {
                      if (response.status === 200)
                        fact.session = {
                          user_id: value.user.id,
                          session_id: value.session.id,
                          role: value.user.role,
                        };
                    } else if (response.status !== 200) {
                      if (typeof value.code === "string")
                        fact.code = value.code;
                    } else if (Array.isArray(value.items))
                      fact.page = {
                        items: value.items,
                        next_cursor: value.next_cursor,
                      };
                    else if (
                      typeof value.kind === "string" &&
                      typeof value.resource_id === "string"
                    )
                      fact.receipt = {
                        kind: value.kind,
                        resource_id: value.resource_id,
                        version: value.version,
                        affected_references: value.affected_references,
                      };
                    else if (typeof value.credential_id === "string")
                      fact.credential = {
                        credential_id: value.credential_id,
                        version: value.version,
                      };
                    else if (typeof value.id === "string" && value.input)
                      fact.provider = value;
                    else if (typeof value.found === "boolean")
                      fact.found = value.found;
                    else if (typeof value.observed === "boolean")
                      fact.found = value.observed;
                  } catch {
                    /* A lost receipt is intentionally not promoted to success. */
                  } finally {
                    text = "";
                  }
                },
                () => {
                  text = "";
                  fact.ended = true;
                },
              );
              return original;
            },
          });
          return reader;
        },
      });
      return response;
    });
  };
}
async function facts(page: Page) {
  return page.evaluate(() => (window as ObservedWindow).__providerFacts ?? []);
}
function canonicalProviders(rows: Provider[]) {
  return JSON.stringify(
    rows.map((p) => [
      p.id,
      p.input.name,
      p.input.protocol,
      p.input.base_url,
      p.input.enabled,
      p.input.credential_ref,
      Object.keys(p.input.options).length,
      p.version,
      p.created_at,
      p.updated_at,
    ]),
  );
}
async function latestProvider(page: Page) {
  const rows = await facts(page);
  const value = rows.filter((f) => f.provider).at(-1)?.provider;
  expect(!!value).toBe(true);
  return value!;
}
async function latestReceipt(page: Page, kind?: string) {
  const rows = await facts(page);
  const value = rows
    .filter((f) => f.receipt && (!kind || f.receipt.kind === kind))
    .at(-1)?.receipt;
  expect(!!value).toBe(true);
  return value!;
}
async function writes(page: Page) {
  return (await facts(page)).filter(
    (f) => f.method !== "GET" && !f.path.endsWith("/lookup"),
  );
}
async function setup(
  page: Page,
  who: "admin" | "member" = "admin",
  path = "/system/providers",
) {
  await page.addInitScript(observeNative);
  await page.goto(path);
  await page.getByLabel("邮箱", { exact: true }).fill(material()[who].email);
  await page.getByLabel("密码", { exact: true }).fill(material()[who].password);
  await page.getByRole("button", { name: "登录", exact: true }).click();
  await expect(page).toHaveURL(new RegExp(path + "$"));
  if (who === "admin")
    await expect(
      page.getByRole("heading", {
        name: path.endsWith("users") ? "用户" : "Providers",
        exact: true,
      }),
    ).toBeVisible();
}
const top = (page: Page) => page.getByRole("dialog").last();

    .getByRole("button", { name: /^(创建 Provider|操作已确认)$/ })

}

  return ipc("facts");
}

  page,
}) => {
  await setup(page);

    page,

  expect(

test("[credential] new and replacement credentials, preserved shared ref, partial success and reviewed rebase", async ({
  page,
}) => {
  await setup(page);
  const data = material();
  const baseline = await ipc("facts");
  await openCreate(page);
  await fillCreate(page, "Credential created Provider");
  await top(page)
    .getByLabel("新凭据（仅写入）", { exact: true })
    .fill(data.value);
  await save(page);
  await confirmed(page);
  const createdProvider = await latestProvider(page);
  expect(
    !!createdProvider.input.credential_ref &&
      createdProvider.input.credential_ref !== data.old_ref,
  ).toBe(true);
  await select(page, "Owned primary");
  await edit(page);
  await top(page)
    .getByLabel("新凭据（仅写入）", { exact: true })
    .fill(data.value);
  await save(page);
  await confirmed(page);
  let current = await latestProvider(page);
  const replacedRef = current.input.credential_ref!;
  expect(
    replacedRef !== data.old_ref &&
      replacedRef !== createdProvider.input.credential_ref,
  ).toBe(true);
  let observed = await ipc("snapshot");
  expect(
    observed.items.find((p: Provider) => p.id === data.shared_id).input
      .credential_ref,
  ).toBe(data.old_ref);
  const old = await ipc("facts", { id: data.old_ref });
  expect(old.credential_exists && old.references === 1).toBe(true);
  await edit(page);
  await ipc("compete", { id: current.id });
  await top(page)
    .getByLabel("新凭据（仅写入）", { exact: true })
    .fill(data.value);
  await save(page);
  await expect(top(page)).toContainText("新凭据已创建，Provider尚未确认保存");
  expect(
    (await top(page)
      .getByLabel("新凭据（仅写入）", { exact: true })
      .inputValue()) === "",
  ).toBe(true);
  const partial = await ipc("facts");
  expect(partial.credential_creates === baseline.credential_creates + 3).toBe(
    true,
  );
  await top(page)
    .getByRole("button", { name: "读取当前配置并核对", exact: true })
    .click();
  await expect(top(page)).toContainText("不会再次创建凭据");
  await save(page);
  await confirmed(page);
  current = await latestProvider(page);
  expect(
    current.input.credential_ref !== replacedRef && current.version === "4",
  ).toBe(true);
  expect((await ipc("facts")).credential_creates).toBe(
    partial.credential_creates,
  );
  await page
    .getByRole("button", { name: "删除 Provider", exact: true })
    .click();
  await top(page)
    .getByRole("button", { name: "确认删除", exact: true })
    .click();
  await expect(
    page.getByText("Provider 已删除，凭据保持不变。", { exact: true }),
  ).toBeVisible();
  for (const ref of [data.old_ref, replacedRef, current.input.credential_ref!])
    expect((await ipc("facts", { id: ref })).credential_exists).toBe(true);
  expect((await ipc("facts")).credentials).toBe(partial.credentials);
  result({
    created: true,
    replaced: true,
    shared_preserved: true,
    partial: true,
    rebase: true,
    credential_retained: true,
  });
});

test("[outcome] actual accepted response loss, bounded historical lookup and explicit exact Execute recovery", async ({
  page,
}) => {
  await setup(page);
  const data = material();
  for (const stage of ["credential", "provider"] as const) {
    await openCreate(page);
    await fillCreate(page, `Recovery ${stage}`);
    if (stage === "credential")
      await top(page)
        .getByLabel("新凭据（仅写入）", { exact: true })
        .fill(data.value);
    await ipc("arm-drop", { stage });
    await save(page);
    await expect(top(page)).toContainText("本次结果尚未确认");
    const lost = await ipc("drop-facts");
    expect(
      lost.dropped === 1 && lost.replayed === 0 && lost.armed === false,
    ).toBe(true);
    const before = await ipc("facts");
    const writeCount = (await writes(page)).length;
    await top(page)
      .getByRole("button", { name: "检查原请求", exact: true })
      .click();
    await expect(top(page)).toContainText("已观察到历史回执");
    expect((await writes(page)).length).toBe(writeCount);
    expect((await ipc("drop-facts")).replayed).toBe(0);
    if (stage === "credential")
      expect((await ipc("facts")).providers).toBe(before.providers);
    await top(page)
      .getByRole("button", { name: "重试原请求", exact: true })
      .click();
    await confirmed(page);
    const replay = await ipc("drop-facts");
    expect(
      replay.dropped === 1 &&
        replay.replayed === 1 &&
        replay.target === lost.target,
    ).toBe(true);
    const after = await ipc("facts");
    expect(after.credential_creates).toBe(before.credential_creates);
    expect(after.providers).toBe(
      before.providers + (stage === "credential" ? 1 : 0),
    );
  }
  await openCreate(page);
  await fillCreate(page, "Confirmed read failure");
  await ipc("arm-get-fail");
  await save(page);
  await expect(
    page.getByText("操作已确认，当前配置读取失败。请仅重读当前配置。", {
      exact: true,
    }),
  ).toBeVisible();
  const count = (await writes(page)).length;
  await page
    .getByRole("button", { name: "重新读取当前配置", exact: true })
    .click();
  await expect(
    page.getByRole("button", { name: "编辑 Provider", exact: true }),
  ).toBeEnabled();
  expect((await writes(page)).length).toBe(count);
  await openCreate(page);
  await fillCreate(page, "Abandoned accepted Provider");
  await ipc("arm-drop", { stage: "provider" });
  await save(page);
  await expect(top(page)).toContainText("本次结果尚未确认");
  await top(page)
    .getByRole("button", { name: "放弃本次操作", exact: true })
    .click();
  await top(page)
    .getByRole("button", { name: "放弃修改", exact: true })
    .click();
  await page.getByRole("button", { name: "刷新", exact: true }).click();
  await expect(
    page.getByRole("button", {
      name: "查看 Abandoned accepted Provider",
      exact: true,
    }),
  ).toBeVisible();
  expect(
    (await writes(page)).some(
      (f) => f.method === "DELETE" && f.path.includes("model-credentials"),
    ),
  ).toBe(false);
  result({
    credential_replayed: true,
    provider_replayed: true,
    historical_only: true,
    abandoned: true,
    confirmed_read_failure: true,
  });
});

async function checkingConfirmation(page: Page, fail: boolean) {
  const originalWrites = (await writes(page)).length;
  if (fail) await ipc("arm-session-fail");
  // Explicit synthetic App lifecycle trigger; subsequent Session is real backend.
  await page.evaluate(() => window.dispatchEvent(new Event("pageshow")));
  if (fail) {
    await expect(page.getByRole("dialog")).toHaveCount(0);
    const restore = page.getByRole("button", {
      name: "检查当前会话",
      exact: true,
    });
    await expect(restore).toBeEnabled();
    expect(
      await restore.evaluate((node) => node.closest("[inert]") === null),
    ).toBe(true);
    await restore.click();
  }
  await expect(page.getByRole("dialog")).toHaveCount(2);
  await expect(top(page)).toHaveAccessibleName("放弃未保存修改？");
  expect(
    await top(page).evaluate(
      (node) =>
        node instanceof HTMLElement &&
        !node.inert &&
        node.getAttribute("aria-hidden") !== "true",
    ),
  ).toBe(true);
  await top(page)
    .getByRole("button", { name: "继续编辑", exact: true })
    .click();
  await expect(page.getByRole("dialog")).toHaveCount(1);
  await focusContained(page);
  expect((await writes(page)).length).toBe(originalWrites);
}
