import os,sys,json,time,subprocess,hashlib,pathlib,signal
r=pathlib.Path(__file__).resolve().parent
name=sys.argv[1]
env=os.environ.copy()
if name=="go01":
 argv=["/workspace/toolchains/go1.27.1/bin/go","run",str(r/"go-url-probe.go")]
 env.update({"GO111MODULE":"off","GOTOOLCHAIN":"local","GOPROXY":"off","GOSUMDB":"off","GOCACHE":str(r/"gocache"),"GOTMPDIR":str(r/"gotmp")})
 cwd=r
 (r/"gotmp").mkdir(exist_ok=True)
 data=(r/"url-cases.json").read_bytes()
else:
 argv=["node","/workspace/agenteam/web/node_modules/vitest/vitest.mjs","run","--config","vitest.config.mjs",sys.argv[2],"--reporter=verbose"]
 env.update({"CI":"1","NO_COLOR":"1","FORCE_COLOR":"0"})
 cwd=r/"candidate/web"
 data=None
log=r/"logs"/name;log.mkdir(exist_ok=False)
files=[p for p in (r/"candidate").rglob("*") if p.is_file() and "node_modules" not in p.parts and ".vite" not in p.parts]
before={str(p.relative_to(r)):hashlib.sha256(p.read_bytes()).hexdigest() for p in files}
meta={"argv":argv,"cwd":str(cwd),"env_overrides":{k:env[k] for k in ["GO111MODULE","GOTOOLCHAIN","GOPROXY","GOSUMDB","GOCACHE","GOTMPDIR"] if name=="go01" and k in env} if name=="go01" else {k:env[k] for k in ["CI","NO_COLOR","FORCE_COLOR"]},"input_sha256":hashlib.sha256((r/"input.json").read_bytes()).hexdigest(),"before":before,"limit_seconds":120 if name=="go01" else 45}
(log/"command.json").write_text(json.dumps(meta,indent=2)+"\n")
t=time.monotonic()
with (log/"raw.log").open("wb") as out:
 p=subprocess.Popen(argv,cwd=cwd,env=env,stdin=subprocess.PIPE if data is not None else subprocess.DEVNULL,stdout=out,stderr=subprocess.STDOUT,start_new_session=True)
 try:
  p.communicate(data,timeout=meta["limit_seconds"])
 except subprocess.TimeoutExpired:
  os.killpg(p.pid,signal.SIGTERM)
  try:p.wait(timeout=5)
  except subprocess.TimeoutExpired:os.killpg(p.pid,signal.SIGKILL);p.wait()
  meta["timed_out"]=True
meta.update({"pid":p.pid,"exit_code":p.returncode,"elapsed_seconds":round(time.monotonic()-t,3),"after":{str(x.relative_to(r)):hashlib.sha256(x.read_bytes()).hexdigest() for x in files},"raw_sha256":hashlib.sha256((log/"raw.log").read_bytes()).hexdigest(),"actual_wait":True})
meta["inputs_unchanged"]=meta["before"]==meta["after"]
(log/"result.json").write_text(json.dumps(meta,indent=2)+"\n")
print(json.dumps({k:meta[k] for k in ["pid","exit_code","elapsed_seconds","inputs_unchanged","actual_wait","raw_sha256"]}))
print((log/"raw.log").read_text())
sys.exit(p.returncode)
