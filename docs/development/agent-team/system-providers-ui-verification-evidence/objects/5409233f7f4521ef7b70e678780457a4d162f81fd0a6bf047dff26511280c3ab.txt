import { describe, expect, it, vi } from 'vitest'
import { accountTransport, type Fetch } from '../api/client'
import {
  captureProviderCommand,
  createSystemProviderAPI,
  validateCredentialValue,
  type ProviderInput,
} from '../api/system-providers'

const id = (n: number) => '01900000-0000-7000-8000-' + n.toString(16).padStart(12, '0')
const time = '2026-10-06T12:34:56.123456Z'
const input = (): ProviderInput => ({
  name: 'Provider',
  protocol: 'openai-chat-completions',
  base_url: 'https://provider.example/v1',
  enabled: false,
  credential_ref: null,
  options: {},
})
const row = (n = 100) => ({
  id: id(n),
  input: input(),
  version: '9007199254740993',
  created_at: time,
  updated_at: time,
})
const model = (n = 100) => ({
  id: id(n),
  provider_id: id(1),
  version: '1',
  created_at: time,
  updated_at: time,
  input: {
    name: 'Model',
    provider_model_id: 'native',
    type: 'chat',
    enabled: true,
    parameters: {},
    request_overwrite: {},
    header_overwrite: {},
    capabilities: {
      tool_calls: true,
      parallel_tool_calls: true,
      streaming: true,
      reasoning: false,
      input_modalities: ['text'],
      output_modalities: ['text'],
      reasoning_efforts: [],
      structured_output_modes: ['text'],
      context_length: '9007199254740993',
      max_output: '9007199254740992',
    },
  },
})
const json = (value: unknown, status = 200) =>
  new Response(JSON.stringify(value), { status, headers: { 'Content-Type': 'application/json' } })
const signal = () => new AbortController().signal
const options = () => ({ csrfToken: 'S'.repeat(43), key: 'provider/command:1', signal: signal() })
const receipt = (kind = 'provider.create', target = id(1), version = '1') => ({
  kind,
  resource_id: target,
  version,
  affected_references: '0',
})
const credential = () => ({ credential_id: id(2), purpose: 'model', version: '1', deleted: false })
const bytes = (value: unknown) => new TextEncoder().encode(JSON.stringify(value)).byteLength
const parsePage = (value: unknown) =>
  createSystemProviderAPI(async () => json(value)).listProviders({}, signal())

describe('System Provider closed transport and strict contract', () => {
  it('uses exactly ten fixed paths, methods, target/query shapes and original command headers', async () => {
    const fetch = vi.fn<Fetch>(async (path, init) => {
      if (path.includes('credential-commands'))
        return json({ observed: true, result: credential() })
      if (path.includes('model-commands')) return json({ found: true, receipt: receipt() })
      if (path.includes('model-credentials'))
        return json(
          init.method === 'GET'
            ? { credential_id: id(2), purpose: 'model', version: '9007199254740993' }
            : credential(),
        )
      if (path.includes('/system/models?')) return json({ items: [model()], next_cursor: null })
      if (init.method === 'GET')
        return json(path.includes('?') ? { items: [], next_cursor: null } : row(1))
      return json(
        receipt(
          init.method === 'PUT'
            ? 'provider.update'
            : init.method === 'DELETE'
              ? 'provider.delete'
              : 'provider.create',
          id(1),
          init.method === 'POST' ? '1' : '9007199254740994',
        ),
      )
    })
    const api = createSystemProviderAPI(fetch),
      write = options(),
      cursor = '../?limit=99&空=#+'
    await api.listProviders({ cursor }, write.signal)
    await api.getProvider(id(1), write.signal)
    await api.createProvider(input(), write)
    await api.updateProvider(id(1), '9007199254740993', input(), write)
    await api.deleteProvider(id(1), '9007199254740993', write)
    await api.listModels(
      { provider_id: id(1), protocol: 'openai-chat-completions', cursor },
      write.signal,
    )
    await api.createCredential('generated-only', write)
    await api.getCredentialMetadata(id(2), write.signal)
    await api.lookupProviderCommand({ kind: 'provider.create', input: input() }, write)
    await api.lookupCredentialCreate(write)
    expect(fetch.mock.calls.map(([path, init]) => [path.split('?')[0], init.method])).toEqual([
      ['/api/v1/system/model-providers', 'GET'],
      [`/api/v1/system/model-providers/${id(1)}`, 'GET'],
      ['/api/v1/system/model-providers', 'POST'],
      [`/api/v1/system/model-providers/${id(1)}`, 'PUT'],
      [`/api/v1/system/model-providers/${id(1)}`, 'DELETE'],
      ['/api/v1/system/models', 'GET'],
      ['/api/v1/system/model-credentials', 'POST'],
      [`/api/v1/system/model-credentials/${id(2)}`, 'GET'],
      ['/api/v1/system/model-commands/lookup', 'POST'],
      ['/api/v1/system/model-credential-commands/lookup', 'POST'],
    ])
    for (const [path, init] of fetch.mock.calls) {
      expect(init).toMatchObject({
        credentials: 'same-origin',
        cache: 'no-store',
        redirect: 'error',
        signal: write.signal,
      })
      const headers = new Headers(init.headers)
      if (init.method === 'GET') {
        expect(init.body).toBeUndefined()
        expect(headers.has('X-CSRF-Token')).toBe(false)
        expect(headers.has('Idempotency-Key')).toBe(false)
      } else {
        expect(headers.get('X-CSRF-Token') === write.csrfToken).toBe(true)
        expect(headers.get('Idempotency-Key') === write.key).toBe(true)
      }
      if (path.includes('?')) {
        const params = [...new URL(path, 'http://localhost').searchParams]
        expect(params).toEqual(
          path.includes('/models?')
            ? [
                ['provider_id', id(1)],
                ['limit', '25'],
                ['cursor', cursor],
              ]
            : [
                ['limit', '25'],
                ['cursor', cursor],
              ],
        )
      }
    }
    expect(JSON.parse(fetch.mock.calls[8]![1].body as string)).toEqual({
      command: 'provider.create',
    })
    expect(JSON.parse(fetch.mock.calls[9]![1].body as string)).toEqual({ kind: 'create' })
  })

  it('closes every endpoint options object and forbids parameters on old endpoints', async () => {
    const fetch = vi.fn<Fetch>(),
      request = accountTransport(fetch) as (
        endpoint: string,
        parse: (value: unknown) => unknown,
        options: unknown,
      ) => Promise<unknown>
    const common = { signal: signal(), body: {}, csrf: 'S'.repeat(43), key: 'stable' }
    const cases: [string, object][] = [
      ['listProviders', { signal: signal(), providers: {} }],
      ['getProvider', { signal: signal(), target: id(1) }],
      ['createProvider', common],
      ['updateProvider', { ...common, target: id(1) }],
      ['deleteProvider', { ...common, target: id(1) }],
      ['listProviderModels', { signal: signal(), models: { provider_id: id(1) } }],
      ['createModelCredential', common],
      ['getModelCredentialMetadata', { signal: signal(), target: id(2) }],
      ['lookupProviderCommand', common],
      ['lookupModelCredentialCreate', common],
    ]
    for (const [endpoint, valid] of cases) {
      for (const extra of [
        { arbitrary: true },
        { limit: 3_000_000 },
        { avatar: {} },
        { users: {} },
      ])
        await expect(request(endpoint, (v) => v, { ...valid, ...extra })).rejects.toMatchObject({
          kind: 'invalid-input',
        })
      for (const missing of Object.keys(valid)) {
        const invalid = { ...valid } as Record<string, unknown>
        delete invalid[missing]
        await expect(request(endpoint, (v) => v, invalid)).rejects.toMatchObject({
          kind: 'invalid-input',
        })
      }
    }
    for (const providers of [
      { cursor: '' },
      { cursor: undefined },
      { cursor: 'x'.repeat(8193) },
      { limit: 25 },
      { url: '/other' },
    ])
      await expect(
        request('listProviders', (v) => v, { signal: signal(), providers }),
      ).rejects.toMatchObject({ kind: 'invalid-input' })
    for (const endpoint of ['profile', 'systemUsers', 'systemInvitations'])
      await expect(
        request(endpoint, (v) => v, { signal: signal(), providers: {} }),
      ).rejects.toMatchObject({ kind: 'invalid-input' })
    await expect(request('https://external.example', (v) => v, common)).rejects.toMatchObject({
      kind: 'invalid-input',
    })
    expect(fetch).not.toHaveBeenCalled()
  })

  it('validates the entire immutable Provider page, calendar, precise order and required null cursor', async () => {
    const value = { items: [row()], next_cursor: null }
    const parsed = await parsePage(value)
    expect(parsed.items[0]!.version).toBe('9007199254740993')
    expect(
      Object.isFrozen(parsed) &&
        Object.isFrozen(parsed.items) &&
        Object.isFrozen(parsed.items[0]!.input),
    ).toBe(true)
    const invalid = [
      { items: [] },
      { ...value, extra: 1 },
      { ...value, next_cursor: '' },
      { ...value, next_cursor: 'cursor' },
      { items: [row(), row()], next_cursor: null },
      { items: [row(1), row(2)], next_cursor: null },
      { items: Array.from({ length: 26 }, (_, n) => row(100 - n)), next_cursor: null },
      ...[0, '01', '0', '9223372036854775808', null].map((version) => ({
        items: [{ ...row(), version }],
        next_cursor: null,
      })),
      ...[
        '2026-02-30T12:34:56.123456Z',
        '2026-10-06T12:34:56.123Z',
        '2026-10-06T12:34:56.123456+00:00',
      ].map((created_at) => ({ items: [{ ...row(), created_at }], next_cursor: null })),
      { items: [{ ...row(), updated_at: '2026-10-06T12:34:56.123455Z' }], next_cursor: null },
      ...[
        { ...input(), options: { arbitrary: 1 } },
        { ...input(), credential_ref: undefined },
        { ...input(), protocol: 'other' },
        { ...input(), enabled: 1 },
        { ...input(), extra: true },
      ].map((input) => ({ items: [{ ...row(), input }], next_cursor: null })),
    ]
    for (const candidate of invalid)
      await expect(parsePage(candidate)).rejects.toMatchObject({ kind: 'invalid-response' })
    const full = Array.from({ length: 25 }, (_, n) => row(100 - n))
    expect((await parsePage({ items: full, next_cursor: 'opaque' })).items).toHaveLength(25)
    await expect(
      createSystemProviderAPI(async () => json(row(3))).getProvider(id(2), signal()),
    ).rejects.toMatchObject({ kind: 'invalid-response' })
  })

  it('parses all Model fields and provider/protocol relationships without silently dropping a bad row', async () => {
    const parse = (value: unknown, protocol = 'openai-chat-completions') =>
      createSystemProviderAPI(async () => json({ items: [value], next_cursor: null })).listModels(
        { provider_id: id(1), protocol: protocol as never },
        signal(),
      )
    expect((await parse(model())).items[0]!.input.capabilities.context_length).toBe(
      '9007199254740993',
    )
    for (const changes of [
      { provider_id: id(2) },
      { extra: true },
      { input: { ...model().input, parameters: { temperature: 1 } } },
      { input: { ...model().input, request_overwrite: { messages: [] } } },
      { input: { ...model().input, header_overwrite: { x: 'a' } } },
      { input: { ...model().input, type: 'embedding' } },
      { input: { ...model().input, provider_model_id: '界'.repeat(86) } },
    ])
      await expect(parse({ ...model(), ...changes })).rejects.toMatchObject({
        kind: 'invalid-response',
      })
    for (const changes of [
      { tool_calls: false },
      { reasoning_efforts: ['low'] },
      { input_modalities: ['text', 'text'] },
      { output_modalities: ['vector'] },
      { context_length: undefined },
      { max_output: '9007199254740994' },
      { max_output: 10 },
      { extra: 1 },
    ])
      await expect(
        parse({
          ...model(),
          input: { ...model().input, capabilities: { ...model().input.capabilities, ...changes } },
        }),
      ).rejects.toMatchObject({ kind: 'invalid-response' })
    const cap = { ...model().input.capabilities, structured_output_modes: ['json_schema'] }
    await expect(
      parse({ ...model(), input: { ...model().input, capabilities: cap } }, 'anthropic-messages'),
    ).rejects.toMatchObject({ kind: 'invalid-response' })
  })

  it('checks exact receipts, lookup nullability and metadata independently of current versions', async () => {
    const command = {
      kind: 'provider.update' as const,
      id: id(1),
      expected_version: '9007199254740993',
      input: input(),
    }
    const parse = (value: unknown) =>
      createSystemProviderAPI(async () => json(value)).lookupProviderCommand(command, options())
    expect(await parse({ found: false, receipt: null })).toEqual({ found: false, receipt: null })
    expect(
      (await parse({ found: true, receipt: receipt('provider.update', id(1), '9007199254740994') }))
        .found,
    ).toBe(true)
    for (const candidate of [
      { found: false },
      { found: false, receipt: receipt() },
      { found: true, receipt: null },
      { found: true, receipt: receipt('provider.delete', id(1), '9007199254740994') },
      { found: true, receipt: receipt('provider.update', id(2), '9007199254740994') },
      { found: true, receipt: receipt('provider.update', id(1), '9007199254740993') },
      {
        found: true,
        receipt: {
          ...receipt('provider.update', id(1), '9007199254740994'),
          affected_references: '1',
        },
      },
    ])
      await expect(parse(candidate)).rejects.toMatchObject({ kind: 'invalid-response' })
    for (const candidate of [
      { observed: false },
      { observed: false, result: credential() },
      { observed: true, result: { ...credential(), version: '2' } },
      { observed: true, result: { ...credential(), deleted: true } },
      { observed: true, result: { ...credential(), purpose: 'other' } },
    ])
      await expect(
        createSystemProviderAPI(async () => json(candidate)).lookupCredentialCreate(options()),
      ).rejects.toMatchObject({ kind: 'invalid-response' })
    expect(
      await createSystemProviderAPI(async () =>
        json({ observed: false, result: null }),
      ).lookupCredentialCreate(options()),
    ).toEqual({ observed: false, result: null })
    expect(
      (
        await createSystemProviderAPI(async () =>
          json({ credential_id: id(2), purpose: 'model', version: '9223372036854775807' }),
        ).getCredentialMetadata(id(2), signal())
      ).version,
    ).toBe('9223372036854775807')
    await expect(
      createSystemProviderAPI(async () =>
        json({ credential_id: id(3), purpose: 'model', version: '1' }),
      ).getCredentialMetadata(id(2), signal()),
    ).rejects.toMatchObject({ kind: 'invalid-response' })
  })

  it('keeps raw fields, permits JSON expansion and rejects local limits before any request', async () => {
    const fetch = vi.fn<Fetch>(async (path) =>
      json(path.includes('credentials') ? credential() : receipt()),
    )
    const api = createSystemProviderAPI(fetch)
    const material = '\0'.repeat(65536)
    expect(bytes({ value: material })).toBeGreaterThan(65536)
    expect(validateCredentialValue(material) === material).toBe(true)
    await api.createCredential(material, options())
    expect(JSON.parse(fetch.mock.calls[0]![1].body as string).value === material).toBe(true)
    await api.createCredential('界'.repeat(21845) + 'x', options())
    const prefix = 'https://provider.example/',
      expanded = {
        ...input(),
        name: '界'.repeat(128),
        base_url: prefix + '\\"'.repeat(Math.floor((8192 - prefix.length) / 2)),
      }
    expect(bytes({ input: expanded })).toBeGreaterThan(16 * 1024)
    await api.createProvider(expanded, options())
    expect(
      JSON.parse(fetch.mock.calls[2]![1].body as string).input.base_url === expanded.base_url,
    ).toBe(true)
    fetch.mockClear()
    for (const value of ['', 'x'.repeat(65537), '界'.repeat(21845) + 'xy', '\ud800', '\udc00'])
      await expect(api.createCredential(value, options())).rejects.toMatchObject({
        kind: 'invalid-input',
      })
    for (const change of [
      { name: '界'.repeat(129) },
      { name: '\0' },
      { base_url: prefix + 'x'.repeat(8192) },
      { base_url: 'https://user@provider.example/v1' },
      { base_url: 'https://provider.example/v1?' },
      { base_url: 'https://provider.example:0/v1' },
      { base_url: 'https://provider.example/v1%' },
      { options: { arbitrary: true } },
    ])
      await expect(
        api.createProvider({ ...input(), ...change } as ProviderInput, options()),
      ).rejects.toMatchObject({ kind: 'invalid-input' })
    expect(() =>
      captureProviderCommand(
        {
          kind: 'provider.update',
          id: id(1),
          expected_version: '9223372036854775807',
          input: input(),
        },
        true,
      ),
    ).toThrowError('invalid-input')
    expect(fetch).not.toHaveBeenCalled()
  })

  it('limits the three request exceptions internally and keeps every other request at 16 KiB', async () => {
    const fetch = vi.fn<Fetch>(async () => json({})),
      request = accountTransport(fetch)
    const common = {
      signal: signal(),
      csrf: 'S'.repeat(43),
      key: 'stable',
      body: { value: 'x'.repeat(17 * 1024) },
    }
    await request('createProvider', (v) => v, common)
    await request('updateProvider', (v) => v, { ...common, target: id(1) })
    await request('createModelCredential', (v) => v, {
      ...common,
      body: { value: 'x'.repeat(500 * 1024) },
    })
    fetch.mockClear()
    for (const endpoint of [
      'lookupProviderCommand',
      'lookupModelCredentialCreate',
      'login',
    ] as const)
      await expect(
        (request as (e: string, p: (v: unknown) => unknown, o: unknown) => Promise<unknown>)(
          endpoint,
          (v) => v,
          common,
        ),
      ).rejects.toMatchObject({ kind: 'invalid-input' })
    await expect(
      request('deleteProvider', (v) => v, { ...common, target: id(1) }),
    ).rejects.toMatchObject({ kind: 'invalid-input' })
    await expect(
      request('createProvider', (v) => v, { ...common, body: { value: 'x'.repeat(32 * 1024) } }),
    ).rejects.toMatchObject({ kind: 'invalid-input' })
    await expect(
      request('createModelCredential', (v) => v, {
        ...common,
        body: { value: 'x'.repeat(512 * 1024) },
      }),
    ).rejects.toMatchObject({ kind: 'invalid-input' })
    expect(fetch).not.toHaveBeenCalled()
  })

  it('accepts a real streamed Go-escaped full 25 row page above the old cap', async () => {
    const prefix = 'https://provider.example/',
      url =
        prefix + '&<>'.repeat(Math.ceil((8192 - prefix.length) / 3)).slice(0, 8192 - prefix.length)
    const rows = Array.from({ length: 25 }, (_, n) => ({
      ...row(100 - n),
      input: { ...input(), base_url: url, name: '<'.repeat(128), credential_ref: id(2) },
      version: '9223372036854775807',
    }))
    const encoded = new TextEncoder().encode(
      JSON.stringify({ items: rows, next_cursor: '<'.repeat(8192) })
        .replaceAll('&', '\\u0026')
        .replaceAll('<', '\\u003c')
        .replaceAll('>', '\\u003e'),
    )
    expect(encoded.byteLength).toBeGreaterThan(600_000)
    expect(encoded.byteLength).toBeLessThan(2097152)
    const stream = new ReadableStream<Uint8Array>({
      start(controller) {
        for (let i = 0; i < encoded.length; i += 17003)
          controller.enqueue(encoded.slice(i, i + 17003))
        controller.close()
      },
    })
    const result = await createSystemProviderAPI(
      async () =>
        new Response(stream, {
          headers: { 'Content-Type': 'application/json', 'Content-Length': '1' },
        }),
    ).listProviders({}, signal())
    expect(result.items).toHaveLength(25)
    expect(result.items.every((p) => p.input.base_url === url)).toBe(true)
  })

  it('joins delayed stream cancellation after actual-byte overflow and never parses a candidate', async () => {
    let release!: () => void, entered!: () => void
    const cancelled = new Promise<void>((resolve) => {
        entered = resolve
      }),
      tail = new Promise<void>((resolve) => {
        release = resolve
      })
    const stream = new ReadableStream<Uint8Array>({
      start(controller) {
        controller.enqueue(new Uint8Array(2097153))
      },
      cancel() {
        entered()
        return tail
      },
    })
    const parse = vi.fn(),
      request = accountTransport(
        async () =>
          new Response(stream, {
            headers: { 'Content-Type': 'application/json', 'Content-Length': '1' },
          }),
      )
    let settled = false
    const actual = request('listProviders', parse, { signal: signal(), providers: {} }).catch(
      (error: unknown) => {
        settled = true
        return error
      },
    )
    await cancelled
    expect(settled).toBe(false)
    expect(parse).not.toHaveBeenCalled()
    release()
    expect(await actual).toMatchObject({ kind: 'invalid-response' })
  })

  it.each(['problem', 'get', 'models', 'account'])(
    'keeps the 600000B cap for %s responses',
    async (kind) => {
      let wasCancelled = false
      const stream = new ReadableStream<Uint8Array>({
        start(controller) {
          controller.enqueue(new Uint8Array(600001))
        },
        cancel() {
          wasCancelled = true
        },
      })
      const request = accountTransport(
        async () =>
          new Response(stream, {
            status: kind === 'problem' ? 503 : 200,
            headers: {
              'Content-Type': kind === 'problem' ? 'application/problem+json' : 'application/json',
            },
          }),
      )
      const parse = vi.fn()
      const actual =
        kind === 'problem'
          ? request('listProviders', parse, { signal: signal(), providers: {} })
          : kind === 'get'
            ? request('getProvider', parse, { signal: signal(), target: id(1) })
            : kind === 'models'
              ? request('listProviderModels', parse, {
                  signal: signal(),
                  models: { provider_id: id(1) },
                })
              : request('profile', parse, { signal: signal() })
      await expect(actual).rejects.toMatchObject({ kind: 'invalid-response' })
      expect(wasCancelled).toBe(true)
      expect(parse).not.toHaveBeenCalled()
    },
  )
})
