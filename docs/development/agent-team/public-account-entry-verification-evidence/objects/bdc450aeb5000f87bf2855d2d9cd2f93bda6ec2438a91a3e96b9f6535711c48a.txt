+const json = (value: unknown, status = 200) =>
+  new Response(JSON.stringify(value), {
+    status,
+    headers: {
+      'Content-Type': status >= 400 ? 'application/problem+json' : 'application/json',
+      'X-Request-ID': id,
+    },

+function problem(code: string, status = 503, commit_state = 'not_started') {
+  return json(
+    {
+      type: 'urn:agenteam:problem:test',
+      title: 'Error',
+      status,
+      detail: 'safe',
+      instance: '/api/v1/session',
+      code,
+      request_id: id,
+      commit_state,
+    },
+    status,
+  )
+}
+function barrier<T>() {
+  let resolve!: (value: T) => void
+  const promise = new Promise<T>((next) => {
+    resolve = next
+  })
+  return { promise, resolve }

+        : json({ valid: true, expires_at: instant })
+    return mutation(path, init)
+  })

+    auth: createSessionController(createAccountAPI(fetch)),
+    fetch,
+    setSession(next: () => Promise<Response>) {
+      session = next
+    },
+    setMutation(next: Fetch) {
+      mutation = next
+    },
+  }
