func newInvitationsWebFixture(t *testing.T, ctx context.Context, mode string) *invitationsWebFixture {
	t.Helper()
	f := &invitationsWebFixture{mode: mode, stable: map[string]invitationStableFacts{}}
	if mode == "delivery" {
		var e error
		f.smtp, e = smtpfixture.Start(ctx)
		if e != nil {
			t.Fatal("owned SMTP fixture unavailable", e)
		}
		t.Cleanup(func() {
			c, cancel := context.WithTimeout(context.Background(), 20*time.Second)
			defer cancel()
			if e := f.smtp.Close(c); e != nil {
				t.Error(e)
			}
		})
		f.smtpFailed, e = f.smtp.Create(ctx, smtpfixture.Scenario{Mode: "none", MailCode: 550})
		if e != nil {
			t.Fatal(e)
		}
		f.smtpSuccess, e = f.smtp.Create(ctx, smtpfixture.Scenario{Mode: "none"})
		if e != nil {
			t.Fatal(e)
		}
	}
	f.authenticationWebFixture = newInvitationsWebRoot(t, ctx, f)
	f.admin = personalWebCredential{Email: f.entry.Email, Password: f.entry.Password, UserID: f.entry.ID}
	f.setup = &entryWebFixture{authenticationWebFixture: f.authenticationWebFixture}
	f.adminClient = f.setup.entryClient()
	anonymous := f.setup.entryHTTP(ctx, f.adminClient, "GET", "/api/v1/auth/bootstrap", nil, "", false, 200)
	f.setup.entryHTTP(ctx, f.adminClient, "POST", "/api/v1/sessions/login", map[string]string{"email": f.admin.Email, "password": f.admin.Password}, httpString(t, anonymous, "csrf_token"), true, 200)
	session := f.setup.entryHTTP(ctx, f.adminClient, "GET", "/api/v1/session", nil, "", false, 200)
	f.csrf = httpString(t, session, "csrf_token")
	helper := &personalWebFixture{authenticationWebFixture: f.authenticationWebFixture}
	f.member = helper.inviteMember(ctx, f.adminClient, f.csrf, "invitations-ui-member@example.com", "invitations-member")
	f.secrets = []string{f.admin.Password, f.member.Password}
	count := 1
	if mode == "read" {
		count = 26
	}
	if mode == "lifecycle" {
		count = 0
	}
	for i := 0; i < count; i++ {
		email := fmt.Sprintf("invitations-ui-%02d-%s@example.com", i, strings.Repeat("long", 30))
		receipt := f.setup.entryHTTP(ctx, f.adminClient, "POST", "/api/v1/system/invitations", map[string]string{"email": email}, f.csrf, true, 201)
		f.waitInvitation(ctx, httpString(t, receipt, "id"), "sent")
	}
	if mode == "delivery" {
		f.configureSMTP(ctx, f.smtpFailed)
		receipt := f.setup.entryHTTP(ctx, f.adminClient, "POST", "/api/v1/system/invitations", map[string]string{"email": "invitations-ui-smtp@example.com"}, f.csrf, true, 201)
		f.waitInvitation(ctx, httpString(t, receipt, "id"), "failed")
	}
	f.expected = f.databaseInvitations(ctx)
	f.private("invitations-material.json", map[string]any{"admin": f.admin, "member": f.member, "expected": f.expected})
	t.Cleanup(func() {
		for _, secret := range f.secrets {
			if secret != "" && f.log.contains(secret) {
				t.Error("invitation private material escaped restricted fixture")
			}
		}
		for _, v := range f.stable {
			clear(v.Token)
		}
		f.secrets = nil
		f.admin.Password = ""
		f.member.Password = ""
	})
	return f
}

func (f *invitationsWebFixture) controlResponse(response *http.Response) error {
	r := response.Request
	if r == nil || r.Method != "POST" || !strings.HasPrefix(r.URL.Path, "/api/v1/system/") || response.StatusCode < 200 || response.StatusCode >= 300 {
		return nil
	}
	f.mu.Lock()
	defer f.mu.Unlock()
	if f.lostKey != "" && r.Header.Get("Idempotency-Key") == f.lostKey {
		f.replayed++
	}
	if !f.dropNext || r.URL.Path != f.dropPath {
		return nil
	}
	f.dropNext = false
	raw, e := io.ReadAll(io.LimitReader(response.Body, 600001))
	_ = response.Body.Close()
	defer clear(raw)
	var receipt map[string]any
	if e != nil || len(raw) > 600000 || json.Unmarshal(raw, &receipt) != nil || (response.StatusCode != 201 && response.StatusCode != 202) {
		return errors.New("owned drop requires actual valid accepted response")
	}
	value, ok := receipt["id"].(string)
	if !ok {
		return errors.New("owned drop accepted receipt absent")
	}
	f.lostID = value
	f.lostKey = r.Header.Get("Idempotency-Key")
	f.dropped++
	return invitationResponseLost
}
func (f *invitationsWebFixture) private(name string, value any) {
	f.t.Helper()
	raw, e := json.Marshal(value)
	if e != nil {
		f.t.Fatal("private invitation encoding failed")
	}
	defer clear(raw)
	path := filepath.Join(f.directory, name)
	if os.WriteFile(path+".tmp", raw, 0600) != nil || os.Rename(path+".tmp", path) != nil {
		f.t.Fatal("private invitation transfer failed")
	}
}

}
func (f *invitationsWebFixture) databaseInvitations(ctx context.Context) []map[string]any {
	f.t.Helper()
	rows, e := f.db.Connect(f.t).Query(ctx, `SELECT v.id::text,v.canonical_email,v.version::text,v.created_at,v.expires_at,i.job_id::text,c.completed_at,coalesce(j.phase,'enqueue_pending'),coalesce(j.attempts,0)::text,coalesce(j.version,1)::text,a.channel,a.result,coalesce(j.reason,'')
 FROM agenteam_account.invitations v JOIN LATERAL (SELECT d.* FROM agenteam_account.delivery_intents d JOIN agenteam_account.commands x ON x.id=d.id WHERE d.link_id=v.id AND d.kind='invitation' AND x.phase='committed' ORDER BY x.completed_at DESC,d.job_id DESC LIMIT 1) i ON true JOIN agenteam_account.commands c ON c.id=i.id LEFT JOIN agenteam_account.mail_jobs j ON j.id=i.job_id LEFT JOIN agenteam_account.mail_attempts a ON a.id=j.current_attempt_id WHERE v.expires_at>clock_timestamp() ORDER BY v.created_at DESC,v.id DESC`)
	if e != nil {
		f.t.Fatal("safe invitation database observation failed", e)
	}
	defer rows.Close()
	out := []map[string]any{}
	for rows.Next() {
		var invitationID, email, version, jobID, phase, attempts, jobVersion, reason string
		var created, expires, accepted time.Time
		var channel, result *string
		if e = rows.Scan(&invitationID, &email, &version, &created, &expires, &jobID, &accepted, &phase, &attempts, &jobVersion, &channel, &result, &reason); e != nil {
			f.t.Fatal(e)
		}
		instant := func(at time.Time) string {
			v, e := foundation.NewInstant(at)
			if e != nil {
				f.t.Fatal(e)
			}
			return v.String()
		}
		var actual any
		if channel != nil {
			if *channel == "log" {
				actual = "backend_log"
			} else {
				actual = *channel
			}
		}
		if phase == "processing" {
			phase = "unknown"
		}
		delivery := map[string]any{"job_id": jobID, "accepted_at": instant(accepted), "phase": phase, "attempts": attempts, "version": jobVersion, "channel": actual, "attempt_result": result}
		if reason != "" {
			delivery["reason"] = reason
		}
		out = append(out, map[string]any{"id": invitationID, "email": email, "version": version, "created_at": instant(created), "expires_at": instant(expires), "latest_delivery": delivery})
	}
	if e = rows.Err(); e != nil {
		f.t.Fatal(e)
	}
	return out
}
func (f *invitationsWebFixture) waitInvitation(ctx context.Context, invitationID, phase string) {
	f.t.Helper()
	bounded, cancel := context.WithTimeout(ctx, 12*time.Second)
	defer cancel()
	tick := time.NewTicker(20 * time.Millisecond)
	defer tick.Stop()
	for {
		var actual string
		e := f.db.Connect(f.t).QueryRow(bounded, `SELECT coalesce(j.phase,'enqueue_pending') FROM agenteam_account.delivery_intents i JOIN agenteam_account.commands c ON c.id=i.id LEFT JOIN agenteam_account.mail_jobs j ON j.id=i.job_id WHERE i.link_id=$1 AND i.kind='invitation' AND c.phase='committed' ORDER BY c.completed_at DESC,i.job_id DESC LIMIT 1`, invitationID).Scan(&actual)
		if e == nil && actual == phase {
			return
		}
		select {
		case <-bounded.Done():
			f.t.Fatal("owned delivery did not reach required phase", phase)
		case <-tick.C:
		}
	}
}

func (f *invitationsWebFixture) ipc(ctx context.Context, r invitationWebIPC) map[string]any {
	f.t.Helper()
	out := map[string]any{"ok": true, "sequence": r.Sequence}
	conn := f.db.Connect(f.t)
	if r.ID != "" {
		if _, e := foundation.ParseID[struct{}](r.ID); e != nil {
			f.t.Fatal("invitation IPC target invalid")
		}
	}
	switch r.Action {
	case "snapshot":
		out["items"] = f.databaseInvitations(ctx)
	case "wait":
		if r.Phase != "sent" && r.Phase != "failed" {
			f.t.Fatal("owned wait phase rejected")
		}
		f.waitInvitation(ctx, r.ID, r.Phase)
		out["items"] = f.databaseInvitations(ctx)
	case "cooldown", "compete":
		var value invitationStableFacts
		if e := conn.QueryRow(ctx, `SELECT token_verifier,expires_at FROM agenteam_account.invitations WHERE id=$1 AND created_by=$2`, r.ID, f.admin.UserID).Scan(&value.Token, &value.Expires); e != nil {
			f.t.Fatal("owned invitation target missing", e)
		}
		if _, ok := f.stable[r.ID]; !ok {
			f.stable[r.ID] = value
		} else {
			clear(value.Token)
		}
		if _, e := conn.Exec(ctx, `UPDATE agenteam_account.invitations SET last_delivery_at=clock_timestamp()-interval '61 seconds' WHERE id=$1 AND created_by=$2`, r.ID, f.admin.UserID); e != nil {
			f.t.Fatal(e)
		}
		if r.Action == "compete" {
			var version string
			if e := conn.QueryRow(ctx, `SELECT version::text FROM agenteam_account.invitations WHERE id=$1`, r.ID).Scan(&version); e != nil {
				f.t.Fatal(e)
			}
			f.setup.entryHTTP(ctx, f.adminClient, "POST", "/api/v1/system/invitations/"+r.ID+"/resend", map[string]string{"version": version}, f.csrf, true, 202)
			f.waitInvitation(ctx, r.ID, "sent")
		}
	case "stable":
		original, ok := f.stable[r.ID]
		if !ok {
			f.t.Fatal("owned link baseline missing")
		}
		var value invitationStableFacts
		if e := conn.QueryRow(ctx, `SELECT token_verifier,expires_at FROM agenteam_account.invitations WHERE id=$1`, r.ID).Scan(&value.Token, &value.Expires); e != nil {
			f.t.Fatal(e)
		}
		out["stable"] = bytes.Equal(original.Token, value.Token) && original.Expires.Equal(value.Expires)
		clear(value.Token)
	case "expire-all":
		if f.mode != "read" {
			f.t.Fatal("owned expiry control not available")
		}
		if _, e := conn.Exec(ctx, `WITH at AS MATERIALIZED(SELECT clock_timestamp() AS value) UPDATE agenteam_account.invitations SET created_at=at.value-interval '25 hours',expires_at=at.value-interval '1 hour' FROM at WHERE created_by=$1`, f.admin.UserID); e != nil {
			f.t.Fatal(e)
		}
	case "redeem", "revoked-link":
		if f.mode != "lifecycle" || !strings.HasPrefix(r.Email, "invitations-ui-") {
			f.t.Fatal("owned redemption material target rejected")
		}
		record := f.record("invitation", r.Email, r.ID)
		link, e := url.Parse(record.URL)
		if e != nil || link.Scheme+"://"+link.Host != f.origin || len(link.Fragment) != 80 {
			f.t.Fatal("owned invitation link invalid")
		}
		f.secrets = append(f.secrets, record.URL, link.Fragment)
		client := f.setup.entryClient()
		bootstrap := f.setup.entryHTTP(ctx, client, "GET", "/api/v1/auth/bootstrap", nil, "", false, 200)
		csrf := httpString(f.t, bootstrap, "csrf_token")
		status := 200
		if r.Action == "revoked-link" {
			status = 410
		}
		inspected := f.setup.entryHTTP(ctx, client, "POST", "/api/v1/invitations/inspect", map[string]string{"token": link.Fragment}, csrf, false, status)
		if r.Action == "revoked-link" {
			out["invalid"] = inspected["code"] == "RESOURCE_DELETED"
			break
		}
		password := "Owned redeemed invitation " + id[struct{}](f.t).String()
		f.secrets = append(f.secrets, password)
		f.setup.entryHTTP(ctx, client, "POST", "/api/v1/invitations/redeem", map[string]string{"token": link.Fragment, "username": "ui-redeemed", "password": password, "confirmation": password}, csrf, true, 201)
		var present bool
		if e = conn.QueryRow(ctx, `SELECT EXISTS(SELECT 1 FROM agenteam_account.users WHERE email=$1) AND NOT EXISTS(SELECT 1 FROM agenteam_account.invitations WHERE id=$2)`, r.Email, r.ID).Scan(&present); e != nil {
			f.t.Fatal(e)
		}
		out["redeemed"] = present
	case "smtp-success":
		if f.mode != "delivery" {
			f.t.Fatal("owned SMTP control unavailable")
		}
		f.configureSMTP(ctx, f.smtpSuccess)
	case "arm-drop":
		if f.mode != "outcome" && f.mode != "navigation" && f.mode != "authority" {
			f.t.Fatal("owned response control unavailable")
		}
		f.mu.Lock()
		f.dropNext = true
		f.dropPath = "/api/v1/system/invitations"
		if r.ID != "" {
			f.dropPath += "/" + r.ID + "/resend"
		}
		f.mu.Unlock()
	case "arm-get-fail":
		if f.mode != "outcome" {
			f.t.Fatal("owned read control unavailable")
		}
		f.failRead.Store(true)
	case "replay-facts":
		f.mu.Lock()
		key, lostID, replayed := f.lostKey, f.lostID, f.replayed
		f.mu.Unlock()
		var commands, intents int
		if e := conn.QueryRow(ctx, `SELECT count(*) FROM agenteam_account.commands WHERE command_key=$1 AND phase='committed' AND resource_id=$2`, key, lostID).Scan(&commands); e != nil {
			f.t.Fatal(e)
		}
		if e := conn.QueryRow(ctx, `SELECT count(*) FROM agenteam_account.delivery_intents i JOIN agenteam_account.commands c ON c.id=i.id WHERE c.command_key=$1 AND i.link_id=$2`, key, lostID).Scan(&intents); e != nil {
			f.t.Fatal(e)
		}
		out["single"] = commands == 1 && intents == 1 && replayed == 1
	case "demote", "promote", "revoke-session":
		if f.mode != "authority" || r.UserID != f.admin.UserID {
			f.t.Fatal("owned authority identity rejected")
		}
		if _, e := foundation.ParseID[struct{}](r.SessionID); e != nil {
			f.t.Fatal("owned authority Session rejected")
		}
		var owned bool
		if e := conn.QueryRow(ctx, `SELECT EXISTS(SELECT 1 FROM agenteam_account.sessions WHERE id=$1 AND user_id=$2 AND revoked_at IS NULL)`, r.SessionID, r.UserID).Scan(&owned); e != nil || !owned {
			f.t.Fatal("owned active Session missing", e)
		}
		if r.Action == "revoke-session" {
			tag, e := conn.Exec(ctx, `UPDATE agenteam_account.sessions SET revoked_at=clock_timestamp(),revoked_reason='administrative' WHERE id=$1 AND user_id=$2 AND revoked_at IS NULL`, r.SessionID, r.UserID)
			if e != nil || tag.RowsAffected() != 1 {
				f.t.Fatal("owned Session revocation failed", e)
			}
		} else {
			role := "user"
			if r.Action == "promote" {
				role = "admin"
			}
			if _, e := conn.Exec(ctx, `UPDATE agenteam_account.users SET role=$1,version=version+1,updated_at=clock_timestamp() WHERE id=$2`, role, r.UserID); e != nil {
				f.t.Fatal(e)
			}
		}
	default:
		f.t.Fatal("unknown invitation IPC action")
	}
	return out
}

func (f *invitationsWebFixture) browserInvitations(ctx context.Context) map[string]any {
	f.t.Helper()
	root, e := filepath.Abs("../account-captcha-web")
	if e != nil {
		f.t.Fatal(e)
	}
	cmd := exec.CommandContext(ctx, "node", filepath.Join(root, "node_modules/@playwright/test/cli.js"), "test", "--config", filepath.Join(root, "system-invitations.config.js"))
	cmd.Dir = root
	cmd.SysProcAttr = &syscall.SysProcAttr{Setpgid: true}
	cmd.Cancel = func() error { return syscall.Kill(-cmd.Process.Pid, syscall.SIGKILL) }
	cmd.WaitDelay = time.Second
	for _, value := range os.Environ() {
		if !strings.HasPrefix(value, "TMPDIR=") && !strings.HasPrefix(value, "AGENTEAM_AUTH_WEB_") && !strings.HasPrefix(value, "AGENTEAM_INVITATIONS_WEB_") && !strings.HasPrefix(value, "PLAYWRIGHT_NO_COPY_PROMPT=") && !strings.HasPrefix(value, "DEBUG=") && !strings.HasPrefix(value, "PWDEBUG=") {
			cmd.Env = append(cmd.Env, value)
		}
	}
	cmd.Env = append(cmd.Env, "TMPDIR="+f.directory, "AGENTEAM_AUTH_WEB_ORIGIN="+f.origin, "AGENTEAM_INVITATIONS_WEB_CASE="+f.mode, "AGENTEAM_AUTH_WEB_PRIVATE="+f.directory, "AGENTEAM_AUTH_WEB_CHROMIUM=/usr/bin/chromium", "PLAYWRIGHT_NO_COPY_PROMPT=1")
	if images := os.Getenv("AGENTEAM_AUTH_WEB_IMAGES"); images != "" {
		if !filepath.IsAbs(images) {
			f.t.Fatal("invitation image path must be absolute")
		}
		cmd.Env = append(cmd.Env, "AGENTEAM_AUTH_WEB_IMAGES="+images)
	}
	var output bytes.Buffer
	cmd.Stdout, cmd.Stderr = &output, &output
	if e = cmd.Start(); e != nil {
		f.t.Fatal("locked invitation browser runner could not start")
	}
	done := make(chan error, 1)
	go func() { done <- cmd.Wait() }()
	joined := false
	defer func() {
		if !joined {
			_ = cmd.Cancel()
			<-done
		}
		_ = syscall.Kill(-cmd.Process.Pid, syscall.SIGKILL)
		cmd.Env = nil
	}()
	tick := time.NewTicker(20 * time.Millisecond)
	defer tick.Stop()
	sequence := 0
	var runErr error
wait:
	for {
		select {
		case runErr = <-done:
			joined = true
			break wait
		case <-tick.C:
			raw, err := os.ReadFile(filepath.Join(f.directory, "invitations-ipc.json"))
			if errors.Is(err, os.ErrNotExist) {
				continue
			}
			var request invitationWebIPC
			decode := json.Unmarshal(raw, &request)
			clear(raw)
			if err != nil || decode != nil || request.Sequence < 1 || request.Sequence > 80 {
				f.t.Error("private invitation IPC rejected")
				_ = cmd.Cancel()
				continue
			}
			if request.Sequence <= sequence {
				continue
			}
			if request.Sequence != sequence+1 {
				f.t.Error("private invitation IPC sequence invalid")
				_ = cmd.Cancel()
				continue
			}
			reply := f.ipc(ctx, request)
			sequence = request.Sequence
			f.private("invitations-ack-"+strconv.Itoa(sequence)+".json", reply)
		}
	}
	safe := output.String()
	for _, secret := range f.secrets {
		if secret != "" {
			safe = strings.ReplaceAll(safe, secret, "[redacted]")
		}
	}
	f.mu.Lock()
	if f.lostKey != "" {
		safe = strings.ReplaceAll(safe, f.lostKey, "[redacted]")
	}
	f.mu.Unlock()
	f.t.Log(safe)
	if runErr != nil {
		f.t.Fatalf("actual invitation production browser failed: %v", runErr)
	}
	raw, e := os.ReadFile(filepath.Join(f.directory, "invitations-result.json"))
	if e != nil {
		f.t.Fatal("safe invitation browser result missing")
	}
	defer clear(raw)
	var result map[string]any
	if json.Unmarshal(raw, &result) != nil || result["completed"] != true {
		f.t.Fatal("safe invitation browser result invalid")
	}
	return result
}
