import { afterEach, beforeEach, expect, it, vi } from 'vitest'
import { flushPromises, mount, type VueWrapper } from '@vue/test-utils'
import { createMemoryHistory, createRouter } from 'vue-router'
import type { SessionController } from '../candidate03/web/src/composables/useSession'
const selected = vi.hoisted(() => ({ auth: null as SessionController | null }))
vi.mock('../candidate03/web/src/composables/useSession', async original => ({
  ...(await original<typeof import('../candidate03/web/src/composables/useSession')>()),
  useSession: () => selected.auth!,
}))
import App from '../candidate03/web/src/App.vue'
import SystemSettingsView from '../candidate03/web/src/views/system/SystemSettingsView.vue'
import SystemUsersView from '../candidate03/web/src/views/system/SystemUsersView.vue'
import SystemInvitationsView from '../candidate03/web/src/views/system/SystemInvitationsView.vue'
import { installAuthentication } from '../candidate03/web/src/router/auth'
import { failure, fixture, gate, id, json, sameCommand, session } from './support'

let wrapper: VueWrapper | undefined
let host: HTMLDivElement
beforeEach(() => {
  host = document.createElement('div'); host.id = 'app'; document.body.append(host)
  // jsdom lacks layout; real browser focus remains a separate acceptance boundary.
  vi.spyOn(HTMLElement.prototype, 'getClientRects').mockImplementation(
    () => [new DOMRect(0,0,80,24)] as unknown as DOMRectList)
})
afterEach(() => {
  wrapper?.unmount()
  wrapper = undefined
  selected.auth?.leave()
  document.body.innerHTML = ''
  vi.useRealTimers()
  vi.restoreAllMocks()
  vi.unstubAllGlobals()
})
const dialog = () => Array.from(document.querySelectorAll<HTMLElement>('[role="dialog"]')).at(-1)!
function button(label: string, scope: ParentNode = document) {
  const found = Array.from(scope.querySelectorAll<HTMLButtonElement>('button')).find(node =>
    (node.querySelector('.button-label[aria-hidden="false"]')?.textContent ?? node.textContent)?.trim() === label)
  expect(!!found, `Visible button ${label}`).toBe(true)
  return found!
}
async function click(label: string, scope?: ParentNode) {
  button(label, scope).click()
  await flushPromises()
}
async function input(value: string) {
  const element = dialog().querySelector<HTMLInputElement>('input')!
  element.value = value
  element.dispatchEvent(new Event('input', { bubbles: true }))
  await flushPromises()
}

it('retains the uncertain App draft through failed Session checking, confirms only the replay, and destroys it on a new Session', async () => {
  vi.stubGlobal('matchMedia', vi.fn(() => ({ matches: false, addEventListener() {}, removeEventListener() {} })))
  vi.useFakeTimers()
  const f = fixture()
  selected.auth = f.auth
  const router = createRouter({ history: createMemoryHistory(), routes: [
    { path: '/', component: { template: '<h1>Home</h1>' }, meta: { authentication: true, protected: true } },
    { path: '/login', name: 'login', component: { template: '<h1>Login</h1>' }, meta: { authentication: true } },
    { path: '/settings/profile', component: { template: '<h1>Profile</h1>' }, meta: { authentication: true, protected: true } },
    { path: '/system', component: SystemSettingsView, redirect: '/system/users',
      meta: { authentication: true, protected: true, systemAdmin: true, navigation: { label: '系统设置', order: 20 } },
      children: [{ path: 'users', component: SystemUsersView }, { path: 'invitations', component: SystemInvitationsView }] },
  ] })
  installAuthentication(router, f.auth)
  await router.push('/system/invitations')
  await router.isReady()
  wrapper = mount(App, { attachTo: host, global: { plugins: [router] } })
  await flushPromises()
  const menu = wrapper.get('nav[aria-label="系统设置"]')
  expect(menu.findAll('a').map(link => link.text())).toEqual(['用户', '待注册邀请'])
  const toggle = menu.get('button')
  await toggle.trigger('click')
  expect(menu.findAll('a').length).toBe(0)
  expect(router.currentRoute.value.path).toBe('/system/invitations')
  await toggle.trigger('click')
  await click('邀请用户')
  const draft = 'original+draft@example.test'
  await input(draft)
  f.setWrite(async () => json({ id: id(10), job_id: id(11), version: '1', unexpected: true }, 201))
  await click('发送邀请')
  expect(dialog().textContent).toContain('请求结果未确认')
  expect(f.auth.system.invitationProgress?.phase).toBe('uncertain')
  const original = f.writes()[0]!
  const beforeNavigation = f.calls.length
  const leaving = router.push('/system/users')
  await flushPromises()
  expect(dialog().textContent).toContain('放弃未保存修改')
  expect(f.calls.length).toBe(beforeNavigation)
  await click('继续编辑', dialog())
  await leaving
  expect(router.currentRoute.value.path).toBe('/system/invitations')
  expect(dialog().querySelector<HTMLInputElement>('input')?.value).toBe(draft)

  const pendingSession = gate<Response>()
  f.setSession(() => pendingSession.promise)
  const readCount = f.reads().length
  await click('检查当前状态', dialog())
  expect(wrapper.text()).toContain('正在确认会话')
  expect(document.querySelector('[role="dialog"]')).toBeNull()
  pendingSession.resolve(failure())
  await flushPromises()
  expect(wrapper.text()).toContain('会话尚未确认')
  expect(f.auth.system.invitationProgress?.phase).toBe('uncertain')
  expect(f.reads().length).toBe(readCount)
  const unload = new Event('beforeunload', { cancelable: true })
  window.dispatchEvent(unload)
  expect(unload.defaultPrevented).toBe(true)
  f.setSession(async () => json(session()))
  await click('检查当前会话')
  expect(dialog().querySelector<HTMLInputElement>('input')?.value).toBe(draft)
  expect(f.reads().length).toBe(readCount + 1)
  expect(f.auth.system.invitationProgress?.phase).toBe('uncertain')
  expect(button('重试原请求', dialog()).disabled).toBe(false)
  f.setWrite(async () => json({ id: id(10), job_id: id(11), version: '3' }, 201))
  f.setList(async () => failure())
  await click('重试原请求', dialog())
  expect(sameCommand(original, f.writes()[1]!)).toBe(true)
  expect(document.querySelector('[role="dialog"]')).toBeNull()
  expect(wrapper.text()).toContain('邀请请求已接受')
  expect(wrapper.text()).toContain('操作已确认，列表读取失败')
  expect(wrapper.text()).not.toContain('请求结果未确认')
  expect(f.auth.system.invitationProgress?.phase).toBe('confirmed')

  f.setList(async () => json({ items: [] }))
  await click('重新读取列表')
  expect(f.writes().length).toBe(2)
  expect(button('邀请请求已接受').disabled).toBe(false)
  await vi.advanceTimersByTimeAsync(2400)
  await click('邀请用户')
  await input('must-not-follow-new-session@example.test')
  f.setWrite(async () => { throw new Error('Controlled lost response') })
  await click('发送邀请')
  expect(f.auth.system.invitationProgress?.phase).toBe('uncertain')
  f.setSession(async () => json(session(200)))
  await click('检查当前状态', dialog())
  expect(f.auth.state.session?.id).toBe(id(200))
  expect(f.auth.system.invitationProgress).toBeNull()
  expect(document.querySelector('[role="dialog"]')).toBeNull()
  await expect(f.auth.system.retryInvitationOriginal()).rejects.toMatchObject({ kind: 'invalid-input' })
  expect(f.writes().length).toBe(3)
  await click('邀请用户')
  expect(dialog().querySelector<HTMLInputElement>('input')?.value).toBe('')
})

    source: 'input03',
    newIntentPosts: f.writes().length - 1,
    newDraftEmpty: dialog().querySelector<HTMLInputElement>('input')?.value === '',
    submitVisibleLabel: label,
    submitAccessibleLabel: submit.getAttribute('aria-label'),
    submitDisabled: submit.disabled,
    priorWriteStillConfirmed: f.auth.system.invitationProgress?.phase === 'confirmed',
    postReceiptListStillFailed: wrapper.text().includes('操作已确认，列表读取失败'),
    advancedMilliseconds: 0,

  expect(evidence.newIntentPosts).toBe(0)
  expect(evidence.newDraftEmpty).toBe(true)
  expect(submit.disabled).toBe(false)
  expect(label).toBe('发送邀请')

it('retains the open discard confirmation as top layer across automatic pageshow Session checking', async () => {
  vi.stubGlobal('matchMedia', vi.fn(() => ({ matches: false, addEventListener() {}, removeEventListener() {} })))
  vi.spyOn(document, 'visibilityState', 'get').mockReturnValue('visible')
  const f = fixture()
  selected.auth = f.auth
  const router = createRouter({ history: createMemoryHistory(), routes: [
    { path: '/', component: { template: '<h1>Home</h1>' }, meta: { authentication: true, protected: true } },
    { path: '/login', name: 'login', component: { template: '<h1>Login</h1>' }, meta: { authentication: true } },
    { path: '/settings/profile', component: { template: '<h1>Profile</h1>' }, meta: { authentication: true, protected: true } },
    { path: '/system', component: SystemSettingsView, redirect: '/system/users',
      meta: { authentication: true, protected: true, systemAdmin: true, navigation: { label: '系统设置', order: 20 } },
      children: [{ path: 'users', component: SystemUsersView }, { path: 'invitations', component: SystemInvitationsView }] },
  ] })
  installAuthentication(router, f.auth)
  await router.push('/system/invitations')
  await router.isReady()
  wrapper = mount(App, { attachTo: host, global: { plugins: [router] } })
  await flushPromises()
  await click('邀请用户')
  await input('held-draft@example.test')
  await click('取消', dialog())
  expect(dialog().textContent).toContain('放弃未保存修改')
  const check = gate<Response>()
  f.setSession(() => check.promise)
  window.dispatchEvent(new Event('pageshow'))
  await flushPromises()
  expect(f.auth.state.phase).toBe('checking')
  expect(document.querySelectorAll('[role="dialog"]').length).toBe(0)
  check.resolve(json(session()))
  await flushPromises()
  const overlays = Array.from(document.querySelectorAll<HTMLElement>('.ui-overlay'))
  const confirmation = Array.from(document.querySelectorAll<HTMLElement>('[role="dialog"]'))
    .find(element => element.textContent?.includes('放弃未保存修改'))!
  const evidence = {
    source: 'input03',
    restoredSession: f.auth.state.phase === 'authenticated',
    overlays: overlays.length,
    confirmationPresent: !!confirmation,
    confirmationLast: confirmation?.closest('.ui-overlay') === overlays.at(-1),
    confirmationHidden: confirmation?.getAttribute('aria-hidden'),
    confirmationInert: confirmation?.inert,
    newIntentPosts: f.writes().length,
  }
  console.info('INDEPENDENT_OPEN_CONFIRMATION_DOM', JSON.stringify(evidence))
  expect(evidence.restoredSession).toBe(true)
  expect(evidence.overlays).toBe(2)
  expect(evidence.confirmationPresent).toBe(true)
  expect(evidence.newIntentPosts).toBe(0)
  expect(evidence.confirmationLast).toBe(true)
  expect(evidence.confirmationHidden).toBeNull()
  expect(evidence.confirmationInert).toBe(false)
})
