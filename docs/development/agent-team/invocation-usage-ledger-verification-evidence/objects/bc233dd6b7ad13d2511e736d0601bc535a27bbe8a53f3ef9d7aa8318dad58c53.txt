type usageObservedStore struct {
	usage.Store
	mu               sync.Mutex
	armed, rollback  bool
	actual, reported f.CommitResult
	marker           f.ID[f.TransactionAttempt]
	afterRead        func(context.Context, f.Tx, f.TransactionCause) error
}

func (s *usageObservedStore) Acquire(ctx context.Context, tx f.Tx, key f.LockKey, mode f.LockMode) error {
	return s.Store.AcquireAll(ctx, tx, []f.LockRequest{{Key: key, Mode: mode}})
}

func (s *usageObservedStore) WithinTx(ctx context.Context, cause f.TransactionCause, fn func(context.Context, f.Tx) error) f.CommitResult {
	s.mu.Lock()
	armed, rollback, marker, afterRead := s.armed, s.rollback, s.marker, s.afterRead
	s.mu.Unlock()
	armed = armed && cause.Kind() == f.CommandsCause && cause.Details().Primary.Namespace() == "model.usage"
	actual := s.Store.WithinTx(ctx, cause, func(ctx context.Context, tx f.Tx) error {
		e := fn(ctx, tx)
		if e == nil && afterRead != nil {
			e = afterRead(ctx, tx, cause)
		}
		if e == nil && armed && rollback {
			return f.NewFault(f.DependencyUnavailable, f.NotStarted)
		}
		return e
	})
	s.mu.Lock()
	s.actual = actual
	s.mu.Unlock()
	if !armed {
		return actual
	}
	reported := f.UnknownResult(marker, cause)
	s.mu.Lock()
	s.reported = reported
	s.mu.Unlock()
	return reported
}

func usageWriterKey(i uc.InvocationIdentity) f.LockRequest {
	c, _ := f.NewCommandIdentity("fixture.runtime", []string{i.Consumer.ProjectID.String()}, "call", f.IdempotencyKey(i.Attempt.CallID.String()))
	k, _ := f.CommandLock(c)
	return f.LockRequest{Key: k, Mode: f.Exclusive}
}
func usageRuntimeLocks(r uc.InvocationRequest, owner id.Actor) []f.LockRequest {
	locks := []f.LockRequest{usageWriterKey(r.Identity)}
	add := func(k f.LockKey, e error) {
		if e != nil {
			panic(e)
		}
		locks = append(locks, f.LockRequest{Key: k, Mode: f.Shared})
	}
	add(f.UserLock(owner.Details().UserID))
	if r.Actor.Details().Kind == id.Human {
		add(f.UserLock(r.Actor.Details().UserID))
	}
	add(f.ProjectLock(r.Identity.Consumer.ProjectID.String()))
	if r.Identity.Consumer.AgentID != nil {
		add(f.AgentLock(r.Identity.Consumer.AgentID.String()))
	}
	if r.Identity.Consumer.ExecutionID != nil {
		add(f.AggregateLock(f.ExecutionAggregate, r.Identity.Consumer.ExecutionID.String()))
	}
	if r.Identity.Consumer.OperationID != "" {
		add(f.AggregateLock(f.OperationAggregate, r.Identity.Consumer.OperationID))
	}
	k, _ := f.RecordLock(f.CommandRecordLock, "fixture-attempt:"+r.Identity.Attempt.InvocationID.String())
	locks = append(locks, f.LockRequest{Key: k, Mode: f.Exclusive})
	sort.Slice(locks, func(i, j int) bool { return f.CompareLockKeys(locks[i].Key, locks[j].Key) < 0 })
	return locks
}
func (p *usageRuntime) load(ctx context.Context, x postgres.SQLExecutor, r uc.InvocationRequest) (usageCall, usageEvent, f.Digest, error) {
	var callRaw, attemptRaw, eventRaw []byte
	var cv, ev int64
	e := x.QueryRow(ctx, `SELECT c.data,a.identity_data,e.data,c.version,e.version FROM usage_runtime_fixture.calls c JOIN usage_runtime_fixture.attempts a ON a.call_id=c.id JOIN usage_runtime_fixture.events e ON e.invocation_id=a.id WHERE c.id=$1 AND a.id=$2 AND e.sequence=$3`, r.Identity.Attempt.CallID.String(), r.Identity.Attempt.InvocationID.String(), int64(r.Sequence)).Scan(&callRaw, &attemptRaw, &eventRaw, &cv, &ev)
	if e != nil {
		return usageCall{}, usageEvent{}, "", f.NewFault(f.Forbidden, f.NotStarted).WithCause(e)
	}
	var c usageCall
	var a mc.AttemptIdentity
	var v usageEvent
	if json.Unmarshal(callRaw, &c) != nil || json.Unmarshal(attemptRaw, &a) != nil || json.Unmarshal(eventRaw, &v) != nil || a != r.Identity.Attempt || !v.Identity.Equal(r.Identity) || c.ID != a.CallID || !c.Consumer.Equal(r.Identity.Consumer) || c.Snapshot != r.Identity.SnapshotID || !bytes.Equal(resolutionJSON(c.Input), resolutionJSON(r.Identity.Input)) || c.Input.Digest != resolutionDigest(c.Messages) || c.Input.SchemaVersion != 1 || v.Action != r.Action || v.Sequence != r.Sequence || c.Initiator != usageStable(p.base.owner) || v.Value.Identity != c.Model {
		return c, v, "", f.NewFault(f.Forbidden, f.NotStarted)
	}
	var snapshot []byte
	if e = x.QueryRow(ctx, `SELECT snapshot_data->'Identity' FROM agenteam_model.snapshots WHERE id=$1 AND project_id=$2`, c.Snapshot.String(), c.Consumer.ProjectID.String()).Scan(&snapshot); e != nil {
		return c, v, "", e
	}
	var model mc.ModelIdentity
	if json.Unmarshal(snapshot, &model) != nil || model != c.Model {
		return c, v, "", f.NewFault(f.Forbidden, f.NotStarted)
	}
	return c, v, resolutionDigest(struct {
		Call    json.RawMessage
		Attempt json.RawMessage
		Event   json.RawMessage
		CV, EV  int64
	}{callRaw, attemptRaw, eventRaw, cv, ev}), nil
}
func (p *usageRuntime) authorize(ctx context.Context, tx f.Tx, r uc.InvocationRequest, c usageCall, v usageEvent) error {
	current := r.Access == uc.ApplyInvocation && (r.Action == uc.ReserveAction || r.Action == uc.ObserveAction && v.Value.Dispatch == uc.Authorized)
	if current {
		if r.Actor.Details().Kind != id.Human {
			return f.NewFault(f.DependencyUnbound, f.NotStarted)
		}
		if usageStable(r.Actor) != c.Initiator {
			return f.NewFault(f.Forbidden, f.NotStarted)
		}
		if e := p.base.accounts.RequireCurrentSession(ctx, tx, r.Actor); e != nil {
			return e
		}
		grant, e := p.base.projects.AuthorizeProject(ctx, tx, r.Actor, c.Consumer.ProjectID, id.Mutate)
		if e != nil {
			return e
		}
		scope, _ := id.InProject(c.Consumer.ProjectID)
		if !grant.Matches(r.Actor, scope, id.Mutate) {
			return f.NewFault(f.Forbidden, f.NotStarted)
		}
	} else {
		a := r.Actor.Details()
		if a.Kind != id.Service || a.ServiceName != id.ModelRuntime || a.CauseRef != r.Identity.Attempt.InvocationID.String() || a.ProjectID != c.Consumer.ProjectID.String() {
			return f.NewFault(f.Forbidden, f.NotStarted)
		}
	}
	if v.Value.Dispatch == uc.Authorized && !v.Authorized || v.Value.Dispatch == uc.Sent && (!v.Authorized || !v.Sent || !v.DoStarted || v.Decision == nil || !v.Decision.Sent) || v.Value.Dispatch == uc.NotSent && (!v.LocalDone || !v.Joined || v.Sent || !v.DoStarted || v.Decision == nil || v.Decision.Sent || v.Decision.Reason == "") || v.Value.Dispatch == uc.DispatchUnknown && (!v.Authorized || !v.ProtocolUnknown) || v.Value.Final != nil && !v.Joined {
		return f.NewFault(f.InvalidState, f.NotStarted)
	}
	if r.Action == uc.ReserveAction && r.Identity.Attempt.AttemptIndex > 1 {
		x, e := p.store.InTx(tx)
		if e != nil {
			return e
		}
		var raw []byte
		e = x.QueryRow(ctx, `SELECT e.data FROM usage_runtime_fixture.attempts a JOIN usage_runtime_fixture.events e ON e.invocation_id=a.id WHERE a.call_id=$1 AND (a.identity_data->>'attempt_index')::bigint=$2 ORDER BY e.sequence DESC LIMIT 1`, c.ID.String(), int64(r.Identity.Attempt.AttemptIndex)-1).Scan(&raw)
		var previous usageEvent
		if e != nil || json.Unmarshal(raw, &previous) != nil || !previous.Joined || previous.Value.Final == nil {
			return f.NewFault(f.ResourceBusy, f.NotStarted)
		}
	}
	return nil
}
func (p *usageRuntime) Discover(ctx context.Context, r uc.InvocationRequest) (uc.InvocationDependencies, error) {
	if r.Validate() != nil {
		return uc.InvocationDependencies{}, f.NewFault(f.InvalidArgument, f.NotStarted)
	}
	var deps uc.InvocationDependencies
	cause, _ := f.NewRecoveryCause("usage-fixture-discover", r.Identity.Attempt.InvocationID.String(), "")
	result := p.store.WithinTx(ctx, cause, func(ctx context.Context, tx f.Tx) error {
		locks := usageRuntimeLocks(r, p.base.owner)
		if e := p.store.AcquireAll(ctx, tx, locks); e != nil {
			return e
		}
		x, e := p.store.InTx(tx)
		if e != nil {
			return e
		}
		c, v, m, e := p.load(ctx, x, r)
		if e != nil {
			return e
		}
		if e = p.authorize(ctx, tx, r, c, v); e != nil {
			return e
		}
		if r.Action == uc.ReserveAction {
			for _, item := range []struct {
				k  f.AggregateKind
				id string
			}{{f.ProviderAggregate, c.Model.ProviderID.String()}, {f.ModelConfigAggregate, c.Model.ModelID.String()}} {
				k, _ := f.AggregateLock(item.k, item.id)
				locks = append(locks, f.LockRequest{Key: k, Mode: f.Shared})
			}
		}
		b, _ := uc.InvocationBinding(r)
		deps, e = uc.NewInvocationDependencies(p.issuer, uc.InvocationDependencyDetails{Binding: b, Mapping: m, Locks: locks})
		return e
	})
	if result.State() != f.Committed {
		return uc.InvocationDependencies{}, result.Fault()
	}
	return deps, nil
}
func (p *usageRuntime) ValidateInTx(ctx context.Context, tx f.Tx, r uc.InvocationRequest, dep uc.InvocationDependencies) (uc.InvocationFact, error) {
	x, e := p.store.InTx(tx)
	if e != nil {
		return uc.InvocationFact{}, e
	}
	if e = p.store.RequireHeldLocks(ctx, tx, dep.RequiredLocks()); e != nil {
		return uc.InvocationFact{}, e
	}
	if e = p.store.RequireHeldLocks(ctx, tx, usageRuntimeLocks(r, p.base.owner)); e != nil {
		return uc.InvocationFact{}, e
	}
	c, v, m, e := p.load(ctx, x, r)
	if e != nil {
		return uc.InvocationFact{}, e
	}
	b, _ := uc.InvocationBinding(r)
	if !dep.Matches(p.issuer, b, m) {
		return uc.InvocationFact{}, f.NewFault(f.ResourceBusy, f.NotStarted)
	}
	if e = p.authorize(ctx, tx, r, c, v); e != nil {
		return uc.InvocationFact{}, e
	}
	return uc.InvocationFact{Identity: v.Identity.Clone(), Initiator: p.base.owner, Sequence: v.Sequence, Value: v.Value.Clone()}, nil
}

func (v *usageFixture) applyPlan(t *testing.T, r uc.InvocationRequest, p uc.InvocationPlan, ignore, rollback bool) (uc.InvocationReceipt, f.CommitResult) {
	t.Helper()
	var receipt uc.InvocationReceipt
	result := v.storeView.WithinTx(testContext(t), p.Cause(), func(ctx context.Context, tx f.Tx) error {
		if e := v.storeView.AcquireAll(ctx, tx, p.RequiredLocks()); e != nil {
			return e
		}
		var e error
		switch r.Action {
		case uc.ReserveAction:
			receipt, e = v.ledger.ReserveInvocationInTx(ctx, tx, r, p)
		case uc.ObserveAction:
			receipt, e = v.ledger.ObserveInvocationInTx(ctx, tx, r, p)
		case uc.FinalizeAction:
			receipt, e = v.ledger.FinalizeInvocationInTx(ctx, tx, r, p)
		}
		if rollback {
			return f.NewFault(f.DependencyUnavailable, f.NotStarted)
		}
		if ignore {
			return nil
		}
		return e
	})
	return receipt, result
}

		t.Fatal("exact invocation Secret read", e)
	}
	c.credential = material
	v.wire.materials = append(v.wire.materials, material)
	appendKey, _ := ac.NewAppendKey(ac.AccessProducer, newID[struct{}](t).String(), 0)
	outboundContext, e := outbound.NewCallContext(v.wire.actor, id.SystemScope(), appendKey, ac.Associations{HTTPTraceID: newID[struct{}](t).String()})
	if e != nil {
		t.Fatal(e)
	}
	req := wire.Request{Snapshot: c.resolved.Snapshot.Clone(), Messages: c.call.Messages, ToolChoice: mc.ToolChoice{Kind: "none"}, ResponseFormat: mc.ResponseFormat{Kind: "text"}, Mode: mode}
	options := wire.CallOptions{ProjectID: c.call.Consumer.ProjectID, Context: outboundContext, Credential: material, Limits: outbound.Limits{Overall: 20 * time.Second}}
	c.exchange = v.wire.start(t, testContext(t), v.wire.adapter, req, options)
}
func (v *usageFixture) observeWire(t *testing.T, c *usageCase) {
	t.Helper()
	observation := c.exchange.Observe()
	if observation.Decision == nil || !observation.Decision.Sent {
		t.Fatal("exact exchange has no sent witness")
	}
	c.event.Action = uc.ObserveAction
	c.event.Sequence++
	c.event.Value.Dispatch = uc.Sent
	if c.event.Value.DispatchedAt == nil {
		now := usageNow(t, v)
		c.event.Value.DispatchedAt = &now
	}
	c.event.Value.Usage = observation.Usage.Clone()
	c.event.Sent = true
	c.event.DoStarted = observation.DoStarted
	c.event.Decision = observation.Decision
	v.apply(t, v.stage(t, c.event))
}
func (v *usageFixture) finishWire(t *testing.T, c *usageCase, status uc.TerminalStatus, modelError *mc.ModelError) {
	t.Helper()
	wireClose(t, c.exchange)
	if !c.exchange.Joined() {
		t.Fatal("actual join missing")
	}
	c.event.Joined = true
	c.event.Action = uc.FinalizeAction
	c.event.Sequence++
	c.event.Value.Final = &uc.Final{Status: status, Version: 1, FinishedAt: usageNow(t, v), Error: modelError}
	v.apply(t, v.stage(t, c.event))
	if got := len(v.wire.state(t, c.key).Requests); got != c.priorRequests+1 {
		t.Fatalf("exact exchange added requests=%d", got-c.priorRequests)
	}
}
func (v *usageFixture) success(t *testing.T, usageJSON json.RawMessage, consumer *mc.Consumer) *usageCase {
	t.Helper()
	scenario := wireJSON(wireReply("result", "stop", usageJSON, nil))
	c := v.newCase(t, &scenario, consumer)
	v.reserve(t, c)
	v.authorize(t, c)
	v.startWire(t, c, wire.JSONResponse)
	_, e := c.exchange.Result(testContext(t))
	if e != nil {
		t.Fatal(e)
	}
	v.observeWire(t, c)
	v.finishWire(t, c, uc.Succeeded, nil)
	return c
}

// Secret's missing invocation dispatch is isolated by the same persisted owner.
// Acquire remains handled by the accepted resolution router; no witness is reused.
func (p *usageRuntime) CheckReferenceInTx(ctx context.Context, tx f.Tx, a id.Actor, ref sc.CredentialRef, purpose sc.Purpose, owner string, retain bool) error {
	return p.base.router.CheckReferenceInTx(ctx, tx, a, ref, purpose, owner, retain)
}
func (p *usageRuntime) secretRequest(ctx context.Context, x postgres.SQLExecutor, r sc.UsageRequest) (uc.InvocationRequest, usageCall, f.Digest, error) {
	var raw []byte
	if e := x.QueryRow(ctx, `SELECT data FROM usage_runtime_fixture.events WHERE invocation_id=$1 ORDER BY sequence DESC LIMIT 1`, r.RequestID).Scan(&raw); e != nil {
		return uc.InvocationRequest{}, usageCall{}, "", e
	}
	var v usageEvent
	if json.Unmarshal(raw, &v) != nil {
		return uc.InvocationRequest{}, usageCall{}, "", f.NewFault(f.Forbidden, f.NotStarted)
	}
	request := uc.InvocationRequest{Actor: p.base.owner, Access: uc.ApplyInvocation, Action: v.Action, Identity: v.Identity.Clone(), Sequence: v.Sequence}
	c, v, m, e := p.load(ctx, x, request)
	if e != nil {
		return request, c, m, e
	}
	a := r.Actor.Details()
	if r.Validate() != nil || r.Action != sc.ReadLeaseUsage || r.Purpose != sc.Model || a.Kind != id.Service || a.ServiceName != id.SecretService || a.CauseRef != r.LeaseOwner.Details().ID || a.ProjectID != r.Ref.Details().Scope.Details().ProjectID || c.CredentialID != r.Ref.Details().ID.String() || c.LeaseID != r.LeaseID.String() || c.LeaseOwner != r.LeaseOwner.Details() || v.Value.Dispatch != uc.Authorized || !v.Authorized || v.Sent || v.Joined {
		return request, c, m, f.NewFault(f.Forbidden, f.NotStarted)
	}
	return request, c, m, nil
}
func (p *usageRuntime) DiscoverUsage(ctx context.Context, r sc.UsageRequest) (sc.UsageDependencies, error) {
	if r.Action != sc.ReadLeaseUsage {
		return p.base.router.DiscoverUsage(ctx, r)
	}
	request, _, m, e := p.secretRequest(ctx, p.store, r)
	if e != nil {
		return sc.UsageDependencies{}, e
	}
	b, _ := sc.UsageBinding(r)
	return sc.NewUsageDependencies(p.secretIssuer, b, m, usageRuntimeLocks(request, p.base.owner))
}
func (p *usageRuntime) ValidateUsageInTx(ctx context.Context, tx f.Tx, r sc.UsageRequest, plan sc.UsageDependencies) error {
	if r.Action != sc.ReadLeaseUsage {
		return p.base.router.ValidateUsageInTx(ctx, tx, r, plan)
	}
	x, e := p.store.InTx(tx)
	if e != nil {
		return e
	}
	request, c, m, e := p.secretRequest(ctx, x, r)
	if e != nil {
		return e
	}
	if e = p.store.RequireHeldLocks(ctx, tx, usageRuntimeLocks(request, p.base.owner)); e != nil {
		return e
	}
	b, _ := sc.UsageBinding(r)
	if !plan.Matches(p.secretIssuer, b, m) {
		return f.NewFault(f.ResourceBusy, f.NotStarted)
	}
	if e = p.base.accounts.RequireCurrentSession(ctx, tx, p.base.owner); e != nil {
		return e
	}
	_, e = p.base.projects.AuthorizeProject(ctx, tx, p.base.owner, c.Consumer.ProjectID, id.Mutate)
	return e
}
func (p *usageRuntime) AuthorizeLeaseInTx(ctx context.Context, tx f.Tx, a id.Actor, ref sc.CredentialRef, owner sc.CredentialLeaseOwner, action sc.LeaseAction) (sc.UseGrant, error) {
	if action != sc.ReadLease {
		return p.base.router.AuthorizeLeaseInTx(ctx, tx, a, ref, owner, action)
	}
	x, e := p.store.InTx(tx)
	if e != nil {
		return sc.UseGrant{}, e
	}
	if ref.Validate() != nil || owner.Validate() != nil || a.Validate() != nil || a.Details().CauseRef != owner.Details().ID {
		return sc.UseGrant{}, f.NewFault(f.Forbidden, f.NotStarted)
	}
	// This legacy port receives an owner, not an invocation RequestID. The
	// planner above already validates the exact RequestID. Re-read that owner's
	// persisted attempts and require the original writer locks in this same Tx;
	// an unrelated authorized attempt or a different lease cannot grant access.
	rows, e := x.Query(ctx, `SELECT a.id::text,c.data->>'LeaseID' FROM usage_runtime_fixture.calls c JOIN usage_runtime_fixture.attempts a ON a.call_id=c.id WHERE c.data->'LeaseOwner'=$1::jsonb AND c.data->>'CredentialID'=$2 ORDER BY a.id`, resolutionJSON(owner.Details()), ref.Details().ID.String())
	if e != nil {
		return sc.UseGrant{}, e
	}
	type candidate struct{ invocation, lease string }
	var candidates []candidate
	for rows.Next() {
		var c candidate
		if e = rows.Scan(&c.invocation, &c.lease); e != nil {
			rows.Close()
			return sc.UseGrant{}, e
		}
		candidates = append(candidates, c)
	}
	e = rows.Err()
	rows.Close()
	if e != nil {
		return sc.UseGrant{}, e
	}
	for _, c := range candidates {
		leaseID, e := f.ParseID[sc.Lease](c.lease)
		if e != nil {
			return sc.UseGrant{}, e
		}
		r := sc.UsageRequest{Actor: a, Ref: ref, Purpose: sc.Model, LeaseOwner: owner, LeaseID: leaseID, Action: sc.ReadLeaseUsage, RequestID: c.invocation}
		request, call, _, e := p.secretRequest(ctx, x, r)
		if e != nil {
			continue
		}
		if e = p.store.RequireHeldLocks(ctx, tx, usageRuntimeLocks(request, p.base.owner)); e != nil {
			continue
		}
		if e = p.base.accounts.RequireCurrentSession(ctx, tx, p.base.owner); e != nil {
			return sc.UseGrant{}, e
		}
		if _, e = p.base.projects.AuthorizeProject(ctx, tx, p.base.owner, call.Consumer.ProjectID, id.Mutate); e != nil {
			return sc.UseGrant{}, e
		}
		return sc.UseGrant{Subject: a, Consumer: sc.Model}, nil
	}
	return sc.UseGrant{}, f.NewFault(f.Forbidden, f.NotStarted)
}

func usageErrorCode(e error) f.Code {
	var ff *f.Fault
	if errors.As(e, &ff) {
		return ff.Code
	}
	return ""
}
