package account_test

import (
	"net/url"
	"strconv"
	"testing"
	"time"

	"github.com/LunaDeerTech/agenteam/internal/central/foundation"
)

func directoryIndependentKeys(t *testing.T, value map[string]any, fields ...string) {
	t.Helper()
	if len(value) != len(fields) {
		t.Fatal("response has an unexpected field count")
	}
	for _, field := range fields {
		if _, ok := value[field]; !ok {
			t.Fatal("response is missing a required field", field)
		}
	}
}

func TestAccountHTTPSystemUserDirectoryIndependent(t *testing.T) {
	f := newHTTPFixture(t)
	bootstrap := f.record("bootstrap", "admin@mail.com", "")
	admin := f.browser()
	login := admin.login(bootstrap.Email, bootstrap.Password, id[struct{}](t).String())
	oldFields := []string{"id", "email", "username", "display_name", "role", "theme", "version", "initial_password_suggestion"}
	directoryIndependentKeys(t, httpObject(t, login.object(t), "user"), oldFields...)
	directoryIndependentKeys(t, httpObject(t, admin.request("GET", "/api/v1/session", nil, "", "").want(t, 200).object(t), "user"), oldFields...)
	directoryIndependentKeys(t, httpObject(t, admin.profile(), "user"), oldFields...)
	member := f.invite(admin, "directory-independent@example.com", "directory-independent")

	conn := f.db.Connect(t)
	rows, err := conn.Query(ctxFor(t), `SELECT id::text,created_at FROM agenteam_account.users`)
	if err != nil {
		t.Fatal("cannot read owned canonical directory facts", err)
	}
	expected := map[string]string{}
	for rows.Next() {
		var userID string
		var created time.Time
		if err := rows.Scan(&userID, &created); err != nil {
			t.Fatal("cannot scan owned canonical directory facts", err)
		}
		instant, err := foundation.NewInstant(created)
		if err != nil {
			t.Fatal("owned database returned an invalid creation instant", err)
		}
		expected[userID] = instant.String()
	}
	rows.Close()
	if rows.Err() != nil || len(expected) != 2 {
		t.Fatal("owned canonical directory fixture is incomplete")
	}
	result := admin.request("GET", "/api/v1/system/users?limit=100", nil, "", "").want(t, 200)
	page := result.object(t)
	directoryIndependentKeys(t, page, "items")
	items := httpItems(t, page)
	if len(items) != len(expected) {
		t.Fatal("directory omitted or added a user")
	}
	seen := map[string]bool{}
	for _, raw := range items {
		item, ok := raw.(map[string]any)
		if !ok {
			t.Fatal("directory item is not a flat object")
		}
		directoryIndependentKeys(t, item, append(append([]string{}, oldFields...), "created_at")...)
		userID := httpString(t, item, "id")
		created := httpString(t, item, "created_at")
		if expected[userID] == "" || created != expected[userID] || seen[userID] {
			t.Fatal("directory creation time or unique identity differs from canonical database fact")
		}
		seen[userID] = true
		parsed, err := foundation.ParseInstant(created)
		if err != nil || parsed.String() != created {
			t.Fatal("directory creation instant is not canonical")
		}
		if version, err := strconv.ParseUint(httpString(t, item, "version"), 10, 64); err != nil || version == 0 {
			t.Fatal("directory version lost its positive decimal string encoding")
		}
	}
	head := admin.request("HEAD", "/api/v1/system/users?limit=100", nil, "", "").want(t, 200)
	if len(head.data) != 0 || head.headers.Get("Content-Length") != strconv.Itoa(len(result.data)) {
		t.Fatal("HEAD did not preserve the encoded directory length with an empty body")
	}
	first := admin.request("GET", "/api/v1/system/users?limit=1", nil, "", "").want(t, 200).object(t)
	cursor := httpString(t, first, "next_cursor")
	queryPath := "/api/v1/system/users?limit=100&cursor=" + url.QueryEscape(cursor)
	remainder := admin.request("GET", queryPath, nil, "", "").want(t, 200)
	if len(httpItems(t, remainder.object(t))) != 1 {
		t.Fatal("users cursor did not resume at the expected single remaining row")
	}
	headQuery := admin.request("HEAD", queryPath, nil, "", "").want(t, 200)
	if len(headQuery.data) != 0 || headQuery.headers.Get("Content-Length") != strconv.Itoa(len(remainder.data)) {
		t.Fatal("HEAD did not accept and execute the existing users cursor and changed limit")
	}
	for _, query := range []string{"limit=01", "cursor=" + url.QueryEscape(cursor+"x")} {
		get := admin.request("GET", "/api/v1/system/users?"+query, nil, "", "").want(t, 400)
		head := admin.request("HEAD", "/api/v1/system/users?"+query, nil, "", "").want(t, get.status)
		if len(head.data) != 0 || head.headers.Get("Content-Type") != get.headers.Get("Content-Type") {
			t.Fatal("HEAD query failure lost the original status, safe representation or empty body")
		}
	}
	unsupported := admin.request("POST", "/api/v1/system/users", nil, "", admin.csrf).want(t, 405)
	if unsupported.headers.Get("Allow") != "GET, HEAD" {
		t.Fatal("user directory method list does not match its explicit GET and HEAD registrations")
	}
	if other := admin.request("HEAD", "/api/v1/system/invitations", nil, "", "").want(t, 405); len(other.data) != 0 {
		t.Fatal("directory change generalized HEAD support to another Account list")
	}
	member.request("GET", "/api/v1/system/users", nil, "", "").problem(t, 403, "FORBIDDEN")
	denied := member.request("HEAD", "/api/v1/system/users", nil, "", "").want(t, 403)
	if len(denied.data) != 0 {
		t.Fatal("nonadministrator HEAD exposed a response body")
	}
	guest := f.browser()
	guest.request("GET", "/api/v1/system/users", nil, "", "").problem(t, 401, "UNAUTHENTICATED")
	if response := guest.request("HEAD", "/api/v1/system/users", nil, "", "").want(t, 401); len(response.data) != 0 {
		t.Fatal("anonymous HEAD exposed a response body")
	}
	oldCookie := admin.cookie("agenteam_local_session")
	admin.request("POST", "/api/v1/sessions/logout", map[string]any{}, id[struct{}](t).String(), admin.csrf).want(t, 204)
	admin.setCookie("agenteam_local_session", oldCookie)
	admin.request("GET", "/api/v1/system/users", nil, "", "").problem(t, 401, "SESSION_REVOKED")
	admin.setCookie("agenteam_local_session", oldCookie)
	if response := admin.request("HEAD", "/api/v1/system/users", nil, "", "").want(t, 401); len(response.data) != 0 {
		t.Fatal("logged-out HEAD exposed a response body")
	}
	t.Log("independent canonical directory, flat nine-field items, legacy eight-field login/session/profile, HEAD and current permission boundaries verified")
}
