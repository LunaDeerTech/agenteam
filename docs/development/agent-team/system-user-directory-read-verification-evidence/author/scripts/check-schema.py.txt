import json,pathlib,subprocess
root=pathlib.Path('/workspace/agenteam')
old=subprocess.check_output(['git','show','125e3c222ffca01ea0fd987d7dc47b8c30fa5f8b:api/openapi/account.json'],cwd=root,text=True)
new=(root/'api/openapi/account.json').read_text()
def block(s,name,next_name):
 start=s.index('      "'+name+'": {')
 return s[start:s.index('      "'+next_name+'": {',start)]
assert block(old,'User','Session')==block(new,'User','SystemUser')
a,b=json.loads(old),json.loads(new)
ua=a['components']['schemas'].pop('UserList'); ub=b['components']['schemas'].pop('UserList')
assert ub['properties']['items']['items']['$ref']=='#/components/schemas/SystemUser'
ub['properties']['items']['items']['$ref']='#/components/schemas/User'
assert ua==ub
b['components']['schemas'].pop('SystemUser')
a['paths']['/api/v1/system/users']['get'].pop('description'); b['paths']['/api/v1/system/users']['get'].pop('description')
assert a==b
print('PASS: original User schema bytes preserved; only SystemUser, UserList reference and listUsers description changed')
