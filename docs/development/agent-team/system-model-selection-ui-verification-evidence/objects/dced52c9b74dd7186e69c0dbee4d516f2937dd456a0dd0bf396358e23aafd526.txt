function gate<T>() { let resolve!: (value: T) => void; const promise = new Promise<T>(r => { resolve = r }); return { promise, resolve } }

  const cancelled = gate<void>(), tail = gate<void>()

}

describe('Independent Selection repaired evidence boundaries', () => {
  it('requires a complete compatible Model parse after same-Provider protocol change and preserves selector version and IDs', async () => {
    const f = await fixture(); f.page.openEditor(); await choose(f, 'memory', 110, 211)
    const original = f.sources.get(id(110))!, draft = { ...f.page.draft }, version = f.page.editor.version
    f.sources.set(original.id, { ...original, version: '2', input: { ...original.input, protocol: 'openai-embeddings' } })
    f.page.choosePurpose('image'); await flushPromises()
    expect(f.page.draftDetails.memory).toBeNull(); expect(f.page.references.memory.value).toBeNull()
    expect(f.page.canSave.value).toBe(false); expect(f.page.draft).toEqual(draft); expect(f.page.editor.version).toBe(version)
    f.page.choosePurpose('memory'); await flushPromises()
    f.page.selectProvider(f.page.choices.memory.providers.rows.find(p => p.id === original.id)!); await flushPromises()
    expect(f.page.choices.memory.models.phase).toBe('error'); expect(f.page.draftDetails.memory).toBeNull()
    await f.page.save(); expect(f.writes()).toHaveLength(0)
    f.sources.set(original.id, { ...original, version: '3' })
    f.page.retryProvider(); await flushPromises()
    expect(f.page.draftDetails.memory?.provider.input.protocol).toBe('openai-chat-completions')
    expect(f.page.canSave.value).toBe(true); expect(f.page.draft).toEqual(draft); expect(f.page.editor.version).toBe('1')
    await f.page.save()
    expect(f.writes()).toHaveLength(1)
    expect(JSON.parse(String(f.writes()[0]!.init.body))).toEqual({ id: id(900), expected_version: '1', ...draft })
  })

  it.each(['disabled', 'capability'] as const)('invalidates an exact selected Model %s observation without adding a Model version fence', async kind => {
    const f = await fixture(); f.page.openEditor(); await choose(f, 'memory', 110, 211)
    const original = f.records.get(id(211))!, draft = { ...f.page.draft }
    f.records.set(original.id, { ...original, version: '2', input: { ...original.input, enabled: kind !== 'disabled',
      capabilities: { ...original.input.capabilities, structured_output_modes: kind === 'capability' ? ['text'] : ['json_schema'] } } })
    f.page.choosePurpose('memory'); await flushPromises()
    f.page.selectProvider(f.page.choices.memory.providers.rows.find(p => p.id === id(110))!); await flushPromises()
    expect(f.page.draftDetails.memory?.model.version).toBe('2'); expect(f.page.canSave.value).toBe(false)
    expect(f.page.draft).toEqual(draft); expect(f.page.editor.version).toBe('1')
    await f.page.save(); expect(f.writes()).toHaveLength(0)
    f.records.set(original.id, { ...original, version: '3' })
    await f.page.pageAction('models', 'refresh')
    expect(f.page.draftDetails.memory?.model.version).toBe('3'); expect(f.page.canSave.value).toBe(true)
    expect(f.page.draft).toEqual(draft); expect(f.page.editor.version).toBe('1')
  })

  it('does not infer deletion from unrelated or absent page rows, but an exact Provider404 invalidates matching evidence', async () => {
    const f = await fixture(); f.page.openEditor(); await choose(f, 'memory', 110, 211)
    const draft = { ...f.page.draft }, unrelated = provider(999, 'openai-chat-completions')
    f.setProviders(async () => json({ items: [{ ...unrelated, input: { ...unrelated.input, enabled: false } }], next_cursor: null }))
    f.page.choosePurpose('image'); await flushPromises()
    expect(f.page.canSave.value).toBe(true); expect(f.page.draftDetails.memory?.provider.id).toBe(id(110))
    f.setProviders(async () => json({ items: [], next_cursor: null })); await f.page.pageAction('providers', 'refresh')
    expect(f.page.canSave.value).toBe(true); expect(f.page.draft).toEqual(draft)
    f.setProviders(async () => json({ items: [f.sources.get(id(110))!], next_cursor: null }))
    f.page.choosePurpose('memory'); await flushPromises()
    f.setProviderDetail(async () => problem('NOT_FOUND', 404))
    f.page.selectProvider(f.page.choices.memory.providers.rows[0]!); await flushPromises()
    expect(f.page.draftDetails.memory).toBeNull(); expect(f.page.canSave.value).toBe(false)
    expect(f.page.references.memory).toMatchObject({ id: id(210), phase: 'error', value: null })
    expect(f.page.draft).toEqual(draft); expect(f.page.editor.version).toBe('1')
    await f.page.save(); expect(f.writes()).toHaveLength(0)
  })

  it('never applies an obsolete disabled Provider DTO after purpose cancellation and its actual body tail', async () => {
    const f = await fixture(); f.page.openEditor(); await choose(f, 'memory', 110, 211)
    const original = f.sources.get(id(110))!, draft = { ...f.page.draft }
    const pending = held({ items: [{ ...original, version: '2', input: { ...original.input, enabled: false } }], next_cursor: null })
    let count = 0
    f.setProviders(async () => ++count === 1 ? pending.response : json({ items: [original], next_cursor: null }))
    f.page.choosePurpose('embedding'); await flushPromises()
    f.page.choosePurpose('image'); await pending.cancelled.promise
    try {
      expect(f.auth.state.busy).toBe(true); expect(count).toBe(1)
      expect(f.page.draftDetails.memory?.provider.input.enabled).toBe(true)
    } finally { pending.tail.resolve(); await flushPromises(); await flushPromises() }
    expect(count).toBe(2); expect(f.page.activePurpose.value).toBe('image')
    expect(f.page.draftDetails.memory?.provider.input.enabled).toBe(true)
    expect(f.page.canSave.value).toBe(true); expect(f.page.draft).toEqual(draft); expect(f.page.editor.version).toBe('1')
    expect(f.writes()).toHaveLength(0)
  })
})
