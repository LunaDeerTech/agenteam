#!/usr/bin/env python3
"""Run only frozen D27 directory UI groups and record actual resources."""
import argparse
import datetime
import ctypes
import tempfile
import hashlib
import json
import os
from pathlib import Path
import re
import subprocess
import time


ROOT = Path(__file__).resolve().parent
REPO = Path('/workspace/agenteam')
OWNED = ['web/src/api/client.ts', 'web/src/api/account.ts', 'web/src/api/system-account.ts', 'web/src/composables/useSession.ts', 'web/src/composables/useSystemUserDirectory.ts', 'web/src/components/layout/SettingsShell.vue', 'web/src/components/layout/SystemNav.vue', 'web/src/views/system/SystemSettingsView.vue', 'web/src/views/system/SystemUsersView.vue', 'web/src/router/index.ts', 'web/src/router/auth.ts', 'web/src/tests/system-account-client.spec.ts', 'web/src/tests/system-user-directory-state.spec.ts', 'web/src/tests/system-user-directory.spec.ts', 'web/src/tests/personal-settings.spec.ts', 'tests/account/system_user_directory_web_test.go', 'tests/account/system_user_directory_web_fixture_test.go', 'tests/account-captcha-web/system-user-directory.config.js', 'tests/account-captcha-web/e2e/system-user-directory.spec.ts', 'docs/development/frontend/README.md']
BASE = "3affc0194214101cfa1e6fdc583afa5d60005db8"
FROZEN = {}


def utc():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()


def write(path, value):
    path.write_text(json.dumps(value, indent=2, sort_keys=True) + '\n')


def command(argv):
    result = subprocess.run(argv, cwd=REPO, capture_output=True, text=True, timeout=20)
    if result.returncode:
        raise RuntimeError('read command failed: ' + ' '.join(argv) + ': ' + result.stderr[:300])
    return result.stdout


def resources():
    result = {}
    for kind in ['container', 'network']:
        for resource_id in command(['docker', kind, 'ls', '-q', '--no-trunc'] if kind == 'network' else ['docker', kind, 'ls', '-aq', '--no-trunc']).split():
            label = '.Config.Labels' if kind == 'container' else '.Labels'
            try:
                raw = command(['docker', kind, 'inspect', '--format', '{{json .Id}}\t{{json .Name}}\t{{json ' + label + '}}', resource_id]).strip().split('\t')
            except RuntimeError as exc:
                if 'No such' in str(exc) or 'not found' in str(exc):
                    continue  # The fixture may remove an ID after the list call.
                raise
            value = dict(zip(['id', 'name', 'labels'], map(json.loads, raw)))
            value['kind'] = kind
            result[resource_id] = value
    return result


def processes():
    result = {}
    for entry in Path('/proc').iterdir():
        if not entry.name.isdigit():
            continue
        try:
            stat = (entry / 'stat').read_text().rsplit(')', 1)[1].split()
            raw = (entry / 'cmdline').read_bytes()
            result[int(entry.name)] = {'ppid': int(stat[1]), 'starttime': stat[19],
                                      'state': stat[0], 'name': (entry / 'comm').read_text().strip(), 'raw': raw}
        except (OSError, ValueError, IndexError):
            pass
    return result


def reap_adopted(observed, main_pid, waited):
    current = processes()
    for key, value in observed.items():
        pid = value['pid']
        row = current.get(pid)
        if pid == main_pid or not row or row['starttime'] != value['starttime'] or row['ppid'] != os.getpid() or row['state'] != 'Z':
            continue
        try:
            got, status = os.waitpid(pid, os.WNOHANG)
        except ChildProcessError:
            continue
        if got == pid:
            waited.append({'pid': pid, 'starttime': value['starttime'], 'actual_wait': True, 'exit': os.waitstatus_to_exitcode(status), 'time': utc()})


def source_input():
    paths = FROZEN['files']
    return {
        'head': command(['git', 'rev-parse', 'HEAD']).strip(),
        'dist': {str(path.relative_to(REPO)): hashlib.sha256(path.read_bytes()).hexdigest() for path in sorted((REPO / 'web/dist').rglob('*')) if path.is_file()},
        'files': {name: hashlib.sha256((REPO / name).read_bytes()).hexdigest() for name in paths},
        'tracked_product_delta': command(['git', 'diff', '--no-ext-diff', '--name-only', BASE, '--', 'web/src', 'internal', 'tests', 'db', 'api', 'scripts', 'go.mod', 'go.sum']).splitlines(),
        'untracked_product_paths': command(['git', 'ls-files', '--others', '--exclude-standard', '--', 'web/src', 'internal', 'tests', 'db', 'api', 'scripts']).splitlines(),
    }


def main():
    global FROZEN
    parser = argparse.ArgumentParser()
    parser.add_argument('--name', required=True)
    parser.add_argument('--group', choices=['new', 'old', 'independent'], required=True)
    parser.add_argument('--frozen', type=Path, required=True)
    parser.add_argument('--modcache', required=True)
    parser.add_argument('--overlay', type=Path)
    args = parser.parse_args()
    if '/' in args.name or '..' in args.name:
        raise ValueError('invalid run name')
    groups = {'new': ['TestAccountSystemUserDirectoryWebReadAndPagination', 'TestAccountSystemUserDirectoryWebAuthorityAndIdentity', 'TestAccountSystemUserDirectoryWebNavigationAndLayouts'], 'old': ['TestAccountAuthenticationWebSessionLifecycle', 'TestAccountPersonalSettingsWebThemeAndNavigation', 'TestAccountPublicEntryWebIdentityNavigation'], 'independent': ['TestAccountSystemUserDirectoryWebIndependent']}
    expected = set(groups[args.group])
    args.selector = '^(' + '|'.join(groups[args.group]) + ')$'
    if bool(args.overlay) != (args.group == 'independent'):
        raise ValueError('overlay must only select the independent group')
    FROZEN = json.loads(args.frozen.read_text())
    if FROZEN['product_base'] != BASE:
        raise ValueError('wrong accepted backend baseline')
    if ctypes.CDLL(None, use_errno=True).prctl(36, 1, 0, 0, 0) != 0:
        raise RuntimeError('cannot become task child subreaper')
    run = ROOT / 'runs' / args.name
    run.mkdir(parents=True, mode=0o700)
    driver_bytes = Path(__file__).read_bytes()
    (run / 'driver.py.txt').write_bytes(driver_bytes)
    verification_inputs = {'driver_sha256': hashlib.sha256(driver_bytes).hexdigest()}
    if args.overlay:
        overlay_bytes = args.overlay.read_bytes()
        (run / 'overlay.json').write_bytes(overlay_bytes)
        replacements = json.loads(overlay_bytes)['Replace']
        if list(replacements) != [str(REPO / 'tests/account/system_user_directory_independent_test.go')]:
            raise ValueError('overlay is outside the independent directory probe')
        probe_path = Path(next(iter(replacements.values())))
        probe_bytes = probe_path.read_bytes()
        (run / 'directory_independent_test.go.txt').write_bytes(probe_bytes)
        verification_inputs.update({'overlay_sha256': hashlib.sha256(overlay_bytes).hexdigest(),
                                    'probe_sha256': hashlib.sha256(probe_bytes).hexdigest()})
        browser_probe_bytes = {name: (ROOT / 'probes' / name).read_bytes() for name in ['directory-independent.config.mjs', 'directory-independent.spec.ts']}
        verification_inputs['browser_probes'] = {name: hashlib.sha256(value).hexdigest() for name, value in browser_probe_bytes.items()}
        for name, value in browser_probe_bytes.items():
            (run / (name + '.txt')).write_bytes(value)
    write(run / 'verification-input.json', verification_inputs)
    runtime = run / 'runtime'
    runtime.mkdir(mode=0o700)
    before = source_input()
    if before['files'] != FROZEN['files'] or before['dist'] != FROZEN['dist']:
        raise RuntimeError('candidate source/dist differs from frozen input')
    if set(before['tracked_product_delta'] + before['untracked_product_paths']) - set(OWNED):
        raise RuntimeError('product changes outside frozen directory ownership')
    (run / 'frozen-input.json').write_bytes(args.frozen.read_bytes())
    browser_runtime = Path(tempfile.mkdtemp(prefix='agtd27ui-', dir='/tmp'))
    if len(str(browser_runtime)) > 45:
        raise RuntimeError('browser runtime is not short')
    images = run / 'images'
    images.mkdir(mode=0o700)
    write(run / 'input-before.json', before)
    baseline = resources()
    write(run / 'baseline.json', baseline)
    env = os.environ.copy()
    flags = '-mod=readonly -buildvcs=false -p=1 -v'
    if args.overlay:
        flags += ' -overlay=' + str(args.overlay.resolve())
    env.update({'AGENTEAM_GO': '/workspace/toolchains/go1.27.1/bin/go',
                'AGENTEAM_MINIO_BINARY': '/workspace/scratch/agenteam-d27-directory-verification-hyadlp89/bin/minio',
                'AGENTEAM_AUTH_WEB_RUNTIME': str(browser_runtime), 'AGENTEAM_AUTH_WEB_IMAGES': str(images),
                'AGENTEAM_DIRECTORY_INDEPENDENT_ROOT': str(ROOT),
                'GOTOOLCHAIN': 'local', 'GOENV': 'off', 'GOWORK': 'off',
                'GOPROXY': 'off', 'GOSUMDB': 'off', 'GOMODCACHE': args.modcache,
                'GOCACHE': str(ROOT / 'gocache'), 'TMPDIR': str(runtime),
                'CGO_ENABLED': '1', 'GOFLAGS': flags})
    argv = ['sh', 'scripts/test-objects.sh', '-run', args.selector]
    record = {'argv': argv, 'cwd': str(REPO), 'started_utc': utc(),
              'env': {key: env[key] for key in ['AGENTEAM_GO', 'AGENTEAM_MINIO_BINARY', 'AGENTEAM_AUTH_WEB_RUNTIME', 'AGENTEAM_AUTH_WEB_IMAGES', 'AGENTEAM_DIRECTORY_INDEPENDENT_ROOT', 'GOTOOLCHAIN', 'GOENV', 'GOWORK', 'GOPROXY', 'GOSUMDB', 'GOMODCACHE', 'GOCACHE', 'TMPDIR', 'CGO_ENABLED', 'GOFLAGS']}}
    if args.overlay:
        record['overlay_sha256'] = hashlib.sha256(args.overlay.read_bytes()).hexdigest()
    write(run / 'command.json', record)
    observed_resources, observed_processes, errors, adopted_waits = {}, {}, [], []
    start = time.monotonic()
    with (run / 'raw.log').open('wb') as raw:
        process = subprocess.Popen(argv, cwd=REPO, env=env, stdout=raw, stderr=subprocess.STDOUT, start_new_session=True)
        print(json.dumps({'run': args.name, 'pid': process.pid, 'selector': args.selector}), flush=True)
        next_resources = 0.0
        while True:
            current = processes()
            owned = {process.pid} | {pid for pid, row in current.items() if row['ppid'] == os.getpid() or str(runtime).encode() in row['raw'] or str(browser_runtime).encode() in row['raw']}
            for _ in range(12):
                larger = owned | {pid for pid, row in current.items() if row['ppid'] in owned}
                if larger == owned:
                    break
                owned = larger
            for pid in owned:
                if pid in current:
                    row = current[pid]
                    key = str(pid) + ':' + row['starttime']
                    observed_processes[key] = {'pid': pid, 'starttime': row['starttime'], 'ppid': row['ppid'], 'name': row['name']}
            if time.monotonic() >= next_resources:
                try:
                    observed_resources.update({key: value for key, value in resources().items() if key not in baseline})
                except Exception as exc:
                    errors.append({'time': utc(), 'error': str(exc)})
                next_resources = time.monotonic() + 1.0
            reap_adopted(observed_processes, process.pid, adopted_waits)
            if process.poll() is not None:
                break
            time.sleep(0.25)
        exit_code = process.wait()
    record.update({'exit': exit_code, 'seconds': round(time.monotonic() - start, 3), 'ended_utc': utc(), 'actual_wait_completed': True})
    write(run / 'command.json', record)
    write(run / 'observed-resources.json', observed_resources)
    write(run / 'observed-processes.json', observed_processes)
    write(run / 'monitor-errors.json', errors)
    cleanups = []
    reap_adopted(observed_processes, process.pid, adopted_waits)
    for _ in range(2):
        reap_adopted(observed_processes, process.pid, adopted_waits)
        current, proc = resources(), processes()
        exact = {}
        for resource_id, row in observed_resources.items():
            result = subprocess.run(['docker', row['kind'], 'inspect', '--format', '{{json .Id}}', resource_id], capture_output=True, timeout=20)
            missing = b'No such' in result.stderr or b'not found' in result.stderr
            exact[resource_id] = {'absent': result.returncode != 0 and resource_id not in current and missing}
        remaining = [row for row in observed_processes.values() if row['pid'] in proc and proc[row['pid']]['starttime'] == row['starttime']]
        cleanups.append({'time': utc(), 'exact_absent': exact,
                         'baseline_unchanged': {key: current.get(key) for key in baseline} == baseline,
                         'remaining_new': sorted(set(current) - set(baseline)), 'owned_processes': remaining,
                         'runtime_entries': sorted(path.name for path in runtime.iterdir()),
                         'browser_runtime': str(browser_runtime),
                         'browser_runtime_entries': sorted(path.name for path in browser_runtime.iterdir())})
        time.sleep(0.5)
    write(run / 'cleanup.json', cleanups)
    write(run / 'adopted-waits.json', adopted_waits)
    if not list(browser_runtime.iterdir()):
        browser_runtime.rmdir()
    after = source_input()
    write(run / 'input-after.json', after)
    clean = all(row['baseline_unchanged'] and not row['remaining_new'] and not row['owned_processes'] and not row['runtime_entries'] and not row['browser_runtime_entries'] and all(value['absent'] for value in row['exact_absent'].values()) for row in cleanups)
    summary = {'run': args.name, 'driver_exit': exit_code, 'source_files_unchanged': before['files'] == after['files'] and before['dist'] == after['dist'],
               'observed_resources': len(observed_resources), 'observed_processes': len(observed_processes),
               'monitor_errors': len(errors), 'double_cleanup': clean, 'run_directory': str(run)}
    verification_unchanged = Path(__file__).read_bytes() == driver_bytes
    if args.overlay:
        verification_unchanged = verification_unchanged and args.overlay.read_bytes() == overlay_bytes and probe_path.read_bytes() == probe_bytes
        verification_unchanged = verification_unchanged and all((ROOT / 'probes' / name).read_bytes() == value for name, value in browser_probe_bytes.items())
    summary['verification_inputs_unchanged'] = verification_unchanged
    tops = re.findall(r'^--- (PASS|FAIL|SKIP): (\S+)', (run / 'raw.log').read_text(errors='replace'), re.MULTILINE)
    summary['browser_processes'] = sum(row['name'] in {'chromium', 'chrome', 'chrome_crashpad'} for row in observed_processes.values())
    summary['node_processes'] = sum(row['name'] == 'node' for row in observed_processes.values())
    summary['adopted_waits'] = len(adopted_waits)
    summary['browser_runtime_removed'] = not browser_runtime.exists()
    summary['top_levels'] = tops
    summary['selected_tests_passed'] = len(tops) == len(expected) and {name for state, name in tops if state == 'PASS'} == expected
    write(run / 'result.json', summary)
    print(json.dumps(summary), flush=True)
    return 0 if exit_code == 0 and clean and len(observed_resources) == 7 and not errors and summary['source_files_unchanged'] and verification_unchanged and summary['selected_tests_passed'] and summary['browser_processes'] and summary['node_processes'] and summary['browser_runtime_removed'] else 1


if __name__ == '__main__':
    raise SystemExit(main())
