package account_test

import (
	"bytes"
	"context"
	"encoding/json"
	"errors"
	"os"
	"os/exec"
	"path/filepath"
	"strconv"
	"strings"
	"syscall"
	"testing"
	"time"

	"github.com/LunaDeerTech/agenteam/internal/central/foundation"
)

type directoryIndependentFact struct {
	ID          string `json:"id"`
	Email       string `json:"email"`
	Username    string `json:"username"`
	DisplayName string `json:"display_name"`
	Role        string `json:"role"`
	CreatedAt   string `json:"created_at"`
}

func TestAccountSystemUserDirectoryWebIndependent(t *testing.T) {
	ctx, cancel := context.WithTimeout(context.Background(), 2*time.Minute)
	defer cancel()
	f := newPersonalWebFixture(t, ctx)
	conn := f.db.Connect(t)
	rows, err := conn.Query(ctx, `SELECT id::text,email,username,display_name,role,created_at FROM agenteam_account.users ORDER BY created_at DESC,id DESC`)
	if err != nil {
		t.Fatal("owned independent canonical facts unavailable", err)
	}
	var facts []directoryIndependentFact
	for rows.Next() {
		var fact directoryIndependentFact
		var created time.Time
		if err := rows.Scan(&fact.ID, &fact.Email, &fact.Username, &fact.DisplayName, &fact.Role, &created); err != nil {
			t.Fatal("owned independent canonical facts scan failed", err)
		}
		instant, err := foundation.NewInstant(created)
		if err != nil {
			t.Fatal("owned independent timestamp invalid", err)
		}
		fact.CreatedAt = instant.String()
		facts = append(facts, fact)
	}
	rows.Close()
	if rows.Err() != nil || len(facts) != 3 {
		t.Fatal("owned real invitation facts incomplete")
	}
	encoded, err := json.Marshal(facts)
	if err != nil || os.WriteFile(filepath.Join(f.directory, "directory-facts.json"), encoded, 0600) != nil {
		t.Fatal("private canonical facts transfer failed")
	}
	clear(encoded)

	probeRoot := os.Getenv("AGENTEAM_DIRECTORY_INDEPENDENT_ROOT")
	if !filepath.IsAbs(probeRoot) {
		t.Fatal("explicit scratch probe root required")
	}
	harness, err := filepath.Abs("../account-captcha-web")
	if err != nil {
		t.Fatal(err)
	}
	cmd := exec.CommandContext(ctx, "node", filepath.Join(harness, "node_modules/@playwright/test/cli.js"), "test", "--config", filepath.Join(probeRoot, "probes/directory-independent.config.mjs"))
	cmd.Dir = filepath.Join(probeRoot, "probes")
	cmd.SysProcAttr = &syscall.SysProcAttr{Setpgid: true}
	cmd.Cancel = func() error { return syscall.Kill(-cmd.Process.Pid, syscall.SIGKILL) }
	cmd.WaitDelay = time.Second
	for _, value := range os.Environ() {
		if !strings.HasPrefix(value, "TMPDIR=") && !strings.HasPrefix(value, "AGENTEAM_AUTH_WEB_") && !strings.HasPrefix(value, "PLAYWRIGHT_NO_COPY_PROMPT=") && !strings.HasPrefix(value, "DEBUG=") && !strings.HasPrefix(value, "PWDEBUG=") {
			cmd.Env = append(cmd.Env, value)
		}
	}
	cmd.Env = append(cmd.Env, "TMPDIR="+f.directory, "AGENTEAM_AUTH_WEB_ORIGIN="+f.origin, "AGENTEAM_AUTH_WEB_PRIVATE="+f.directory, "AGENTEAM_AUTH_WEB_CHROMIUM=/usr/bin/chromium", "PLAYWRIGHT_NO_COPY_PROMPT=1")
	if images := os.Getenv("AGENTEAM_AUTH_WEB_IMAGES"); images != "" {
		if !filepath.IsAbs(images) {
			t.Fatal("owned screenshot directory must be absolute")
		}
		cmd.Env = append(cmd.Env, "AGENTEAM_AUTH_WEB_IMAGES="+images)
	}
	var output bytes.Buffer
	cmd.Stdout, cmd.Stderr = &output, &output
	if err := cmd.Start(); err != nil {
		t.Fatal("locked independent Playwright did not start")
	}
	defer func() { _ = syscall.Kill(-cmd.Process.Pid, syscall.SIGKILL); cmd.Env = nil }()
	done := make(chan error, 1)
	go func() { done <- cmd.Wait() }()
	ticker := time.NewTicker(20 * time.Millisecond)
	defer ticker.Stop()
	sequence := 0
	var runErr error
wait:
	for {
		select {
		case runErr = <-done:
			break wait
		case <-ticker.C:
			raw, err := os.ReadFile(filepath.Join(f.directory, "directory-control.json"))
			if errors.Is(err, os.ErrNotExist) {
				continue
			}
			var request struct {
				Sequence  int    `json:"sequence"`
				Action    string `json:"action"`
				SessionID string `json:"session_id"`
			}
			if err != nil || json.Unmarshal(raw, &request) != nil {
				t.Error("private independent control invalid")
				_ = cmd.Cancel()
				continue
			}
			clear(raw)
			if request.Sequence == sequence {
				continue
			}
			if request.Sequence != sequence+1 || request.Sequence > 3 {
				t.Error("private independent control sequence invalid")
				_ = cmd.Cancel()
				continue
			}
			var changed bool
			switch {
			case request.Sequence == 1 && request.Action == "demote" && request.SessionID == "":
				tag, err := conn.Exec(ctx, `UPDATE agenteam_account.users SET role='user' WHERE id=$1 AND role='admin'`, f.admin.UserID)
				changed = err == nil && tag.RowsAffected() == 1
			case request.Sequence == 2 && request.Action == "promote" && request.SessionID == "":
				tag, err := conn.Exec(ctx, `UPDATE agenteam_account.users SET role='admin' WHERE id=$1 AND role='user'`, f.admin.UserID)
				changed = err == nil && tag.RowsAffected() == 1
			case request.Sequence == 3 && request.Action == "revoke":
				if _, err := foundation.ParseID[struct{}](request.SessionID); err == nil {
					tag, err := conn.Exec(ctx, `UPDATE agenteam_account.sessions SET revoked_at=clock_timestamp(),revoked_reason='logout' WHERE id=$1 AND user_id=$2 AND revoked_at IS NULL`, request.SessionID, f.admin.UserID)
					changed = err == nil && tag.RowsAffected() == 1
				}
			}
			if !changed {
				t.Error("exact owned fixture change did not affect the expected single row")
				_ = cmd.Cancel()
				continue
			}
			ack := filepath.Join(f.directory, "directory-ack-"+strconv.Itoa(request.Sequence)+".json")
			if os.WriteFile(ack+".tmp", []byte(`{"completed":true}`), 0600) != nil || os.Rename(ack+".tmp", ack) != nil {
				t.Error("private fixture control acknowledgement failed")
				_ = cmd.Cancel()
				continue
			}
			sequence = request.Sequence
			t.Logf("owned independent fixture transition %d applied to exactly one owned row", sequence)
		}
	}
	safe := output.String()
	for _, value := range []string{f.admin.Password, f.member.Password, f.newPassword} {
		safe = strings.ReplaceAll(safe, value, "[redacted]")
	}
	t.Log(safe)
	if runErr != nil {
		t.Fatalf("actual independent browser failed: %v", runErr)
	}
	raw, err := os.ReadFile(filepath.Join(f.directory, "directory-independent-result.json"))
	if err != nil {
		t.Fatal("independent browser completion record missing")
	}
	defer clear(raw)
	var result struct {
		Completed            bool   `json:"completed"`
		UserID               string `json:"user_id"`
		RevokedSessionID     string `json:"revoked_session_id"`
		MemberID             string `json:"member_id"`
		CanonicalRows        int    `json:"canonical_rows"`
		Transitions          int    `json:"transitions"`
		MemberDirectoryReads int    `json:"member_directory_reads"`
	}
	if json.Unmarshal(raw, &result) != nil || !result.Completed || result.UserID != f.admin.UserID || result.MemberID != f.member.UserID || result.CanonicalRows != len(facts) || result.Transitions != 3 || sequence != 3 || result.MemberDirectoryReads != 0 {
		t.Fatal("independent canonical/authority/identity completion incomplete")
	}
	var role, reason string
	var revoked bool
	if err := conn.QueryRow(ctx, `SELECT u.role,s.revoked_at IS NOT NULL,s.revoked_reason FROM agenteam_account.users u JOIN agenteam_account.sessions s ON s.user_id=u.id WHERE u.id=$1 AND s.id=$2`, f.admin.UserID, result.RevokedSessionID).Scan(&role, &revoked, &reason); err != nil || role != "admin" || !revoked || reason != "logout" {
		t.Fatal("exact owned role recovery or revoked Session fact missing", err)
	}
	t.Log("independent browser matched real DB registration time, observed real403/recovery and real401, then showed no directory for a different ordinary identity; actual cmd.Wait returned")
}
