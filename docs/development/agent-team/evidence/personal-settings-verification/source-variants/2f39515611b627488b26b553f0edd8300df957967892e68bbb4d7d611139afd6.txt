import { afterEach, describe, expect, it, vi } from 'vitest'
import { flushPromises, mount, type VueWrapper } from '@vue/test-utils'
import { createMemoryHistory, createRouter } from 'vue-router'
import { Rotate } from 'go-captcha-vue'
import type { AccountAPI, SessionView } from '../api/account'
import { AccountFailure } from '../api/client'
import type { SessionController } from '../composables/useSession'
const selected = vi.hoisted(() => ({ auth: null as SessionController | null }))
vi.mock('../composables/useSession', async (original) => {
  const actual = await original<typeof import('../composables/useSession')>()
  return { ...actual, useSession: () => selected.auth! }
})
import { createSessionController } from '../composables/useSession'
import { installAuthentication, safeReturnTarget } from '../router/auth'
import App from '../App.vue'
import LoginView from '../views/auth/LoginView.vue'
import HomeView from '../views/HomeView.vue'
import NotFoundView from '../views/NotFoundView.vue'

const id = '01900000-0000-7000-8000-000000000001'
const time = '2026-10-05T12:34:56.123456Z'
const view: SessionView = {
  user: {
    id,
    email: 'person@example.com',
    username: '',
    display_name: '',
    role: 'user',
    theme: 'dark',
    version: '1',
    initial_password_suggestion: true,
  },
  session: { id, issued_at: time, absolute_expires_at: time, idle_expires_at: time },
  csrf_token: 'S'.repeat(43),
}
const fault = (code: string, status = 401) =>
  new AccountFailure('problem', {
    type: 'urn:agenteam:problem:test',
    title: 'Test',
    status,
    detail: 'safe error',
    instance: '/api/v1/session',
    code,
    request_id: id,
    commit_state: code === 'COMMIT_UNKNOWN' ? 'unknown' : 'not_started',
  })
let wrapper: VueWrapper | undefined
async function page(path = '/login', unavailable = false) {
  let signedIn = false
  const api = {
    getProfile: vi.fn<AccountAPI['getProfile']>(async () => {
      throw new Error('unexpected personal request')
    }),
    updateProfile: vi.fn<AccountAPI['updateProfile']>(async () => {
      throw new Error('unexpected personal request')
    }),
    getPreferences: vi.fn<AccountAPI['getPreferences']>(async () => {
      throw new Error('unexpected personal request')
    }),
    setPreferences: vi.fn<AccountAPI['setPreferences']>(async () => {
      throw new Error('unexpected personal request')
    }),
    readAvatar: vi.fn<AccountAPI['readAvatar']>(async () => {
      throw new Error('unexpected personal request')
    }),
    putAvatar: vi.fn<AccountAPI['putAvatar']>(async () => {
      throw new Error('unexpected personal request')
    }),
    deleteAvatar: vi.fn<AccountAPI['deleteAvatar']>(async () => {
      throw new Error('unexpected personal request')
    }),
    changePassword: vi.fn<AccountAPI['changePassword']>(async () => {
      throw new Error('unexpected personal request')
    }),
    bootstrap: vi.fn<AccountAPI['bootstrap']>(async () => ({
      csrf_token: 'A'.repeat(43),
      challenge_modes: ['rotate'],
      delivery_channel: 'backend_log',
    })),
    getSession: vi.fn<AccountAPI['getSession']>(async () => {
      if (unavailable) throw fault('DEPENDENCY_UNAVAILABLE', 503)
      if (!signedIn) throw fault('UNAUTHENTICATED')
      return view
    }),
    login: vi.fn<AccountAPI['login']>(async () => {
      signedIn = true
      return { user: view.user, session: view.session, next_path: '/' }
    }),
    logout: vi.fn<AccountAPI['logout']>(async () => {
      signedIn = false
    }),
    createChallenge: vi.fn<AccountAPI['createChallenge']>(async () => ({
      id,
      mode: 'rotate',
      master: 'data:image/png;base64,AA==',
      thumb: 'data:image/png;base64,AA==',
      expires_at: time,
    })),
    verifyChallenge: vi.fn<AccountAPI['verifyChallenge']>(async () => ({
      pass: id + '.' + 'P'.repeat(43),
    })),
  }
  selected.auth = createSessionController(api)
  const router = createRouter({
    history: createMemoryHistory(),
    routes: [
      {
        path: '/',
        name: 'home',
        component: HomeView,
        meta: { authentication: true, protected: true },
      },
      { path: '/login', name: 'login', component: LoginView, meta: { authentication: true } },
      { path: '/debug', component: { template: '<p>Development</p>' } },
      { path: '/:pathMatch(.*)*', component: NotFoundView },
    ],
  })
  installAuthentication(router, selected.auth)
  await router.push(path)
  await router.isReady()
  wrapper = mount(App, { attachTo: document.body, global: { plugins: [router] } })
  await flushPromises()
  return { api, router, auth: selected.auth, wrapper }
}
afterEach(() => {
  wrapper?.unmount()
  wrapper = undefined
  document.body.innerHTML = ''
  vi.useRealTimers()
  vi.restoreAllMocks()
})
describe('formal authentication pages', () => {
  it('has a real labelled form and protected home, preserves password bytes, confirms and logs out', async () => {
    const p = await page()
    expect(p.wrapper.find('.system-nav').exists()).toBe(false)
    expect(p.wrapper.find('h1').text()).toBe('登录')
    expect(document.activeElement?.id).toBe('login-title')
    expect(p.wrapper.find('#login-password').attributes('autocomplete')).toBe('current-password')
    expect(p.wrapper.find('#login-password').attributes('maxlength')).toBeUndefined()
    const password = '  Original 密码 12345  '
    await p.wrapper.get('#login-email').setValue(view.user.email)
    await p.wrapper.get('#login-password').setValue(password)
    await p.wrapper.get('form').trigger('submit')
    await flushPromises()
    expect(p.api.login).toHaveBeenCalledTimes(1)
    expect(p.api.login.mock.calls[0][0].password === password).toBe(true)
    expect(p.router.currentRoute.value.path).toBe('/')
    expect(p.wrapper.get('h1').text()).toBe('首页')
    expect(p.wrapper.get('.account-name').text()).toBe(view.user.email)
    expect(p.wrapper.text()).toContain('建议之后更换')
    expect(document.documentElement.dataset.theme).toBe('dark')
    expect(p.wrapper.find('a[href*="settings"]').exists()).toBe(false)
    expect(p.wrapper.find('a[href*="project"]').exists()).toBe(false)
    // Initial restore, login confirmation, and the single protected navigation.
    expect(p.api.getSession).toHaveBeenCalledTimes(3)
    await p.wrapper.get('.account-actions button').trigger('click')
    await flushPromises()
    expect(p.router.currentRoute.value.path).toBe('/login')
    expect(p.auth.state.phase).toBe('anonymous')
    expect(p.api.logout).toHaveBeenCalledTimes(1)
    expect(p.api.getSession).toHaveBeenCalledTimes(4)
    // The logout Cookie owner must release before the new anonymous bootstrap.
    expect(p.api.bootstrap.mock.calls.length).toBeGreaterThan(1)
    expect(p.wrapper.get('#login-password').element).toHaveProperty('value', '')
  })
  it('hides protected contents when authority is unavailable and provides explicit recovery', async () => {
    const p = await page('/', true)
    expect(p.router.currentRoute.value.path).toBe('/login')
    expect(p.wrapper.find('[aria-label="Dashboard"]').exists()).toBe(false)
    expect(p.wrapper.find('#login-password').exists()).toBe(false)
    expect(p.wrapper.text()).toContain('检查当前会话')
    expect(p.api.bootstrap).not.toHaveBeenCalled()
  })
  it('keeps Unknown inputs locked and offers an explicit original-request retry without optimistic success', async () => {
    const p = await page()
    p.api.login.mockRejectedValue(fault('COMMIT_UNKNOWN', 503))
    const local = vi.spyOn(Storage.prototype, 'setItem')
    await p.wrapper.get('#login-email').setValue(view.user.email)
    await p.wrapper.get('#login-password').setValue('OriginalPassword123')
    await p.wrapper.get('form').trigger('submit')
    await flushPromises()
    expect(p.auth.state.phase).toBe('uncertain')
    expect(p.wrapper.get('#login-password').attributes()).toHaveProperty('readonly')
    expect(p.wrapper.find('[aria-label="Dashboard"]').exists()).toBe(false)
    const retry = p.wrapper
      .findAll('button')
      .find((button) => button.text().includes('重试原请求'))!
    await retry.trigger('click')
    await flushPromises()
    expect(p.api.login).toHaveBeenCalledTimes(2)
    expect(
      JSON.stringify(p.api.login.mock.calls[0].slice(0, 3)) ===
        JSON.stringify(p.api.login.mock.calls[1].slice(0, 3)),
    ).toBe(true)
    expect(local).not.toHaveBeenCalled()
    expect(p.router.currentRoute.value.path).toBe('/login')
  })
  it('keeps Debug and unknown routes independent and accepts only the implemented return path', async () => {
    const p = await page('/debug')
    expect(p.api.getSession).not.toHaveBeenCalled()
    await p.router.push('/no-such-page')
    await flushPromises()
    expect(p.api.getSession).not.toHaveBeenCalled()
    for (const target of [
      'https://example.com/',
      '//example.com',
      '/projects/123',
      ['/'],
      '/#other',
      undefined,
    ])
      expect(safeReturnTarget(target)).toBe('/')
    expect(safeReturnTarget('/')).toBe('/')
  })
  for (const reason of ['abandon', 'csrf'])
    it(`clears unnecessary form material after ${reason} while preserving ordinary credential failures`, async () => {
      const p = await page()
      p.api.login.mockRejectedValueOnce(
        fault(
          reason === 'abandon' ? 'COMMIT_UNKNOWN' : 'CSRF_FAILED',
          reason === 'abandon' ? 503 : 403,
        ),
      )
      await p.wrapper.get('#login-email').setValue(view.user.email)
      await p.wrapper.get('#login-password').setValue('Old-Original-Password-83!')
      await p.wrapper.get('form').trigger('submit')
      await flushPromises()
      const action = reason === 'abandon' ? '放弃原请求' : '检查当前会话'
      await p.wrapper
        .findAll('button')
        .find((button) => button.text().includes(action))!
        .trigger('click')
      await flushPromises()
      expect(p.auth.state.phase).toBe('anonymous')
      expect(p.wrapper.get('#login-email').element).toHaveProperty('value', '')
      expect(p.wrapper.get('#login-password').element).toHaveProperty('value', '')
      expect(p.auth.state.canRetryOriginal).toBe(false)
      await p.auth.retryOriginal()
      expect(p.api.login).toHaveBeenCalledTimes(1)
      p.api.login.mockRejectedValueOnce(fault('UNAUTHENTICATED'))
      const replacement = 'New-Explicit-Password-92!'
      await p.wrapper.get('#login-email').setValue('new@example.com')
      await p.wrapper.get('#login-password').setValue(replacement)
      await p.wrapper.get('form').trigger('submit')
      await flushPromises()
      expect(p.api.login).toHaveBeenCalledTimes(2)
      expect(p.api.login.mock.calls[1][0].password === replacement).toBe(true)
      expect(p.api.login.mock.calls[1][2] !== p.api.login.mock.calls[0][2]).toBe(true)
      expect(
        (p.wrapper.get('#login-password').element as HTMLInputElement).value === replacement,
      ).toBe(true)
    })
  for (const cause of ['rejected', 'timeout'])
    it(`restores the challenge action focus after ${cause} without claiming verification`, async () => {
      const p = await page()
      p.api.login.mockRejectedValueOnce(fault('CHALLENGE_REQUIRED', 403))
      await p.wrapper.get('#login-email').setValue(view.user.email)
      await p.wrapper.get('#login-password').setValue('Original-Password-92!')
      await p.wrapper.get('form').trigger('submit')
      await flushPromises()
      await p.wrapper.get('[data-action="challenge"]').trigger('click')
      await flushPromises()
      expect(p.auth.state.challenge !== null && !p.auth.state.challengeBusy).toBe(true)
      ;(p.wrapper.get('#rotate-angle').element as HTMLInputElement).focus()
      expect(document.activeElement?.id).toBe('rotate-angle')
      let release: (() => void) | undefined
      if (cause === 'rejected')
        p.api.verifyChallenge.mockRejectedValueOnce(fault('CHALLENGE_INVALID', 403))
      else {
        vi.useFakeTimers()
        p.api.verifyChallenge.mockImplementationOnce(
          () =>
            new Promise((resolve) => {
              release = () => resolve({ pass: id + '.' + 'P'.repeat(43) })
            }),
        )
      }
      await p.wrapper.get('#rotate-angle').trigger('keydown', { key: 'Enter' })
      if (cause === 'timeout') await vi.advanceTimersByTimeAsync(30_000)
      await flushPromises()
      expect(p.auth.state.passReady).toBe(false)
      expect(p.auth.state.challenge).toBe(null)
      expect(document.activeElement === p.wrapper.get('[data-action="challenge"]').element).toBe(
        true,
      )
      release?.()
      await flushPromises()
      expect(p.auth.state.passReady).toBe(false)
    })
  it('offers one explicit Session recovery after an old login tail settles on a returned auth route', async () => {
    const p = await page()
    let release!: (value: {
      user: typeof view.user
      session: typeof view.session
      next_path: '/'
    }) => void
    p.api.login.mockImplementationOnce(
      () =>
        new Promise((resolve) => {
          release = resolve
        }),
    )
    await p.wrapper.get('#login-email').setValue(view.user.email)
    await p.wrapper.get('#login-password').setValue('Original-Password-92!')
    await p.wrapper.get('form').trigger('submit')
    await flushPromises()
    expect(p.api.login).toHaveBeenCalledTimes(1)
    await p.router.push('/no-such-page')
    await p.router.push('/login')
    await flushPromises()
    expect(p.auth.state.busy).toBe(true)
    expect(p.api.getSession).toHaveBeenCalledTimes(1)
    release({ user: view.user, session: view.session, next_path: '/' })
    await flushPromises()
    expect(p.auth.state.busy).toBe(false)
    expect(p.auth.state.user).toBe(null)
    expect(p.wrapper.find('[aria-label="Dashboard"]').exists()).toBe(false)
    const recovery = p.wrapper
      .findAll('button')
      .find((button) => button.text().includes('检查当前会话'))
    expect(recovery !== undefined).toBe(true)
    p.api.getSession.mockResolvedValue(view)
    await recovery!.trigger('click')
    await flushPromises()
    expect(p.router.currentRoute.value.path).toBe('/')
    expect(p.api.getSession).toHaveBeenCalledTimes(3)
    expect(p.api.login).toHaveBeenCalledTimes(1)
  })
  it('focuses an enabled new challenge, returns successful verification to login, and never steals edited-input focus', async () => {
    const p = await page()
    p.api.login.mockRejectedValueOnce(fault('CHALLENGE_REQUIRED', 403))
    await p.wrapper.get('#login-email').setValue(view.user.email)
    await p.wrapper.get('#login-password').setValue('Original-Password-92!')
    await p.wrapper.get('form').trigger('submit')
    await flushPromises()
    await p.wrapper.get('[data-action="challenge"]').trigger('click')
    await flushPromises()
    expect(document.activeElement?.id).toBe('rotate-angle')
    await p.wrapper.get('#rotate-angle').trigger('keydown', { key: 'Enter' })
    await flushPromises()
    expect(p.auth.state.passReady).toBe(true)
    expect(document.activeElement === p.wrapper.get('[data-action="login"]').element).toBe(true)
    await p.auth.createChallenge()
    await flushPromises()
    const close = p.wrapper.findAll('button').find((button) => button.text().includes('关闭验证'))!
    ;(close.element as HTMLButtonElement).focus()
    await close.trigger('click')
    await flushPromises()
    expect(document.activeElement === p.wrapper.get('[data-action="challenge"]').element).toBe(true)
    await p.wrapper.get('[data-action="challenge"]').trigger('click')
    await flushPromises()
    ;(p.wrapper.get('#login-email').element as HTMLInputElement).focus()
    await p.wrapper.get('#login-email').setValue('another@example.com')
    await flushPromises()
    expect(document.activeElement?.id).toBe('login-email')
    expect(p.auth.state.challenge).toBe(null)
    expect(p.auth.state.passReady).toBe(false)
  })
  for (const outcome of ['rejected', 'timeout', 'success', 'edited'])
    it(`hands off focus after disabled controls lose browser focus: ${outcome}`, async () => {
      const p = await page()
      p.api.login.mockRejectedValueOnce(fault('CHALLENGE_REQUIRED', 403))
      await p.wrapper.get('#login-email').setValue(view.user.email)
      await p.wrapper.get('#login-password').setValue('Original-Password-92!')
      await p.wrapper.get('form').trigger('submit')
      await flushPromises()
      await p.wrapper.get('[data-action="challenge"]').trigger('click')
      await flushPromises()
      expect(document.activeElement?.id).toBe('rotate-angle')
      let succeed!: () => void, reject!: () => void
      p.api.verifyChallenge.mockImplementationOnce(
        () =>
          new Promise((resolve, fail) => {
            succeed = () => resolve({ pass: id + '.' + 'P'.repeat(43) })
            reject = () => fail(fault('CHALLENGE_INVALID', 403))
          }),
      )
      if (outcome === 'timeout') vi.useFakeTimers()
      await p.wrapper.get('#rotate-angle').trigger('keydown', { key: 'Enter' })
      await flushPromises()
      const range = p.wrapper.get('#rotate-angle').element as HTMLInputElement
      expect(range.disabled).toBe(true)
      // jsdom does not implement the browser's disabled-control focus loss.
      // Explicitly model that native transition; this is pure, not browser evidence.
      range.blur()
      // jsdom also ignores blur() on disabled elements; move focus to the same
      // native fallback explicitly without enabling or changing the control.
      document.body.tabIndex = -1
      document.body.focus()
      document.body.removeAttribute('tabindex')
      expect(document.activeElement === document.body).toBe(true)
      if (outcome === 'edited') {
        const email = p.wrapper.get('#login-email').element as HTMLInputElement
        email.focus()
        email.blur()
        expect(document.activeElement === document.body).toBe(true)
        reject()
      } else if (outcome === 'rejected') reject()
      else if (outcome === 'success') succeed()
      else await vi.advanceTimersByTimeAsync(30_000)
      await flushPromises()
      const target = outcome === 'success' ? 'login' : 'challenge'
      expect(
        document.activeElement ===
          (outcome === 'edited'
            ? document.body
            : p.wrapper.get(`[data-action="${target}"]`).element),
      ).toBe(true)
      expect(p.auth.state.passReady).toBe(outcome === 'success')
      if (outcome === 'timeout') {
        succeed()
        await flushPromises()
        expect(p.auth.state.passReady).toBe(false)
      }
    })
  for (const outcome of ['success', 'rejected', 'timeout', 'edited', 'outside-pointer'])
    it(`remembers a current-question drag through body focus without stealing later interaction: ${outcome}`, async () => {
      const p = await page()
      p.api.login.mockRejectedValueOnce(fault('CHALLENGE_REQUIRED', 403))
      await p.wrapper.get('#login-email').setValue(view.user.email)
      await p.wrapper.get('#login-password').setValue('Original-Password-92!')
      await p.wrapper.get('form').trigger('submit')
      await flushPromises()
      await p.wrapper.get('[data-action="challenge"]').trigger('click')
      await flushPromises()
      const range = p.wrapper.get('#rotate-angle').element as HTMLInputElement
      expect(document.activeElement === range && !range.disabled).toBe(true)
      const drag = p.wrapper.get('.gc-drag-block')
      expect(drag.element.tagName).toBe('DIV')
      expect(drag.attributes('tabindex')).toBeUndefined()
      const pointerDown = (target: Element) => {
        const event = new MouseEvent('pointerdown', { button: 0, bubbles: true })
        Object.defineProperty(event, 'isPrimary', { value: true })
        target.dispatchEvent(event)
      }
      pointerDown(drag.element)
      // Model the non-focusable drag's native focus loss. jsdom does not
      // perform that default pointer action; this is a pure interaction case.
      range.blur()
      expect(document.activeElement === document.body).toBe(true)
      let succeed!: () => void, reject!: () => void
      p.api.verifyChallenge.mockImplementationOnce(
        () =>
          new Promise((resolve, fail) => {
            succeed = () => resolve({ pass: id + '.' + 'P'.repeat(43) })
            reject = () => fail(fault('CHALLENGE_INVALID', 403))
          }),
      )
      if (outcome === 'timeout') vi.useFakeTimers()
      const events = p.wrapper.getComponent(Rotate).props('events') as {
        confirm: (angle: number) => void
      }
      events.confirm(137)
      await flushPromises()
      expect(p.api.verifyChallenge).toHaveBeenCalledTimes(1)
      expect(p.auth.state.challengeBusy && range.disabled).toBe(true)
      if (outcome === 'edited') {
        const email = p.wrapper.get('#login-email').element as HTMLInputElement
        email.focus()
        email.blur()
      } else if (outcome === 'outside-pointer') {
        pointerDown(p.wrapper.get('.login-description').element)
      }
      expect(document.activeElement === document.body).toBe(true)
      if (outcome === 'timeout') await vi.advanceTimersByTimeAsync(30_000)
      else if (outcome === 'success') succeed()
      else reject()
      await flushPromises()
      const movedElsewhere = outcome === 'edited' || outcome === 'outside-pointer'
      const target = outcome === 'success' ? 'login' : 'challenge'
      expect(
        document.activeElement ===
          (movedElsewhere ? document.body : p.wrapper.get(`[data-action="${target}"]`).element),
      ).toBe(true)
      expect(p.auth.state.passReady).toBe(outcome === 'success')
      if (outcome === 'timeout') {
        succeed()
        await flushPromises()
        expect(p.auth.state.passReady).toBe(false)
      }
    })
})
