import { test, expect, type Page, type Browser, type Locator } from "@playwright/test";
import { readFileSync, writeFileSync, renameSync, existsSync } from "node:fs";
import { join } from "node:path";

const directory = process.env.AGENTEAM_AUTH_WEB_PRIVATE!;
type Credential = { email: string; password: string; user_id: string };
type Material = { admin: Credential; member: Credential; new_password: string };
type Profile = {
  user: { id: string; email: string; username: string; display_name: string; role: string; theme: string; version: string; initial_password_suggestion: boolean };
  avatar: { media_type: string; byte_size: string; sha256: string } | null;
};
type Observation = { version: string; theme: string; writes: number; audits: number; avatar_id: string };
const material = (): Material => JSON.parse(readFileSync(join(directory, "personal-credentials.json"), "utf8"));
function result(value: Record<string, unknown>) {
  writeFileSync(join(directory, "personal-result.json"), JSON.stringify({ ...value, completed: true }), { mode: 0o600 });
}
let observationSequence = 0;
async function observe(userID: string): Promise<Observation> {
  const sequence = ++observationSequence;
  const request = join(directory, "observe.json");
  writeFileSync(request + ".tmp", JSON.stringify({ sequence, user_id: userID }), { mode: 0o600 });
  renameSync(request + ".tmp", request);
  const ack = join(directory, `observation-${sequence}.json`);
  await expect.poll(() => existsSync(ack)).toBe(true);
  return JSON.parse(readFileSync(ack, "utf8"));
}
async function privateFill(input: Locator, value: string) {
  try { await input.fill(value); } catch { throw new Error("private password entry failed"); }
}
async function identity(page: Page) {
  return page.evaluate(async () => {
    const response = await fetch("/api/v1/session", { credentials: "same-origin", cache: "no-store" });
    if (response.status !== 200) return { status: response.status, user_id: "", session_id: "", csrf_length: 0 };
    const value = await response.json();
    return { status: response.status, user_id: value.user.id as string, session_id: value.session.id as string, csrf_length: value.csrf_token.length as number };
  });
}
async function profile(page: Page): Promise<Profile> {
  return page.evaluate(async () => {
    const response = await fetch("/api/v1/me", { credentials: "same-origin", cache: "no-store" });
    if (response.status !== 200) throw new Error("confirmed current Profile unavailable");
    return response.json();
  });
}
async function enter(page: Page, credential: Credential, target = "/settings/profile") {
  await page.goto(target);
  await expect(page.getByRole("heading", { name: "登录", exact: true })).toBeVisible();
  await expect(page.getByRole("textbox", { name: "邮箱", exact: true })).toBeEditable();
  await expect(page.locator("#login-password")).toHaveAccessibleName("密码");
  await page.getByRole("textbox", { name: "邮箱", exact: true }).fill(credential.email);
  await privateFill(page.locator("#login-password"), credential.password);
  await page.getByRole("button", { name: "登录", exact: true }).click();
  const title = target.endsWith("appearance") ? "主题" : target.endsWith("password") ? "修改密码" : target === "/" ? "首页" : "基本资料";
  await expect(page.getByRole("heading", { name: title, exact: true, level: 1 })).toBeVisible();
  await expect(page.getByRole("heading", { name: title, exact: true, level: 1 })).toBeFocused();
  if (title === "基本资料") await expect(page.locator("#profile-username")).toBeEditable();
  if (title === "主题") await expect(page.getByRole("radio", { name: "跟随系统", exact: true })).toBeEnabled();
  if (title === "修改密码") await expect(page.locator("#settings-current_password")).toBeEditable();
  expect(new URL(page.url()).pathname).toBe(target === "/settings" ? "/settings/profile" : target);
  const current = await identity(page);
  expect(current.status === 200 && current.user_id === credential.user_id && current.csrf_length === 43).toBe(true);
  return current;
}
async function context(browser: Browser) {
  return browser.newContext({ baseURL: process.env.AGENTEAM_AUTH_WEB_ORIGIN, viewport: { width: 1440, height: 1000 } });
}
async function responseFor(page: Page, action: () => Promise<unknown>, path: string, method: string, status = 200) {
  const pending = page.waitForResponse((response) => new URL(response.url()).pathname === path && response.request().method() === method);
  await action();
  const response = await pending;
  expect(response.status()).toBe(status);
  return response;
}
async function navigate(page: Page, label: string) {
  const menu = page.getByRole("button", { name: "个人设置栏目", exact: true });
  if (await menu.isVisible()) await menu.click();
  await page.getByRole("navigation", { name: "个人设置", exact: true }).getByRole("link", { name: label, exact: true }).click();
  await expect(page.getByRole("heading", { name: label, exact: true, level: 1 })).toBeVisible();
}
async function logout(page: Page) {
  await page.locator(".account-actions").getByRole("button", { name: "退出登录", exact: true }).click();
  await expect(page.getByRole("heading", { name: "登录", exact: true })).toBeVisible();
}
async function checkedWrite(page: Page, method: string, path: string, body: Record<string, unknown>, wrongCSRF = false) {
  return page.evaluate(async ({ method, path, body, wrongCSRF }) => {
    const session = await (await fetch("/api/v1/session", { cache: "no-store", credentials: "same-origin" })).json();
    const response = await fetch(path, { method, credentials: "same-origin", cache: "no-store", headers: { "Content-Type": "application/json", "X-CSRF-Token": wrongCSRF ? "x".repeat(43) : session.csrf_token, "Idempotency-Key": crypto.randomUUID() }, body: JSON.stringify(body) });
    const value = response.status === 204 ? {} : await response.json();
    return { status: response.status, code: value.code as string | undefined };
  }, { method, path, body, wrongCSRF });
}
async function storedMaterialsAbsent(page: Page) {
  expect(await page.evaluate(() => localStorage.length === 0 && sessionStorage.length === 0)).toBe(true);
  expect(new URL(page.url()).search === "" && new URL(page.url()).hash === "").toBe(true);
}
async function noOverflow(page: Page) {
  const value = await page.evaluate(() => ({ fits: document.documentElement.scrollWidth <= innerWidth, width: innerWidth,
    elements: [...document.querySelectorAll("html,body,.settings-shell,.settings-content,.settings-sidebar,.account-actions,.ui-input")].slice(0, 20).map((element) => { const r = element.getBoundingClientRect(); return { tag: element.tagName, class: element.getAttribute("class") || "", x: r.x, width: r.width, right: r.right, client: element.clientWidth, scroll: element.scrollWidth }; }) }));
  if (!value.fits) console.info("settings geometry", JSON.stringify(value));
  expect(value.fits).toBe(true);
}
