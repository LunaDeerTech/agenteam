type secretAuditIndependentStore struct {
	*postgres.Store
	wrongLock foundation.LockRequest
	failCheck atomic.Bool
	faults    atomic.Int32
}

func (s *secretAuditIndependentStore) RequireHeldLocks(ctx context.Context, tx foundation.Tx, locks []foundation.LockRequest) error {
	if s.failCheck.CompareAndSwap(true, false) {
		s.faults.Add(1)
		// Call the actual Store's check on this original live Tx. Its poison,
		// not a fabricated CommitResult, must prevent a commit even if ignored.
		return s.Store.RequireHeldLocks(ctx, tx, append(append([]foundation.LockRequest(nil), locks...), s.wrongLock))
	}
	return s.Store.RequireHeldLocks(ctx, tx, locks)
}

type secretAuditIndependentAppendPort struct {
	*secretProjectAuditAppendPorts
	last error
}

func (p *secretAuditIndependentAppendPort) CheckAppendInTx(ctx context.Context, tx foundation.Tx, entry ac.Entry, key ac.AppendKey) error {
	p.last = p.secretProjectAuditAppendPorts.CheckAppendInTx(ctx, tx, entry, key)
	return p.last
}

func secretAuditIndependentAssembly(t *testing.T, f *secretProjectAuditFixture) (*secret.Service, *secretAuditIndependentStore, *secretAuditIndependentAppendPort) {
	t.Helper()
	key, err := foundation.AggregateLock(foundation.CredentialRefAggregate, newID[sc.Credential](t).String())
	if err != nil {
		t.Fatal(err)
	}
	store := &secretAuditIndependentStore{Store: f.store, wrongLock: foundation.LockRequest{Key: key, Mode: foundation.Exclusive}}
	checker, err := secret.NewProjectAuditAuthority(store)
	if err != nil {
		t.Fatal(err)
	}
	ports := &secretProjectAuditPorts{secretAuthority: &secretAuthority{auditAuthority: &auditAuthority{store: f.store}}, checker: checker}
	appendPort := &secretAuditIndependentAppendPort{secretProjectAuditAppendPorts: &secretProjectAuditAppendPorts{secretProjectAuditPorts: ports}}
	auditor, err := audit.New(store, auditKeys(t), audit.Authorizations{Sessions: ports, System: ports, Projects: appendPort})
	if err != nil {
		t.Fatal(err)
	}
	service, err := secret.New(store, masterKeys(t, 1, 1), auditor, secret.Authorizations{Sessions: ports, System: ports, Projects: ports, Usage: ports})
	if err != nil {
		t.Fatal(err)
	}
	if err = service.Initialize(auditContext(t)); err != nil {
		t.Fatal(err)
	}
	t.Cleanup(service.StopMaintenance)
	f.ports = ports
	return service, store, appendPort
}

func secretAuditIndependentSnapshot(t *testing.T, f *secretProjectAuditFixture) string {
	t.Helper()
	var result string
	err := f.store.QueryRow(auditContext(t), `SELECT jsonb_build_array(
(SELECT coalesce(jsonb_agg(to_jsonb(s) ORDER BY id),'[]') FROM agenteam_secret.secrets s),
(SELECT coalesce(jsonb_agg(to_jsonb(r) ORDER BY id),'[]') FROM agenteam_secret.secret_command_receipts r),
(SELECT coalesce(jsonb_agg(jsonb_build_array(payload_id,scope,project_id,owner_kind,owner_id) ORDER BY payload_id),'[]') FROM agenteam_secret.secret_payloads),
(SELECT coalesce(jsonb_agg(to_jsonb(a) ORDER BY id),'[]') FROM agenteam_audit.audit_records a))::text`).Scan(&result)
	if err != nil {
		t.Fatal(err)
	}
	return result
}

func TestSecretAuditIndependentFactRollbackAndPoison(t *testing.T) {
	for _, mode := range []string{"receipt-payload-owner", "different-wrapper-same-pg", "checker-missing-lock-poisons-ignored-error"} {
		t.Run(mode, func(t *testing.T) {
			f := newSecretProjectAuditFixture(t)
			service, store, port := secretAuditIndependentAssembly(t, f)
			originalChecker := f.ports.checker
			if mode == "different-wrapper-same-pg" {
				other := &secretAuditIndependentStore{Store: f.store}
				var err error
				f.ports.checker, err = secret.NewProjectAuditAuthority(other)
				if err != nil {
					t.Fatal(err)
				}
			}
			entered := false
			f.ports.before = func(ctx context.Context, tx foundation.Tx, e ac.Entry, k ac.AppendKey) (ac.Entry, ac.AppendKey, error) {
				entered = true
				if mode == "receipt-payload-owner" {
					x, err := f.store.InTx(tx)
					if err != nil {
						return e, k, err
					}
					tag, err := x.Exec(ctx, `UPDATE agenteam_secret.secret_payloads SET owner_id=$2 WHERE payload_id=(SELECT digest_payload_id FROM agenteam_secret.secret_command_receipts WHERE command_digest=$1)`, k.Details().CauseRef, newID[sc.Credential](t).String())
					if err != nil {
						return e, k, err
					}
					if tag.RowsAffected() != 1 {
						t.Fatal("actual current receipt payload not reached")
					}
				}
				if mode == "checker-missing-lock-poisons-ignored-error" {
					store.failCheck.Store(true)
				}
				return e, k, nil
			}
			request := f.request(t, sc.Create, []byte("independent-fact-checker-payload"))
			prepared, err := service.PrepareWrite(auditContext(t), request)
			if err != nil {
				t.Fatal(err)
			}
			before := secretAuditIndependentSnapshot(t, f)
			var callback error
			command, _ := foundation.NewCommandsCause(request.Identity)
			result := store.WithinTx(auditContext(t), command, func(ctx context.Context, tx foundation.Tx) error {
				if err := store.AcquireAll(ctx, tx, prepared.RequiredLocks()); err != nil {
					return err
				}
				_, callback = service.ApplyPreparedWriteInTx(ctx, tx, prepared)
				if mode == "checker-missing-lock-poisons-ignored-error" {
					return nil
				}
				return callback
			})
			if !entered || callback == nil || port.last == nil || result.State() != foundation.NotCommitted {
				t.Fatal("actual checker rejection did not roll back original Tx", entered, result.State())
			}
			requireCode(t, callback, foundation.DependencyUnavailable)
			if mode == "checker-missing-lock-poisons-ignored-error" {
				requireCode(t, port.last, foundation.DependencyUnavailable)
				requireCode(t, result.Fault(), foundation.InternalError)
				if store.faults.Load() != 1 {
					t.Fatal("actual Store poison boundary not reached")
				}
			} else {
				requireCode(t, port.last, foundation.Forbidden)
			}
			if secretAuditIndependentSnapshot(t, f) != before {
				t.Fatal("rejected checker leaked canonical, receipt, payload ownership or audit rows")
			}
			// Retry exactly the originally prepared command with the proper checker;
			// success must atomically commit one canonical, receipt and real Audit.
			f.ports.before = nil
			f.ports.checker = originalChecker
			port.last = nil
			result = store.WithinTx(auditContext(t), command, func(ctx context.Context, tx foundation.Tx) error {
				if err := store.AcquireAll(ctx, tx, prepared.RequiredLocks()); err != nil {
					return err
				}
				_, err := service.ApplyPreparedWriteInTx(ctx, tx, prepared)
				return err
			})
			if result.State() != foundation.Committed || f.counts(t) != (secretProjectAuditCounts{canonical: 1, receipts: 1, audits: 1}) {
				t.Fatal("original command retry lost atomic facts", result.State())
			}
			t.Log("real final Tx: checker rejection, complete rollback, exact original prepared retry committed once")
		})
	}
}
