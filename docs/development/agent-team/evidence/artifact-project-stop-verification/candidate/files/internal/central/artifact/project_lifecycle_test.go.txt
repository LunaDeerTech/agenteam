func TestArtifactStopSnapshotRejectsRegistryABA(t *testing.T) {
	s := producerFixture()
	project := testProject(t)
	p, err := s.beginProducer(context.Background(), project)
	if err != nil {
		t.Fatal(err)
	}
	r := s.state()
	r.mu.Lock()
	b := &artifactStopPlan{service: s, hasLocalSnapshot: true, localRevision: r.localRevision, local: []capturedProducer{captureProducer(p)}}
	err = s.validateLocalSnapshot(b)
	r.mu.Unlock()
	if err != nil {
		t.Fatal("unchanged original snapshot rejected", err)
	}
	q, err := s.beginProducer(context.Background(), project)
	if err != nil {
		t.Fatal(err)
	}
	s.finishProducer(q)
	r.mu.Lock()
	sameMembers := len(r.controls) == 1 && r.controls[p]
	err = s.validateLocalSnapshot(b)
	r.mu.Unlock()
	if !sameMembers || err == nil {
		t.Fatal("new lifetime disappeared back into an accepted old snapshot", sameMembers, err)
	}
	s.finishProducer(p)
}

func TestArtifactStopUnconfirmedBatchPreservesOriginalUnionAndOutcome(t *testing.T) {
	attempt, _ := foundation.NewID[foundation.TransactionAttempt]()
	cause, err := txCause()
	if err != nil {
		t.Fatal(err)
	}
	for _, original := range []foundation.CommitResult{
		foundation.NotCommittedResult(foundation.NewFault(foundation.Forbidden, foundation.NotCommitted)),
		foundation.UnknownResult(attempt, cause),
	} {
		t.Run(string(original.State()), func(t *testing.T) {
			s := producerFixture()
			p, err := s.beginProducer(context.Background(), testProject(t))
			if err != nil {
				t.Fatal(err)
			}
			first := []foundation.LockRequest{projectLock(p.project, foundation.Exclusive), controlRecordLock(p.id)}
			if _, started := s.startStopBatch(p, first); !started {
				t.Fatal("first bounded batch rejected")
			}
			s.finishStopBatch(p, original)
			next := []foundation.LockRequest{projectLock(testProject(t), foundation.Exclusive)}
			kept, started := s.startStopBatch(p, next)
			if started || kept.State() != original.State() || kept.AttemptID() != original.AttemptID() || kept.Cause().Details().RecoveryRunID != original.Cause().Details().RecoveryRunID {
				t.Fatal("unconfirmed original batch replaced", started, kept.State())
			}
			if original.Fault() != nil && (kept.Fault() == nil || kept.Fault().Code != original.Fault().Code) {
				t.Fatal("original business fault changed")
			}
			if len(p.pendingBatch) != len(first) {
				t.Fatal("pending union grew or disappeared")
			}
			for i, lock := range first {
				if foundation.CompareLockKeys(p.pendingBatch[i].Key, lock.Key) != 0 || p.pendingBatch[i].Mode != lock.Mode {
					t.Fatal("original pending key replaced")
				}
			}
			// This models a confirmed barrier result, never a NotCommitted
			// callback. Only confirmation releases the slot for a new batch.
			s.finishStopBatch(p, foundation.CommittedResult())
			if _, started := s.startStopBatch(p, next); !started {
				t.Fatal("confirmed batch prevented forward progress")
			}
			s.finishStopBatch(p, foundation.CommittedResult())
			s.finishProducer(p)
		})
	}
}

func TestArtifactProjectStopMergeRequiresBothExactTerminalReports(t *testing.T) {
	cause := testArtifactStopCause(t)
	stopped, _ := oc.NewObjectStopReport(oc.ObjectStopComponent, cause, oc.ObjectStopDetails{State: oc.ProjectStopped})
	merged, err := mergeArtifactReport(cause, oc.ObjectStopDetails{State: oc.ProjectStopPending}, stopped)
	if err != nil || merged.Details().State != oc.ProjectStopPending {
		t.Fatal("empty local diagnostics became stopped", err)
	}
	for _, state := range []oc.ProjectStopState{oc.ProjectStopPending, oc.ProjectStopFailed} {
		d := oc.ObjectStopDetails{State: state}
		if state == oc.ProjectStopFailed {
			d.SafeReason = oc.ProjectStopDependencyUnavailable
		}
		object, _ := oc.NewObjectStopReport(oc.ObjectStopComponent, cause, d)
		out, err := mergeArtifactReport(cause, oc.ObjectStopDetails{State: oc.ProjectStopped}, object)
		if err != nil || out.Details().State != state {
			t.Fatal("Object state was hidden", err)
		}
	}
	wrong, _ := oc.NewObjectStopReport(oc.ObjectStopComponent, testArtifactStopCause(t), oc.ObjectStopDetails{State: oc.ProjectStopped})
	if _, err = mergeArtifactReport(cause, oc.ObjectStopDetails{State: oc.ProjectStopped}, wrong); err == nil {
		t.Fatal("wrong cause accepted")
	}
	wrong, _ = oc.NewObjectStopReport(oc.ArtifactStopComponent, cause, oc.ObjectStopDetails{State: oc.ProjectStopped})
	if _, err = mergeArtifactReport(cause, oc.ObjectStopDetails{State: oc.ProjectStopped}, wrong); err == nil {
		t.Fatal("wrong component accepted")
	}
}
func TestArtifactProjectStopMergeUnknownWinsAndDeduplicates(t *testing.T) {
	cause := testArtifactStopCause(t)
	id, _ := foundation.NewID[oc.ProjectStopResource]()
	ref := oc.ProjectStopRef{Kind: oc.StopObjectLease, ID: id}
	object, _ := oc.NewObjectStopReport(oc.ObjectStopComponent, cause, oc.ObjectStopDetails{State: oc.ProjectStopPending, UnknownRefs: []oc.ProjectStopRef{ref}, SafeReason: oc.ProjectStopOutcomeUnknown})
	out, err := mergeArtifactReport(cause, oc.ObjectStopDetails{State: oc.ProjectStopFailed, SafeReason: oc.ProjectStopOperationFailed, ActiveRefs: []oc.ProjectStopRef{ref, ref}}, object)

		t.Fatal("unknown fact was weakened or duplicated", err)
	}
}

func TestArtifactProjectStopLocalJoinUnknownSurvivesObjectTerminal(t *testing.T) {
	cause := testArtifactStopCause(t)
	id, _ := foundation.NewID[oc.ProjectStopResource]()
	ref := oc.ProjectStopRef{Kind: oc.StopArtifactCreation, ID: id}
	for _, state := range []oc.ProjectStopState{oc.ProjectStopped, oc.ProjectStopFailed} {
		details := oc.ObjectStopDetails{State: state}
		if state == oc.ProjectStopFailed {
			details.SafeReason = oc.ProjectStopDependencyUnavailable
		}
		object, err := oc.NewObjectStopReport(oc.ObjectStopComponent, cause, details)
		if err != nil {
			t.Fatal(err)
		}
		out, err := mergeArtifactReport(cause, oc.ObjectStopDetails{
			State: oc.ProjectStopPending, SafeReason: oc.ProjectStopOutcomeUnknown,
			ActiveRefs: []oc.ProjectStopRef{ref}, UnknownRefs: []oc.ProjectStopRef{ref, ref},
		}, object)
		if err != nil || out.Details().State != oc.ProjectStopPending || out.Details().SafeReason != oc.ProjectStopOutcomeUnknown || len(out.Details().UnknownRefs) != 1 || len(out.Details().ActiveRefs) != 0 {
			t.Fatal("local join uncertainty lost to Object terminal", state, err, out.Details())
		}
	}
}

func TestArtifactStopPendingBatchPreservesPriorUnknownAndNormalizesRefs(t *testing.T) {
	cause := testArtifactStopCause(t)
	id, _ := foundation.NewID[oc.ProjectStopResource]()
	ref := oc.ProjectStopRef{Kind: oc.StopArtifactCreation, ID: id}
	original := foundation.NotCommittedResult(foundation.NewFault(foundation.Forbidden, foundation.NotCommitted))
	observed := foundation.NewFault(foundation.CommitUnknown, foundation.Unknown)
	local := oc.ObjectStopDetails{State: oc.ProjectStopPending, SafeReason: oc.ProjectStopOutcomeUnknown, UnknownRefs: []oc.ProjectStopRef{ref, ref}, ActiveRefs: []oc.ProjectStopRef{ref}}
	report, err := pendingConvergenceReport(cause, local, observed, original)
	if err != nil || report.Validate() != nil || report.Details().SafeReason != oc.ProjectStopOutcomeUnknown || len(report.Details().UnknownRefs) != 1 || len(report.Details().ActiveRefs) != 0 {
		t.Fatal("later NotCommitted hid earlier real Unknown or duplicate refs broke report", err)
	}
	if original.State() != foundation.NotCommitted || original.Fault().Code != foundation.Forbidden {
		t.Fatal("owned pending business result was rewritten")
	}
	_, err = pendingConvergenceReport(cause, oc.ObjectStopDetails{State: oc.ProjectStopPending}, nil, original)
	var fault *foundation.Fault
	if !errors.As(err, &fault) || fault.CommitState != foundation.NotCommitted || fault.Code != foundation.Forbidden {
		t.Fatal("NotCommitted without a real earlier Unknown was changed", err)
	}
}
