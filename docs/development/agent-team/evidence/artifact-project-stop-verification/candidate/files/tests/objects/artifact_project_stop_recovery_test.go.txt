// Only the actual same-Tx changes arm the existing PostgreSQL wire proxy.
// No transaction result, authorization success, or storage response is forged.
type artifactStopCommitStore struct {
	*postgres.Store
	proxy              *commitProxy
	authority          *artifactStopAuthority
	mu                 sync.Mutex
	wanted, hit, tuple string
	armed              atomic.Bool
}

func (s *artifactStopCommitStore) arm(phase string) {
	s.mu.Lock()
	s.wanted, s.hit, s.tuple = phase, "", ""
	s.mu.Unlock()
	s.armed.Store(true)
}
func (s *artifactStopCommitStore) WithinTx(ctx context.Context, cause foundation.TransactionCause, fn func(context.Context, foundation.Tx) error) foundation.CommitResult {
	return s.Store.WithinTx(ctx, cause, func(ctx context.Context, tx foundation.Tx) error {
		if !s.armed.Load() {
			return fn(ctx, tx)
		}
		e, err := s.Store.InTx(tx)
		if err != nil {
			return err
		}
		var beforeWork, beforeJoined int
		if err = e.QueryRow(ctx, `SELECT count(*),count(*) FILTER(WHERE joined_at IS NOT NULL) FROM agenteam_artifact.project_work`).Scan(&beforeWork, &beforeJoined); err != nil {
			return err
		}
		if err = fn(ctx, tx); err != nil {
			return err
		}
		var work, joined, newWork, ownJoin, gate, stopped, source, publication int
		err = e.QueryRow(ctx, `SELECT (SELECT count(*) FROM agenteam_artifact.project_work),(SELECT count(*) FROM agenteam_artifact.project_work WHERE joined_at IS NOT NULL),
 (SELECT count(*) FROM agenteam_artifact.project_work WHERE xmin::text=pg_current_xact_id_if_assigned()::text AND joined_at IS NULL),
 (SELECT count(*) FROM agenteam_artifact.project_work WHERE xmin::text=pg_current_xact_id_if_assigned()::text AND joined_at IS NOT NULL),
 (SELECT count(*) FROM agenteam_artifact.project_stops WHERE xmin::text=pg_current_xact_id_if_assigned()::text AND state='stopping'),
 (SELECT count(*) FROM agenteam_artifact.project_stops WHERE xmin::text=pg_current_xact_id_if_assigned()::text AND state='stopped'),
 (SELECT count(*) FROM agenteam_artifact.commands c JOIN agenteam_artifact.project_work w ON w.resource_id=c.artifact_id AND w.source_lease_id=c.source_lease_id JOIN agenteam_object.object_leases l ON l.id=c.source_lease_id WHERE c.xmin::text=pg_current_xact_id_if_assigned()::text AND w.xmin=c.xmin AND l.xmin=c.xmin AND c.state='pending'),
 (SELECT count(*) FROM agenteam_artifact.commands c JOIN agenteam_artifact.artifacts a ON a.id=c.artifact_id JOIN agenteam_object.object_references r ON r.owner_id=a.id AND r.object_id=a.object_id AND r.kind='canonical' JOIN agenteam_audit.audit_records audit ON audit.resource_id=a.id AND audit.action='artifact.create' WHERE c.xmin::text=pg_current_xact_id_if_assigned()::text AND a.xmin=c.xmin AND r.xmin=c.xmin AND audit.xmin=c.xmin AND c.state='completed')`).Scan(&work, &joined, &newWork, &ownJoin, &gate, &stopped, &source, &publication)
		if err != nil {
			return err
		}
		phase := ""
		switch {
		case publication > 0:
			phase = "publication"
		case source > 0:
			phase = "source_capture"
		case stopped > 0:
			phase = "stopped"
		case gate > 0:
			phase = "gate"
		case ownJoin > 0 && joined > beforeJoined:
			phase = "join"
		case newWork > 0 && work > beforeWork:
			phase = "work"
		}
		if phase == "" && s.authority != nil {
			s.authority.mu.Lock()
			_, valid := s.authority.validated[tx]
			s.authority.mu.Unlock()
			if valid && newWork == 0 && ownJoin == 0 {
				phase = "preflight"
			}
		}
		s.mu.Lock()
		wanted := s.wanted
		s.mu.Unlock()
		if phase == wanted && s.armed.CompareAndSwap(true, false) {
			var tuple string
			if err = e.QueryRow(ctx, `SELECT coalesce(jsonb_agg(jsonb_build_array(id,project_id,process_id,kind,resource_id,encode(command_hash,'hex'),source_project_id,source_lease_id,admission_version,admission_operation,joined_at IS NOT NULL) ORDER BY id),'[]'::jsonb)::text FROM agenteam_artifact.project_work WHERE xmin::text=pg_current_xact_id_if_assigned()::text`).Scan(&tuple); err != nil {
				return err
			}
			s.mu.Lock()
			s.hit, s.tuple = phase, tuple
			s.mu.Unlock()
			s.proxy.armed.Store(1)
		}
		return nil
	})
}
func (s *artifactStopCommitStore) assertHit(t *testing.T, phase string) {
	t.Helper()
	s.mu.Lock()
	defer s.mu.Unlock()
	if s.hit != phase {
		t.Fatal("wrong semantic COMMIT", s.hit, phase)
	}
	t.Log("actual Artifact COMMIT", phase, "work tuple", s.tuple)
}
func artifactCommitFixture(t *testing.T, f *artifactStopFixture, after, late bool, options artifactStopOptions) (*artifactStopFixture, *artifactStopCommitStore, *commitProxy) {
	t.Helper()
	raw, proxy := proxyStore(t, f.fixture, after)

	wrapped := &artifactStopCommitStore{Store: raw, proxy: proxy}
	options.store, options.port = raw, wrapped
	on := bindArtifactStop(t, f.fixture, f.proxy, f.stopAuthority, options)
	wrapped.authority = on.stopAuthority
	return on, wrapped, proxy
}
func TestArtifactProjectStopUnknownAdmissionNeverStartsIO(t *testing.T) {
	for _, outcome := range []string{"committed", "rolled_back", "late"} {
		for _, kind := range []string{"content", "source", "upload"} {
			t.Run(outcome+"/"+kind, func(t *testing.T) {
				f := newArtifactStopFixture(t, artifactStopOptions{})
				source := f.create(t, "base-source", "text/plain", "body")
				ref, _ := source.Details().Reference.BusinessFile()
				on, wrapped, proxy := artifactCommitFixture(t, f, outcome == "committed", outcome == "late", artifactStopOptions{})
				var release sync.Once
				defer release.Do(func() { close(proxy.release) })
				var target art.UploadTarget
				var err error
				if kind == "upload" {
					target, err = on.artifact.BeginUpload(contextFor(t), on.invocation(t), command(t, "unknown-target"), art.Display{Name: "body"}, "text/plain", 4, nil)
					if err != nil {
						t.Fatal(err)
					}
				}
				body := &measuredStopBody{}
				gets, puts := f.proxy.gets.Load(), f.proxy.puts.Load()
				wrapped.arm("work")
				done := make(chan error, 1)
				go func() {
					var e error
					switch kind {
					case "content":
						_, e = on.artifact.CreateFromContent(contextFor(t), on.invocation(t), command(t, "unknown-content"), art.Display{Name: "body"}, "text/plain", "body")
					case "source":
						_, e = on.artifact.CreateFromSource(contextFor(t), on.invocation(t), command(t, "unknown-source"), art.Display{Name: "body"}, ref)
					case "upload":
						_, e = on.artifact.UploadPayload(contextFor(t), on.invocation(t), target, body)
					}
					done <- e
				}()
				artifactStopWait(t, proxy.reached)
				wrapped.assertHit(t, "work")
				if on.resolver.resolves.Load() != 0 || body.reads.Load() != 0 || f.proxy.gets.Load() != gets || f.proxy.puts.Load() != puts {
					t.Fatal("I/O preceded confirmed Artifact work")
				}
				if outcome == "late" {
					requireCode(t, artifactStopResult(t, done), foundation.CommitUnknown)
					budget, cancel := context.WithTimeout(context.Background(), 60*time.Millisecond)
					err = on.artifact.Drain(budget)
					cancel()
					if err == nil {
						t.Fatal("unknown original writer was forgotten")
					}
				}
				release.Do(func() { close(proxy.release) })
				if outcome == "late" {
					artifactStopWait(t, proxy.committed)
				} else {
					requireCode(t, artifactStopResult(t, done), foundation.CommitUnknown)
				}
				budget, cancel := context.WithTimeout(context.Background(), time.Second)
				defer cancel()
				if err = on.artifact.Drain(budget); err != nil {
					t.Fatal("original admission did not settle", err)
				}
				var count int
				if err = f.store.QueryRow(contextFor(t), `SELECT count(*) FROM agenteam_artifact.project_work WHERE process_id<>(SELECT process_id FROM agenteam_artifact.project_work ORDER BY id LIMIT 1)`).Scan(&count); err != nil {
					t.Fatal(err)
				}
				if (outcome == "rolled_back" && count != 0) || (outcome != "rolled_back" && count != 1) {
					t.Fatal("wrong durable admission outcome", outcome, count)
				}
				if on.resolver.resolves.Load() != 0 || body.reads.Load() != 0 || f.proxy.gets.Load() != gets || f.proxy.puts.Load() != puts {
					t.Fatal("unknown admission eventually started I/O")
				}
			})
		}
	}
}

type artifactDiscardTiming struct {
	duration time.Duration
	err      error
}
type artifactObservedDiscard struct {
	oc.Uploads
	seen     atomic.Bool
	entered  chan struct{}
	returned chan artifactDiscardTiming
}

func (o *artifactObservedDiscard) DiscardPrepared(p oc.PreparedPayload) error {
	first := o.seen.CompareAndSwap(false, true)
	started := time.Now()
	if first {
		close(o.entered)
	}
	err := o.Uploads.DiscardPrepared(p)
	if first {
		o.returned <- artifactDiscardTiming{time.Since(started), err}
	}
	return err
}

					t.Log("actual Discard duration", timing.duration, "result", timing.err, "public Unknown after COMMIT barrier", time.Since(commitReached))

					t.Fatal("publication returned before actual Discard returned")
				}

			}
		})
	}

func TestArtifactProjectStopUnknownPreflightAndGateNeverCancel(t *testing.T) {
	for _, phase := range []string{"preflight", "gate"} {
		t.Run(phase, func(t *testing.T) {
			f := newArtifactStopFixture(t, artifactStopOptions{})
			source := f.create(t, "stop-unknown-source", "text/plain", "body")
			ref, _ := source.Details().Reference.BusinessFile()
			var resolver *artifactHeldResolver
			on, wrapped, proxy := artifactCommitFixture(t, f, false, true, artifactStopOptions{resolve: func(r oc.SourceResolver) oc.SourceResolver { resolver = newArtifactHeldResolver(r); return resolver }})
			var release, unblock sync.Once
			defer release.Do(func() { close(proxy.release) })
			defer unblock.Do(func() { close(resolver.release) })
			copyDone := make(chan error, 1)
			go func() {
				_, e := on.artifact.CreateFromSource(contextFor(t), on.invocation(t), command(t, "held-stop-copy"), art.Display{Name: "copy"}, ref)
				copyDone <- e
			}()
			original := artifactResolverEntered(t, resolver)
			var work string
			if err := f.store.QueryRow(contextFor(t), `SELECT id::text FROM agenteam_artifact.project_work WHERE project_id=$1 AND joined_at IS NULL`, f.project.String()).Scan(&work); err != nil {
				t.Fatal("exact live resolver work", err)
			}
			actor, cause := f.activate(t, f.project, oc.ProjectStopArchive)
			requestCtx := contextFor(t)
			wrapped.arm(phase)
			type answer struct {
				report oc.ObjectStopReport
				err    error
			}
			reports := make(chan answer, 1)
			go func() { r, e := on.artifact.RequestProjectStop(requestCtx, actor, cause); reports <- answer{r, e} }()
			artifactStopWait(t, proxy.reached)
			wrapped.assertHit(t, phase)
			var first answer
			select {
			case first = <-reports:
			case <-time.After(3 * time.Second):
				t.Fatal("stop unknown did not return")
			}
			if first.err != nil || first.report.Details().SafeReason != oc.ProjectStopOutcomeUnknown || original.Err() != nil || on.stopObjects.requests.Load() != 0 {
				t.Fatal("unconfirmed stop cancelled or invoked Object", first.err, first.report.Details())
			}
			budget, cancel := context.WithTimeout(requestCtx, 80*time.Millisecond)
			_, err := on.artifact.RequestProjectStop(budget, actor, cause)
			cancel()
			if err == nil || original.Err() != nil || on.stopObjects.requests.Load() != 0 {
				t.Fatal("repeat crossed unresolved original stop writer")
			}
			release.Do(func() { close(proxy.release) })
			artifactStopWait(t, proxy.committed)
			cursor := func() (int, bool) {
				t.Helper()
				var lane int
				var revoked bool
				if err := f.store.QueryRow(requestCtx, `SELECT COALESCE((SELECT scan_kind FROM agenteam_artifact.project_stops WHERE project_id=$1 AND operation_id=$2),-1),EXISTS(SELECT 1 FROM agenteam_artifact.project_work WHERE id=$3 AND revoked_by=$2)`, f.project.String(), cause.Details().OperationID.String(), work).Scan(&lane, &revoked); err != nil {
					t.Fatal("exact recovered cursor/work", err)
				}
				return lane, revoked
			}
			lane, revoked := cursor()
			t.Log("original COMMIT terminal", phase, "cursor", lane, "work revoked", revoked, "original context", original.Err())
			if phase == "gate" && (lane != 1 || !revoked) || phase == "preflight" && (lane != -1 || revoked) {
				t.Fatal("late COMMIT did not preserve its actual original phase", lane, revoked)
			}
			// The confirmed late gate advanced the persistent cursor before the
			// caller learned its outcome. Drive one fixed four-lane cycle using
			// the same original request budget; no new timeout or polling sleep.
			for round := 0; round < 4; round++ {
				r, e := on.artifact.RequestProjectStop(requestCtx, actor, cause)
				lane, revoked = cursor()
				t.Log("recovery round", round, "cursor", lane, "work revoked", revoked, "report", r.Details(), "error", e, "original context", original.Err())
				if e != nil || r.Validate() != nil || r.Details().State != oc.ProjectStopPending {
					t.Fatal("held original resolver failed conservative recovery", e, r.Details())
				}
				select {
				case e := <-copyDone:
					t.Fatal("original resolver returned before actual release", e)
				default:
				}
				if original.Err() != nil {
					break
				}
			}
			if original.Err() == nil {
				t.Fatal("one complete cursor cycle did not cancel exact original")
			}
			unblock.Do(func() { close(resolver.release) })
			if artifactStopResult(t, copyDone) == nil {
				t.Fatal("revoked copy succeeded")
			}
			artifactStopSettled(t, on.artifact, actor, cause)
		})
	}
}

type artifactDiscardBarrier struct {
	oc.Uploads
	entered, release chan struct{}
	once             sync.Once
}

func (p *artifactDiscardBarrier) DiscardPrepared(v oc.PreparedPayload) error {
	err := p.Uploads.DiscardPrepared(v)
	p.once.Do(func() { close(p.entered); <-p.release })
	return err
}
func holdArtifactRecord(t *testing.T, f *fixture, id string) func() {
	t.Helper()
	key, _ := foundation.RecordLock(foundation.ReferenceRecordLock, "artifact-work:"+id)
	held, release := make(chan struct{}), make(chan struct{})
	done := make(chan foundation.CommitResult, 1)
	var once sync.Once
	go func() {
		done <- f.store.WithinTx(contextFor(t), cause(t), func(ctx context.Context, tx foundation.Tx) error {
			if err := f.store.AcquireAll(ctx, tx, []foundation.LockRequest{{Key: key, Mode: foundation.Exclusive}}); err != nil {
				return err
			}
			close(held)
			<-release
			return nil
		})
	}()
	artifactStopWait(t, held)
	unlock := func() {
		once.Do(func() { close(release) })
		select {
		case r := <-done:
			if r.State() != foundation.Committed {
				t.Error(r.Fault())
			}
		case <-time.After(time.Second):
			t.Error("work blocker did not join")
		}
	}
	t.Cleanup(func() { once.Do(func() { close(release) }) })
	return unlock
}

func testArtifactLocalJoinUnknown(t *testing.T, late bool) {
	f := newArtifactStopFixture(t, artifactStopOptions{})
	var discard *artifactDiscardBarrier
	on, wrapped, proxy := artifactCommitFixture(t, f, !late, late, artifactStopOptions{uploads: func(u oc.Uploads) oc.Uploads {
		discard = &artifactDiscardBarrier{Uploads: u, entered: make(chan struct{}), release: make(chan struct{})}
		return discard
	}})
	var release, body sync.Once
	defer release.Do(func() { close(proxy.release) })
	defer body.Do(func() { close(discard.release) })
	done := make(chan error, 1)
	go func() {
		_, e := on.artifact.CreateFromContent(contextFor(t), on.invocation(t), command(t, "join-unknown"), art.Display{Name: "body"}, "text/plain", "body")
		done <- e
	}()
	artifactStopWait(t, discard.entered)
	var work string
	if err := f.store.QueryRow(contextFor(t), `SELECT w.id::text FROM agenteam_artifact.project_work w JOIN agenteam_artifact.commands c ON c.artifact_id=w.resource_id WHERE c.command_key='join-unknown' AND c.state='completed' AND w.joined_at IS NULL`).Scan(&work); err != nil {
		t.Fatal("final publication not committed before join barrier", err)
	}
	unlock := holdArtifactRecord(t, f.fixture, work)
	body.Do(func() { close(discard.release) })
	if err := artifactStopResult(t, done); err != nil {
		t.Fatal("business publication", err)
	}
	unlock()
	actor, cause := f.activate(t, f.project, oc.ProjectStopArchive)
	wrapped.arm("join")
	type answer struct {
		r oc.ObjectStopReport
		e error
	}
	reports := make(chan answer, 1)
	requestCtx := contextFor(t)
	if late {
		var cancel context.CancelFunc
		requestCtx, cancel = context.WithTimeout(requestCtx, 500*time.Millisecond)
		defer cancel()
	}
	go func() { r, e := on.artifact.RequestProjectStop(requestCtx, actor, cause); reports <- answer{r, e} }()
	artifactStopWait(t, proxy.reached)
	wrapped.assertHit(t, "join")
	if !late {
		release.Do(func() { close(proxy.release) })
	}
	select {
	case a := <-reports:
		if late && a.e != nil {
			requireCode(t, a.e, foundation.CommitUnknown)
		} else if a.e != nil || a.r.Details().State != oc.ProjectStopPending || a.r.Details().SafeReason != oc.ProjectStopOutcomeUnknown || len(a.r.Details().UnknownRefs) == 0 {
			t.Fatal("join Unknown was downgraded to ordinary work pending", a.e, a.r.Details())
		}
	case <-time.After(3 * time.Second):
		t.Fatal("join unknown stop stuck")
	}
	if late {
		var joined bool
		if err := f.store.QueryRow(contextFor(t), `SELECT joined_at IS NOT NULL FROM agenteam_artifact.project_work WHERE id=$1`, work).Scan(&joined); err != nil || joined {
			t.Fatal("uncommitted join writer was not actually held", joined, err)
		}
		budget, cancel := context.WithTimeout(context.Background(), 60*time.Millisecond)
		err := on.artifact.Drain(budget)

// This observer delegates every query, lock and transaction to the real store.
// It records completed physical read batches; it does not supply SQL results.
type artifactBatchFacts struct {
	reads    map[string]bool
	controls map[string]bool
	writes   bool
}
type artifactBatchStore struct {
	*postgres.Store
	authority    *artifactStopAuthority
	mu           sync.Mutex
	enabled      bool
	transactions map[foundation.Tx]*artifactBatchFacts
	readMembers  []int
	readRows     []int
	maxControls  int
	afterFirst   func()
}

func (s *artifactBatchStore) start(after func()) {
	s.mu.Lock()
	s.enabled, s.afterFirst = true, after
	s.transactions = map[foundation.Tx]*artifactBatchFacts{}
	s.readMembers, s.readRows = nil, nil
	s.maxControls = 0
	s.mu.Unlock()
}
func (s *artifactBatchStore) WithinTx(ctx context.Context, cause foundation.TransactionCause, fn func(context.Context, foundation.Tx) error) foundation.CommitResult {
	var token foundation.Tx
	var facts *artifactBatchFacts
	result := s.Store.WithinTx(ctx, cause, func(ctx context.Context, tx foundation.Tx) error {
		token = tx
		s.mu.Lock()
		if s.enabled {
			facts = &artifactBatchFacts{reads: map[string]bool{}, controls: map[string]bool{}}
			s.transactions[tx] = facts
		}
		s.mu.Unlock()
		return fn(ctx, tx)
	})
	if facts == nil {
		return result
	}
	s.authority.mu.Lock()
	_, authorized := s.authority.validated[token]
	s.authority.mu.Unlock()
	s.mu.Lock()
	if authorized && len(facts.controls) > s.maxControls {
		s.maxControls = len(facts.controls)
	}
	var after func()
	if authorized && !facts.writes && result.State() == foundation.Committed {
		s.readMembers = append(s.readMembers, len(facts.controls)-1)
		s.readRows = append(s.readRows, len(facts.reads))
		after, s.afterFirst = s.afterFirst, nil
	}
	delete(s.transactions, token)
	s.mu.Unlock()
	if after != nil {
		after()
	}
	return result
}
func (s *artifactBatchStore) AcquireAll(ctx context.Context, tx foundation.Tx, locks []foundation.LockRequest) error {
	s.mu.Lock()
	if f := s.transactions[tx]; f != nil {
		for _, lock := range locks {
			if strings.Contains(lock.Key.Canonical(), ":artifact-control:") {
				f.controls[lock.Key.Canonical()] = true
			}
		}
	}
	s.mu.Unlock()
	return s.Store.AcquireAll(ctx, tx, locks)
}
func (s *artifactBatchStore) InTx(tx foundation.Tx) (postgres.SQLExecutor, error) {
	e, err := s.Store.InTx(tx)
	if err != nil {
		return nil, err
	}
	return &artifactBatchExecutor{SQLExecutor: e, owner: s, tx: tx}, nil
}

type artifactBatchExecutor struct {
	postgres.SQLExecutor
	owner *artifactBatchStore
	tx    foundation.Tx
}

func (e *artifactBatchExecutor) QueryRow(ctx context.Context, query string, args ...any) postgres.Row {
	if strings.Contains(query, "FROM agenteam_artifact.project_work WHERE id=$1") && len(args) > 0 {
		e.owner.mu.Lock()
		if f := e.owner.transactions[e.tx]; f != nil {
			f.reads[fmt.Sprint(args[0])] = true
		}
		e.owner.mu.Unlock()
	}
	return e.SQLExecutor.QueryRow(ctx, query, args...)
}
func (e *artifactBatchExecutor) Exec(ctx context.Context, query string, args ...any) (pgconn.CommandTag, error) {
	e.owner.mu.Lock()
	if f := e.owner.transactions[e.tx]; f != nil {
		f.writes = true
	}
	e.owner.mu.Unlock()
	return e.SQLExecutor.Exec(ctx, query, args...)
}
func (s *artifactBatchStore) assertTwoBatches(t *testing.T, durable bool) {
	t.Helper()
	s.mu.Lock()
	defer s.mu.Unlock()
	if len(s.readMembers) < 2 || s.readMembers[0] != 64 || s.readMembers[1] != 1 || s.maxControls > 65 {
		t.Fatal("local snapshot or final gate was not bounded", s.readMembers, s.readRows, s.maxControls)
	}
	if durable && (s.readRows[0] != 64 || s.readRows[1] != 1) {
		t.Fatal("local DB tuple reads were not real bounded batches", s.readRows)
	}
	t.Log("actual readonly local batches", s.readMembers, "work row identities", s.readRows, "largest gate/control union", s.maxControls)
}

// This observer delegates the original transaction and one complete lock
// acquisition unchanged. It records normal Store outcomes; it neither changes
// protocol frames nor keeps an original backend alive after ROLLBACK.
type artifactFirstAttempt struct {
	tx    foundation.Tx
	pid   int32
	locks []foundation.LockRequest
}
type artifactFirstAttemptStore struct {
	*postgres.Store
	mu                  sync.Mutex
	original            foundation.Tx
	first, confirmation chan artifactFirstAttempt
	result              chan foundation.CommitResult
	seen                int
}

func (s *artifactFirstAttemptStore) WithinTx(ctx context.Context, cause foundation.TransactionCause, fn func(context.Context, foundation.Tx) error) foundation.CommitResult {
	var tx foundation.Tx
	r := s.Store.WithinTx(ctx, cause, func(ctx context.Context, token foundation.Tx) error { tx = token; return fn(ctx, token) })
	s.mu.Lock()
	original := tx == s.original && tx.Valid()
	s.mu.Unlock()
	if original {
		s.result <- r
	}
	return r
}
func (s *artifactFirstAttemptStore) AcquireAll(ctx context.Context, tx foundation.Tx, locks []foundation.LockRequest) error {
	e, err := s.Store.InTx(tx)
	if err != nil {
		return err
	}
	var pid int32
	if err = e.QueryRow(ctx, `SELECT pg_backend_pid()`).Scan(&pid); err != nil {
		return err
	}
	s.mu.Lock()
	s.seen++
	which := s.seen
	if which == 1 {
		s.original = tx
	}
	s.mu.Unlock()
	a := artifactFirstAttempt{tx: tx, pid: pid, locks: append([]foundation.LockRequest{}, locks...)}
	if which == 1 {
		s.first <- a
	} else if which == 2 {
		s.confirmation <- a
	}
	return s.Store.AcquireAll(ctx, tx, locks)
}
func artifactWaitProjectLock(t *testing.T, f *fixture, a artifactFirstAttempt, project identity.ProjectID) {
	t.Helper()
	key, _ := foundation.ProjectLock(project.String())
	v := uint64(key.AdvisoryKey())
	ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
	defer cancel()
	for {
		var waiting bool
		if err := f.store.QueryRow(ctx, `SELECT EXISTS(SELECT 1 FROM pg_locks WHERE pid=$1 AND locktype='advisory' AND NOT granted AND classid::bigint=$2 AND objid::bigint=$3 AND objsubid=1)`, a.pid, int64(v>>32), int64(v&0xffffffff)).Scan(&waiting); err != nil {
			t.Fatal(err)
		}
		if waiting {
			return
		}
		select {
		case <-ctx.Done():
			t.Fatal("exact backend did not wait for owned Project holder", a.pid)
		case <-time.After(5 * time.Millisecond):
		}
	}
}
func TestArtifactProjectStopFirstLookupFailureRetainsConfirmation(t *testing.T) {
	f := newArtifactStopFixture(t, artifactStopOptions{})
	observer := &artifactFirstAttemptStore{Store: f.store, first: make(chan artifactFirstAttempt, 1), confirmation: make(chan artifactFirstAttempt, 1), result: make(chan foundation.CommitResult, 1)}
	on := bindArtifactStop(t, f.fixture, f.proxy, f.stopAuthority, artifactStopOptions{port: observer, guard: true})
	key, _ := foundation.ProjectLock(f.project.String())
	held, release := make(chan struct{}), make(chan struct{})
	holderDone := make(chan foundation.CommitResult, 1)
	go func() {
		holderDone <- f.store.WithinTx(contextFor(t), cause(t), func(ctx context.Context, tx foundation.Tx) error {
			if err := f.store.AcquireAll(ctx, tx, []foundation.LockRequest{{Key: key, Mode: foundation.Exclusive}}); err != nil {
				return err
			}
			close(held)
			<-release
			return nil
		})
	}()
	artifactStopWait(t, held)
	var open sync.Once
	unblock := func() {
		open.Do(func() {
			close(release)
			select {
			case r := <-holderDone:
				if r.State() != foundation.Committed {
					t.Error("normal holder did not confirm", r.State())
				}
			case <-time.After(3 * time.Second):
				t.Error("normal holder did not return")
			}
		})
	}
	defer unblock()
	ctx, cancel := context.WithCancel(contextFor(t))
	defer cancel()
	done := make(chan error, 1)
	beforePUT := f.proxy.puts.Load()
	go func() {
		_, err := on.artifact.CreateFromContent(ctx, on.invocation(t), command(t, "first-lookup-lock"), art.Display{Name: "new"}, "text/plain", "body")
		done <- err
	}()
	var first artifactFirstAttempt
	select {
	case first = <-observer.first:
	case <-time.After(3 * time.Second):
		t.Fatal("first physical lookup was not reached")
	}
	artifactWaitProjectLock(t, f.fixture, first, f.project)
	var commandHeld, controlHeld bool
	for _, l := range first.locks {
		if !strings.HasPrefix(l.Key.Canonical(), "command:") && !strings.Contains(l.Key.Canonical(), ":artifact-control:") {
			continue
		}
		v := uint64(l.Key.AdvisoryKey())
		var granted bool
		if err := f.store.QueryRow(contextFor(t), `SELECT EXISTS(SELECT 1 FROM pg_locks WHERE pid=$1 AND locktype='advisory' AND granted AND classid::bigint=$2 AND objid::bigint=$3 AND objsubid=1)`, first.pid, int64(v>>32), int64(v&0xffffffff)).Scan(&granted); err != nil {
			t.Fatal(err)
		}
		if strings.HasPrefix(l.Key.Canonical(), "command:") {
			commandHeld = granted
		} else {
			controlHeld = granted
		}
	}
	if !commandHeld || controlHeld {
		t.Fatal("first physical Acquire was not partial before control EX", commandHeld, controlHeld)
	}
	cancel()
	var original foundation.CommitResult
	select {
	case original = <-observer.result:
	case <-time.After(3 * time.Second):
		t.Fatal("normal Store cancellation did not return")
	}
	if original.State() != foundation.NotCommitted {
		t.Fatal("normal Store result was not NotCommitted", original.State())
	}
	var confirmation artifactFirstAttempt
	select {
	case confirmation = <-observer.confirmation:
	case <-time.After(3 * time.Second):
		t.Fatal("failed first lookup retired without full union confirmation")
	}
	if confirmation.tx == first.tx || len(confirmation.locks) != len(first.locks) {
		t.Fatal("cleanup did not confirm the original physical union")
	}
	for i, l := range first.locks {
		if foundation.CompareLockKeys(confirmation.locks[i].Key, l.Key) != 0 || confirmation.locks[i].Mode != foundation.Exclusive {
			t.Fatal("cleanup changed a key or failed to confirm with EX")
		}
	}
	artifactWaitProjectLock(t, f.fixture, confirmation, f.project)
	on.artifact.StopAdmission()
	budget, stop := context.WithTimeout(context.Background(), 60*time.Millisecond)
	err := on.artifact.Drain(budget)
	stop()
	if err == nil || on.artifact.Joined() {
		t.Fatal("whole call retired before full union confirmation")
	}
	assertArtifactGuardHeld(t, on)
	var work, native, published int
	if err = f.store.QueryRow(contextFor(t), `SELECT (SELECT count(*) FROM agenteam_artifact.project_work),(SELECT count(*) FROM agenteam_artifact.commands),(SELECT count(*) FROM agenteam_artifact.artifacts)`).Scan(&work, &native, &published); err != nil || work != 0 || native != 0 || published != 0 || f.proxy.puts.Load() != beforePUT {
		t.Fatal("failed first lookup started durable work or object I/O", work, native, published, err)
	}
	unblock()
	err = artifactStopResult(t, done)
	var fault *foundation.Fault
	if !errors.As(err, &fault) || fault.CommitState != foundation.NotCommitted || fault.Code != original.Fault().Code {
		t.Fatal("public call rewrote the original Store failure", err)
	}
	budget, stop = context.WithTimeout(context.Background(), 2*time.Second)
	defer stop()
	if err = on.artifact.Drain(budget); err != nil || !on.artifact.Joined() {
		t.Fatal("released real confirmation failed to join", err)
	}
	t.Log("normal first lookup backend", first.pid, "partial Command held before control; separate confirmation backend", confirmation.pid, "full EX union", len(confirmation.locks), "original result", original.State())
}
