from pathlib import Path
from urllib.parse import unquote
import hashlib
import json
import re
import subprocess

REPO = Path('/workspace/agenteam')
OUT = REPO/'docs/development/agent-team/evidence/artifact-project-stop-verification'
TMP = Path('/tmp/agenteam-artifact-archive-t64zx6Zx')
DOCS = [
    'docs/development/agent-team/artifact-project-stop-verification.md',
    'docs/development/agent-team/evidence/artifact-project-stop-verification/README.md',

TRACKED = DOCS[2:]

def sha(data):
    return hashlib.sha256(data).hexdigest()

commands = []
def run(args):
    result = subprocess.run(args, cwd=REPO, capture_output=True, text=True)
    commands.append({'argv': args, 'cwd': str(REPO), 'exit_code': result.returncode, 'stdout_sha256': sha(result.stdout.encode()), 'stdout_bytes': len(result.stdout.encode()), 'stderr': result.stderr})
    no_index_difference = '--no-index' in args and result.returncode == 1 and not result.stdout and not result.stderr
    if no_index_difference:
        commands[-1]['interpretation'] = 'expected no-index difference exit 1; no whitespace diagnostics'
    assert result.returncode == 0 or no_index_difference, commands[-1]
    return result.stdout

raw = json.loads((OUT/'source-map.json').read_text())['raw_files']
for name, meta in raw.items():
    data = (OUT/name).read_bytes()
    assert sha(data) == meta['sha256']
    for source in meta['original_paths']:
        assert Path(source).read_bytes() == data, ('original changed', source)

link_count = 0
fragment_count = 0
def slugs(path):
    result = set()
    duplicates = {}
    for match in re.finditer(r'^#{1,6}\s+(.+?)\s*#*$', path.read_text(), re.M):
        title = re.sub(r'<[^>]*>', '', match.group(1)).strip().lower()
        title = re.sub(r'[^\w\-\s\u3400-\u9fff]', '', title).replace(' ', '-')
        n = duplicates.get(title, 0)
        duplicates[title] = n+1
        result.add(title if n == 0 else f'{title}-{n}')
    return result

for name in DOCS:
    p = REPO/name
    data = p.read_bytes()
    text = data.decode('utf-8')
    assert b'\r' not in data
    assert data.endswith(b'\n')
    assert not re.search(r'[ \t]+$', text, re.M), name
    fences = re.findall(r'^```', text, re.M)
    assert len(fences) % 2 == 0, ('unbalanced fences', name)
    for destination in re.findall(r'\[[^\]]*\]\(([^)]+)\)', text):
        destination = destination.strip('<>')
        if re.match(r'^[a-zA-Z][a-zA-Z0-9+.-]*:', destination):
            continue
        destination = unquote(destination)
        relative, _, fragment = destination.partition('#')
        target = (p.parent/relative).resolve() if relative else p
        assert target.exists(), (name, destination)
        link_count += 1
        if fragment:
            fragment_count += 1
            assert fragment in slugs(target), ('fragment missing', name, destination)
