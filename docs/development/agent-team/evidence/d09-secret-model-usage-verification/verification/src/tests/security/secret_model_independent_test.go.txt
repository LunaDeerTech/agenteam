type independentModelAuditResult struct {
	next ac.Appender
	err  error
}

func (a *independentModelAuditResult) AppendInTx(ctx context.Context, tx foundation.Tx, e ac.Entry, k ac.AppendKey) (ac.AppendReceipt, error) {
	r, err := a.next.AppendInTx(ctx, tx, e, k)
	a.err = err
	return r, err
}

func TestIndependentSecretModelRequestAndWitness(t *testing.T) {
	t.Run("other-real-invocation", func(t *testing.T) {
		f, request, _ := newModelReadCase(t, false, sc.ModelCallOwner)
		s, tap := newModelReadService(t, f, f.store)
		owner, actor := bindModelOwner(t, f.secretFixture, request.Ref, sc.Model, sc.ModelCallOwner)
		lease := f.acquire(t, request.Ref, owner, actor)
		other := f.modelReadRequest(t, actor, lease.LeaseID)
		material, err := s.ReadCredentialForUsage(auditContext(t), other)
		if err != nil {
			t.Fatal("alternate Invocation is not independently valid", err)
		}
		assertModelValue(t, material, "model-usage-material")
		request.RequestID = other.RequestID
		material, err = s.ReadCredentialForUsage(auditContext(t), request)
		modelNoMaterial(t, material, err)
		requireCode(t, err, foundation.Forbidden)
		if tap.calls != 1 || resolveCount(t, f.secretFixture) != 1 {
			t.Fatal("alternate valid Invocation was rebound to a different exact lease")
		}
	})
	t.Run("foreign-provider-issuer", func(t *testing.T) {
		f, request, _ := newModelReadCase(t, false, sc.ExecutionOwner)
		s, tap := newModelReadService(t, f, f.store)
		if _, err := s.DiscoverUsage(auditContext(t), request); err != nil {
			t.Fatal("valid persisted facts did not plan", err)
		}
		changed := 0
		f.ports.transformPlan = func(_ sc.UsageRequest, original sc.UsageDependencies) (sc.UsageDependencies, error) {
			changed++
			return sc.NewUsageDependencies(sc.NewPlanIssuer(), original.Binding(), original.Mapping(), original.RequiredLocks())
		}
		material, err := s.ReadCredentialForUsage(auditContext(t), request)
		modelNoMaterial(t, material, err)
		requireCode(t, err, foundation.ResourceBusy)
		if changed != 1 || tap.calls != 0 || resolveCount(t, f.secretFixture) != 0 {
			t.Fatal("wrong issuer test did not exercise its exact validated plan")
		}
	})
	t.Run("typed-valid-requestid-tamper-reaches-checker", func(t *testing.T) {
		f, request, _ := newModelReadCase(t, false, sc.ModelCallOwner)
		s, tap := newModelReadService(t, f, f.store)
		observed := &independentModelAuditResult{next: tap.next}
		tap.next = observed
		checks := f.ports.checks
		tap.before = func(ctx context.Context, tx foundation.Tx, entry ac.Entry, key ac.AppendKey) (context.Context, ac.Entry, ac.AppendKey, error) {
			fields := entry.Fields()
			fields.Associations.RequestID = newID[struct{}](t).String()
			changed, err := ac.NewEntry(fields)
			if err != nil {
				t.Fatal("negative Entry is not typed-valid", err)
			}
			return ctx, changed, key, nil
		}
		material, err := s.ReadCredentialForUsage(auditContext(t), request)
		modelNoMaterial(t, material, err)
		requireCode(t, err, foundation.DependencyUnavailable)
		requireCode(t, observed.err, foundation.Forbidden)
		var inner *secret.Error
		if !errors.As(observed.err, &inner) || inner.Code() != secret.AuthorizationRejected || tap.calls != 1 || f.ports.checks != checks+1 || resolveCount(t, f.secretFixture) != 0 {
			t.Fatal("RequestID rejection was not the actual Secret fact checker")
		}
	})
	t.Run("fresh-tx-with-live-context-cannot-reuse-proof", func(t *testing.T) {
		f, request, _ := newModelReadCase(t, false, sc.ExecutionOwner)
		s, tap := newModelReadService(t, f, f.store)
		material, err := s.ReadCredentialForUsage(auditContext(t), request)
		if err != nil {
			t.Fatal(err)
		}
		material.Destroy()
		proofContext := context.WithoutCancel(tap.ctx)
		if proofContext.Err() != nil {
			t.Fatal("retained witness context is already cancelled")
		}
		plan, err := s.DiscoverUsage(auditContext(t), request)
		if err != nil {
			t.Fatal(err)
		}
		called := false
		var rejected error
		result := f.store.WithinTx(auditContext(t), txCause(t), func(ctx context.Context, tx foundation.Tx) error {
			if err := f.store.AcquireAll(ctx, tx, plan.RequiredLocks()); err != nil {
				return err
			}
			if tx == tap.tx {
				t.Fatal("expected a different actual transaction")
			}
			called = true
			rejected = f.ports.checker.CheckProjectAuditInTx(proofContext, tx, tap.entry, tap.key)
			return rejected
		})
		requireCode(t, rejected, foundation.Forbidden)
		requireCode(t, result.Fault(), foundation.Forbidden)
		var inner *secret.Error
		if !called || result.State() != foundation.NotCommitted || !errors.As(rejected, &inner) || inner.Code() != secret.AuthorizationRejected || resolveCount(t, f.secretFixture) != 1 {
			t.Fatal("cross-Tx proof test stopped at a different boundary")
		}
	})
}

type independentModelOutcomeStore struct {
	*modelReadStore
	cancel context.CancelFunc
	calls  int
}

func (s *independentModelOutcomeStore) WithinTx(ctx context.Context, cause foundation.TransactionCause, fn func(context.Context, foundation.Tx) error) foundation.CommitResult {
	result := s.modelReadStore.WithinTx(ctx, cause, fn)
	if s.armed && s.cancel != nil {
		s.calls++
		s.cancel()
	}
	return result
}

func TestIndependentSecretModelRevalidationAndOutcome(t *testing.T) {
	t.Run("actual-lock-then-valid-mapping-change", func(t *testing.T) {
		f, request, _ := newModelReadCase(t, false, sc.ExecutionOwner)
		store := &modelReadStore{Store: f.store}
		s, tap := newModelReadService(t, f, store)
		store.unions = nil
		original, err := s.DiscoverUsage(auditContext(t), request)
		if err != nil {
			t.Fatal(err)
		}
		var call string
		if err := f.store.QueryRow(auditContext(t), `SELECT call_id::text FROM audit_fixture.secret_invocations WHERE id=$1`, request.RequestID).Scan(&call); err != nil {
			t.Fatal(err)
		}
		gate, _ := foundation.AggregateLock(foundation.OperationAggregate, call)
		process := newID[struct{}](t).String()
		ctx, cancel := context.WithCancel(auditContext(t))
		defer cancel()
		held, release := make(chan int, 1), make(chan struct{})
		var releaseOnce sync.Once
		unlock := func() { releaseOnce.Do(func() { close(release) }) }
		defer unlock()
		holder := make(chan foundation.CommitResult, 1)
		go func() {
			holder <- f.store.WithinTx(ctx, txCause(t), func(ctx context.Context, tx foundation.Tx) error {
				if err := f.store.AcquireAll(ctx, tx, []foundation.LockRequest{{Key: gate, Mode: foundation.Exclusive}}); err != nil {
					return err
				}
				x, err := f.store.InTx(tx)
				if err != nil {
					return err
				}
				var pid int
				if err := x.QueryRow(ctx, `SELECT pg_backend_pid()`).Scan(&pid); err != nil {
					return err
				}
				held <- pid
				select {
				case <-release:
				case <-ctx.Done():
					return ctx.Err()
				}
				if _, err := x.Exec(ctx, `UPDATE audit_fixture.secret_calls SET process_id=$2 WHERE id=$1`, call, process); err != nil {
					return err
				}
				_, err = x.Exec(ctx, `UPDATE audit_fixture.secret_invocations SET process_id=$2 WHERE id=$1`, request.RequestID, process)
				return err
			})
		}()
		var pid int
		select {
		case pid = <-held:
		case result := <-holder:
			t.Fatal("holder exited before acquiring its real gate", result.Fault())
		case <-ctx.Done():
			t.Fatal("holder failed to enter")
		}
		type outcome struct {
			material sc.SecretMaterial
			err      error
		}
		done := make(chan outcome, 1)
		go func() { m, err := s.ReadCredentialForUsage(ctx, request); done <- outcome{m, err} }()
		ticker := time.NewTicker(5 * time.Millisecond)
		defer ticker.Stop()
		for {
			var waiting bool
			if err := f.store.QueryRow(ctx, `SELECT EXISTS(SELECT 1 FROM pg_stat_activity WHERE datname=$1 AND wait_event_type='Lock' AND $2=ANY(pg_blocking_pids(pid)))`, f.db.Name, pid).Scan(&waiting); err != nil {
				t.Fatal(err)
			}
			if waiting {
				break
			}
			select {
			case <-ticker.C:
			case result := <-done:
				result.material.Destroy()
				t.Fatalf("read did not block on exact holder: %v", result.err)
			case <-ctx.Done():
				t.Fatal("actual blocking relationship missing")
			}
		}
		if resolveCount(t, f.secretFixture) != 0 {
			t.Fatal("read wrote Audit before acquiring its complete union")
		}
		unlock()
		if result := <-holder; result.State() != foundation.Committed {
			t.Fatal(result.Fault())
		}
		result := <-done
		modelNoMaterial(t, result.material, result.err)
		requireCode(t, result.err, foundation.ResourceBusy)
		current, err := s.DiscoverUsage(auditContext(t), request)
		if err != nil || current.Mapping() == original.Mapping() {
			t.Fatal("changed tuple was not still valid with a distinct mapping", err)
		}
		store.mu.Lock()
		unions := append([][]foundation.LockRequest(nil), store.unions...)
		store.mu.Unlock()
		want := original.RequiredLocks()
		if len(unions) != 1 || len(unions[0]) != len(want) {
			t.Fatal("read did not acquire exactly one complete planned union")
		}
		for i, lock := range want {
			if unions[0][i].Key.Canonical() != lock.Key.Canonical() || unions[0][i].Mode != lock.Mode {
				t.Fatal("planned lock identity/order/mode changed")
			}
		}
		if tap.calls != 0 || resolveCount(t, f.secretFixture) != 0 {
			t.Fatal("stale mapping reached Audit")
		}
		t.Log("exact PG blocker observed; one unchanged union; valid post-lock mapping changed and was rejected")
	})
	for _, rollback := range []bool{false, true} {
		name := "returned-unknown-after-real-commit-and-cancel"
		if rollback {
			name = "returned-unknown-after-real-rollback-and-cancel"
		}
		t.Run(name, func(t *testing.T) {
			f, request, _ := newModelReadCase(t, false, sc.ModelCallOwner)
			base := &modelReadStore{Store: f.store, marker: newID[foundation.TransactionAttempt](t), rollback: rollback}
			store := &independentModelOutcomeStore{modelReadStore: base}
			s, tap := newModelReadService(t, f, store)
			ctx, cancel := context.WithCancel(auditContext(t))
			defer cancel()
			store.cancel, base.armed = cancel, true
			material, err := s.ReadCredentialForUsage(ctx, request)
			modelNoMaterial(t, material, err)
			requireCode(t, err, foundation.CommitUnknown)
			var fault *foundation.Fault
			var original interface {
				error
				AttemptID() foundation.ID[foundation.TransactionAttempt]
				Cause() foundation.TransactionCause
			}
			if ctx.Err() != context.Canceled || !errors.As(err, &fault) || fault.CommitState != foundation.Unknown || !errors.As(err, &original) || original.AttemptID() != base.returned.AttemptID() || !reflect.DeepEqual(original.Cause().Details(), base.returned.Cause().Details()) {
				t.Fatal("caller cancellation replaced original returned Unknown identity", err)
			}
			wantState, wantRows := foundation.Committed, 1
			if rollback {
				wantState, wantRows = foundation.NotCommitted, 0
			}
			if base.actual.State() != wantState || store.calls != 1 || tap.calls != 1 || resolveCount(t, f.secretFixture) != wantRows || tap.entry.Fields().Associations.RequestID != request.RequestID || original.Cause().Details().RecoveryRunID != tap.key.Details().CauseRef {
				t.Fatal("real transaction/canonical Audit proof was lost")
			}
			t.Logf("actual Tx=%s; decorated original marker/cause preserved despite cancelled caller; zero material; no physical backend attempt claim", wantState)
		})
	}
}
