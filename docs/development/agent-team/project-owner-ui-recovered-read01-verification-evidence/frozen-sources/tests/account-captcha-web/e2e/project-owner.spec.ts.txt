const acknowledgementFields = {
  observe: [
    "project",
    "update_commands",
    "update_audits",
    "update_events",
    "dropped",
    "replayed",
    "same_original",
    "patch_requests",
    "lookup_requests",
  ],
  "drop-next-update": [],
  "fail-next-read": [],
  "fail-session": [],
  "hold-next-read": [],
  "hold-status": ["started", "finished"],
  "release-read": [],
  update: ["project"],
  archive: ["fact_only"],
  "reuse-name": ["project"],
} as const;
let sequence = 0;
async function ipc(
  action: keyof typeof acknowledgementFields,
  fields: Record<string, string> = {},
) {
  const current = ++sequence;
  const request = join(directory, "project-owner-ipc.json");
  const ack = join(directory, `project-owner-ack-${current}.json`);
  writeFileSync(
    request + ".tmp",
    JSON.stringify({ sequence: current, action, ...fields }),
    {
      mode: 0o600,
    },
  );
  renameSync(request + ".tmp", request);
  await expect
    .poll(() => existsSync(ack), { timeout: 4_000, intervals: [20, 40, 100] })
    .toBe(true);
  const value = JSON.parse(readFileSync(ack, "utf8"));
  // The frozen Go protocol returns sequence and action-specific fields, with
  // no synthetic "ok" flag. Service ProjectRef may omit archived_at, so only
  // the safe fields used by this private observation are validated here.
  const keys = ["sequence", ...acknowledgementFields[action]];
  expect(
    value !== null &&
      typeof value === "object" &&
      !Array.isArray(value) &&
      Object.keys(value).length === keys.length &&
      keys.every((key) => Object.hasOwn(value, key)) &&
      value.sequence === current,
  ).toBe(true);
  if (action === "observe" || action === "update" || action === "reuse-name") {
    const project = value.project;
    const data = material();
    const key = fields.project ?? "main";
    const owner =
      action !== "reuse-name" && key === "admin" ? data.admin : data.owner;
    expect(
      project !== null &&
        typeof project === "object" &&
        !Array.isArray(project) &&
        typeof project.id === "string" &&
        uuid7.test(project.id) &&
        project.owner_user_id === owner.user_id &&
        (action === "reuse-name" || project.id === data.ids[key]) &&
        typeof project.description === "string" &&
        typeof project.version === "string" &&
        /^[1-9][0-9]{0,18}$/.test(project.version) &&
        BigInt(project.version) <= 9223372036854775807n,
    ).toBe(true);
  }
  if (action === "observe") {
    expect(
      [
        "update_commands",
        "update_audits",
        "update_events",
        "dropped",
        "replayed",
        "patch_requests",
        "lookup_requests",
      ].every((key) => Number.isSafeInteger(value[key]) && value[key] >= 0) &&
        typeof value.same_original === "boolean",
    ).toBe(true);
  }
  if (action === "hold-status")
    expect(
      typeof value.started === "boolean" && typeof value.finished === "boolean",
    ).toBe(true);
  if (action === "archive") expect(value.fact_only === true).toBe(true);
  return value as Observation & {
    started: boolean;
    finished: boolean;
    fact_only: boolean;
  };
}
function complete(value: Record<string, unknown>) {

    JSON.stringify({ completed: true, ...value }),
    {
      mode: 0o600,
    },
  );
}

// Observe only safe request coordinates and the response's request ID. Never
// inspect Cookie, CSRF, command keys, request bodies or clone a response stream.
type ReadFact = { url: URL; ownerID: string; status: number };
const readFacts = new Map<string, ReadFact>();
const observationTails: Promise<void>[] = [];
function observe(page: Page, initialOwner: string) {
  let ownerID = initialOwner;
  const requests: { method: string; endpoint: string }[] = [];
  page.on("request", (request) => {
    const endpoint = new URL(request.url()).pathname;
    if (endpoint.startsWith("/api/v1/"))
      requests.push({ method: request.method(), endpoint });
  });
  page.on("response", (response) => {
    const url = new URL(response.url());
    if (
      response.request().method() !== "GET" ||
      !url.pathname.startsWith("/api/v1/projects")
    )
      return;
    const captured = ownerID;
    observationTails.push(
      (async () => {
        const id = await response.headerValue("x-request-id");
        if (id)
          readFacts.set(id, {
            url,
            ownerID: captured,
            status: response.status(),
          });
      })(),
    );
  });
  return {
    requests,
    identity: (value: string) => {
      ownerID = value;
    },
  };
}
async function login(page: Page, credential: Credential) {
  await expect(page.locator("#login-email")).toBeVisible();
  // Playwright errors must not print a credential passed to fill().
  try {
    await page.locator("#login-email").fill(credential.email);
    await page.locator("#login-password").fill(credential.password);
  } catch {
    throw new Error("PRIVATE_PROJECT_LOGIN_INPUT_FAILED");
  }
  await button(page, "登录").click();
  await expect(button(page, "退出登录")).toBeEnabled();
  await expect(page).not.toHaveURL(/\/login(?:\?|$)/);
}

async function factsUnchanged(before: Observation, after: Observation) {
  expect(after.update_commands === before.update_commands).toBe(true);
  expect(after.update_audits === before.update_audits).toBe(true);
  expect(after.update_events === before.update_events).toBe(true);
}

// The fixture stores the original, complete safe bytes before any deliberate
// response loss. Schema validation and the public client below read those same
// bytes; successful mutation decoding remains the application's own operation.
const schemaProgram = String.raw`
import hashlib,json,pathlib,sys
from jsonschema import Draft202012Validator
from referencing import Registry,Resource
from referencing.jsonschema import DRAFT202012
try:
 root=pathlib.Path(sys.argv[1]); evidence=pathlib.Path(sys.argv[2]); base=root/'api/openapi'
 docs={n:json.loads((base/n).read_bytes()) for n in ['common.json','project-owner.json','project-usage.json']}
 registry=Registry().with_resources(((base/n).as_uri(),Resource.from_contents(d,default_specification=DRAFT202012)) for n,d in docs.items())
 count=0
 for path in sorted(evidence.glob('response-*.json')):
  m=json.loads(path.read_bytes()); filename='body-'+m['body_sha256']+'.json'
  assert m['body_file']==filename and len(m['input_hash'])==64 and m['source_run'].startswith('TestAccountProjectOwnerWeb')
  raw=(evidence/filename).read_bytes(); assert hashlib.sha256(raw).hexdigest()==m['body_sha256']
  endpoint=m['endpoint']; name='project-usage.json' if endpoint=='/api/v1/projects/resolve' else 'project-owner.json'
  route=endpoint if endpoint in ['/api/v1/projects','/api/v1/projects/resolve'] else '/api/v1/projects/{id}'+('/commands/lookup' if endpoint.endswith('/commands/lookup') else '')
  response=docs[name]['paths'][route][m['method'].lower()]['responses'][str(m['status'])]
  media=m['content_type'].split(';')[0].strip().lower(); ref=response['content'][media]['schema']
  schema=dict(docs[name]); schema['$id']=(base/name).as_uri(); schema.update(ref)
  Draft202012Validator(schema,registry=registry).validate(json.loads(raw)); count+=1
 assert count>0
 print(count)
except Exception:
 print('SAFE_PROJECT_BODY_SCHEMA_FAILED',file=sys.stderr); sys.exit(1)
`;
async function verifyBodies(requireReads = false) {
  await Promise.all(observationTails);
  const checked = spawnSync(
    "python3",
    ["-c", schemaProgram, repository, evidence],
    {
      encoding: "utf8",
      timeout: 6_000,
      maxBuffer: 4096,
    },
  );
  writeFileSync(
    join(evidence, "schema-validation.json"),
    JSON.stringify({
      status: checked.status,
      signal: checked.signal,
      stdout: checked.stdout,
      stderr: checked.stderr,
      failed_to_run: !!checked.error,
    }),
    { mode: 0o600 },
  );
  expect(checked.status === 0 && /^\d+\s*$/.test(checked.stdout)).toBe(true);
  const counts = { list: 0, resolve: 0, get: 0, problem: 0 };
  for (const name of readdirSync(evidence).filter((name) =>
    /^response-\d+\.json$/.test(name),
  )) {
    const meta = JSON.parse(readFileSync(join(evidence, name), "utf8"));
    const fact = readFacts.get(meta.request_id);
    if (!fact || meta.method !== "GET") continue;
    expect(
      meta.endpoint === fact.url.pathname && meta.status === fact.status,
    ).toBe(true);
    expect(meta.body_file === `body-${meta.body_sha256}.json`).toBe(true);
    const raw = readFileSync(join(evidence, meta.body_file));
    expect(
      createHash("sha256").update(raw).digest("hex") === meta.body_sha256,
    ).toBe(true);
    const api = createProjectOwnerAPI(
      async () =>
        new Response(new Uint8Array(raw).buffer, {
          status: meta.status,
          headers: {
            "Content-Type": meta.content_type,
            "X-Request-ID": meta.request_id,
          },
        }),
    );
    const signal = new AbortController().signal;
    const url = fact.url;
    const kind =
      url.pathname === "/api/v1/projects"
        ? "list"
        : url.pathname.endsWith("/resolve")
          ? "resolve"
          : "get";
    try {
      if (kind === "list")
        await api.list(
          {
            limit: Number(url.searchParams.get("limit") ?? "50"),
            ...(url.searchParams.has("cursor")
              ? { cursor: url.searchParams.get("cursor")! }
              : {}),
            ...(url.searchParams.has("lifecycle")
              ? {
                  lifecycle: url.searchParams
                    .get("lifecycle")!
                    .split(",") as ProjectLifecycle[],
                }
              : {}),
          },
          signal,
        );
      else if (kind === "resolve")
        await api.resolve(
          {
            username: url.searchParams.get("username")!,
            project_name: url.searchParams.get("project_name")!,
          },
          fact.ownerID,
          signal,
        );
      else await api.get(meta.project_id, fact.ownerID, signal);
      expect(meta.status === 200).toBe(true);
      counts[kind]++;
    } catch (error) {
      expect(
        meta.status !== 200 &&
          error instanceof AccountFailure &&
          error.kind === "problem",
      ).toBe(true);
      counts.problem++;
    }
  }
  if (requireReads)
    expect(counts.list > 0 && counts.resolve > 0 && counts.get > 0).toBe(true);
  const value = { schemas: Number(checked.stdout), client_reads: counts };
  writeFileSync(
    join(evidence, "same-body-validation.json"),
    JSON.stringify(value),
    { mode: 0o600 },
  );
  return value;
}

async function refreshSession(page: Page) {
  await page.evaluate(() => window.dispatchEvent(new Event("pageshow")));
}
async function sessionID(page: Page) {
  return page.evaluate(async () => {
    const response = await fetch("/api/v1/session", {
      credentials: "same-origin",
      cache: "no-store",
    });
    if (response.status !== 200) return "";
    const value = await response.json();
    return typeof value.session?.id === "string" ? value.session.id : "";
  });
}
async function revokeCurrentSession(page: Page) {
  const status = await page.evaluate(async () => {
    const current = await fetch("/api/v1/session", {
      credentials: "same-origin",
      cache: "no-store",
    });
    const value = await current.json();
    const response = await fetch("/api/v1/sessions/logout", {
      method: "POST",
      credentials: "same-origin",
      cache: "no-store",
      redirect: "error",
      headers: {
        "Content-Type": "application/json",
        "X-CSRF-Token": value.csrf_token,
        "Idempotency-Key": crypto.randomUUID(),
      },
      body: "{}",
    });
    return response.status;
  });
  expect(status === 204).toBe(true);
  await refreshSession(page);
  await expect(page).toHaveURL(/\/login$/);
}

async function noOverflow(page: Page) {
  expect(
    await page.evaluate(() => {
      const nodes = [
        document.documentElement,
        document.body,
        document.querySelector("main"),
        document.querySelector(".project-form"),
      ].filter((n): n is HTMLElement => n instanceof HTMLElement);
      return nodes.every((node) => node.scrollWidth <= node.clientWidth + 1);
    }),
  ).toBe(true);
}
async function focusedInside(page: Page) {
  expect(
    await dialog(page).evaluate((node) =>
      node.contains(document.activeElement),
    ),
  ).toBe(true);
}
