const invitationIndexMigration = "-- agenteam:transaction tx\n-- +goose Up\nCREATE INDEX account_invitation_delivery_link ON agenteam_account.delivery_intents(link_id,id) WHERE kind='invitation';\n"

// Read the actual private SQL literal rather than maintaining a second query.
func invitationReadSQL(t *testing.T) string {
	t.Helper()
	raw, e := os.ReadFile("../../internal/central/account/http_facade.go")
	if e != nil {
		t.Fatal(e)
	}
	tree, e := parser.ParseFile(token.NewFileSet(), "http_facade.go", raw, 0)
	if e != nil {
		t.Fatal(e)
	}
	var query string
	ast.Inspect(tree, func(node ast.Node) bool {
		v, ok := node.(*ast.ValueSpec)
		if ok && len(v.Names) == 1 && v.Names[0].Name == "accountHTTPInvitationSelect" {
			literal, ok := v.Values[0].(*ast.BasicLit)
			if !ok {
				t.Fatal("query is no longer one literal")
			}
			query, e = strconv.Unquote(literal.Value)
			if e != nil {
				t.Fatal(e)
			}
		}
		return true
	})
	if query == "" {
		t.Fatal("actual invitation query missing")
	}
	return query
}

func invitationExplain(t *testing.T, conn *pgx.Conn, query, stage string) {
	t.Helper()
	if _, e := conn.Exec(ctxFor(t), `ANALYZE agenteam_account.invitations; ANALYZE agenteam_account.delivery_intents; ANALYZE agenteam_account.commands; ANALYZE agenteam_account.mail_jobs; ANALYZE agenteam_account.mail_attempts`); e != nil {
		t.Fatal(e)
	}
	var created time.Time
	if e := conn.QueryRow(ctxFor(t), `SELECT created_at FROM agenteam_account.invitations LIMIT 1`).Scan(&created); e != nil {
		t.Fatal(e)
	}
	for _, tc := range []struct {
		name  string
		at    any
		id    any
		limit int
	}{
		{"normal", nil, nil, 26}, {"next-page", created, "01930000-0000-7000-8000-000000000015", 26},
		{"empty", time.Date(2000, 1, 1, 0, 0, 0, 0, time.UTC), "01930000-0000-7000-8000-000000000001", 26},
		{"time-ties", nil, nil, 31}, {"skew", created, "01930000-0000-7000-8000-000000000002", 2},
	} {
		ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
		var raw []byte
		e := conn.QueryRow(ctx, `EXPLAIN (ANALYZE, BUFFERS, FORMAT JSON) `+query, tc.at, tc.id, tc.limit).Scan(&raw)
		cancel()
		if e != nil {
			t.Fatalf("%s/%s actual plan within budget: %v", stage, tc.name, e)
		}
		if !json.Valid(raw) {
			t.Fatal("plan JSON invalid")
		}
		t.Logf("EXPLAIN %s/%s %s", stage, tc.name, raw)
		if stage == "after" && tc.name != "empty" && !strings.Contains(string(raw), "account_invitation_delivery_link") {
			t.Fatal("scale plan did not locate invitation history with link index", tc.name)
		}
	}
}
func TestAccountInvitationDeliveryReadMigrationAndPlan(t *testing.T) {
	raw, e := fs.ReadFile(migrations.SQL, "00019_account_invitation_delivery_read.sql")
	if e != nil || string(raw) != invitationIndexMigration {
		t.Fatal("fixed migration bytes", e)
	}
	db := pgfixture.NewDatabase(t)
	cfg := db.Config(t, nil)
	files := migrationFiles(t, "00018")
	source, e := postgres.NewSource(files, nil)
	if e != nil {
		t.Fatal(e)
	}
	migrate(t, cfg, source)
	conn := db.Connect(t)
	invitationSeedScale(t, conn)
	query := invitationReadSQL(t)
	rows, e := conn.Query(ctxFor(t), `SELECT pg_get_indexdef(indexrelid) FROM pg_index WHERE indrelid='agenteam_account.delivery_intents'::regclass ORDER BY indexrelid`)
	if e != nil {
		t.Fatal(e)
	}
	for rows.Next() {
		var definition string
		if e = rows.Scan(&definition); e != nil {
			t.Fatal(e)
		}
		t.Log("BEFORE pg_index", definition)
	}
	rows.Close()
	if rows.Err() != nil {
		t.Fatal(rows.Err())
	}
	invitationExplain(t, conn, query, "before")
	var before string
	fingerprint := `SELECT md5(concat((SELECT string_agg(row_to_json(i)::text,',' ORDER BY id) FROM agenteam_account.delivery_intents i),(SELECT string_agg((id,intent_id,phase,attempts,version,fence,current_attempt_id,reason)::text,',' ORDER BY id) FROM agenteam_account.mail_jobs),(SELECT string_agg((id,job_id,fence,channel,result,phase,terminal,io_joined,token_ref,credential_ref)::text,',' ORDER BY id) FROM agenteam_account.mail_attempts),(SELECT string_agg((id,version,created_at,expires_at,material_ref)::text,',' ORDER BY id) FROM agenteam_account.invitations),(SELECT string_agg((id,version,password_version,auth_sequence)::text,',' ORDER BY id) FROM agenteam_account.users)))`
	if e = conn.QueryRow(ctxFor(t), fingerprint).Scan(&before); e != nil {
		t.Fatal(e)
	}
	// The same candidate first passed the private before/after plan before the
	// global file was written; the final run binds its exact repository bytes.
	files["00019_account_invitation_delivery_read.sql"] = &fstest.MapFile{Data: []byte(invitationIndexMigration)}
	proposed, e := postgres.NewSource(files, nil)
	if e != nil {
		t.Fatal(e)
	}
	migrate(t, cfg, proposed)
	invitationExplain(t, conn, query, "after")
	var after, definition string
	if e = conn.QueryRow(ctxFor(t), fingerprint).Scan(&after); e != nil || before != after {
		t.Fatal("migration modified history", e)
	}
	if e = conn.QueryRow(ctxFor(t), `SELECT pg_get_indexdef('agenteam_account.account_invitation_delivery_link'::regclass)`).Scan(&definition); e != nil || definition != "CREATE INDEX account_invitation_delivery_link ON agenteam_account.delivery_intents USING btree (link_id, id) WHERE (kind = 'invitation'::text)" {
		t.Fatal("index definition", definition, e)
	}
	t.Log("AFTER pg_index", definition)
	// Exercise the actual SELECT, including the 2048-cycle tie break.
	rows, e = conn.Query(ctxFor(t), query, nil, nil, 31)
	if e != nil {
		t.Fatal(e)
	}
	count := 0
	for rows.Next() {
		values, e := rows.Values()
		if e != nil {
			t.Fatal(e)
		}
		count++
		if values[0] == "01930000-0000-7000-8000-000000000001" && values[5] != "01920000-0000-7000-8000-000000018ea0" {
			t.Fatal("skew latest accepted tie-break", values[5])
		}
	}
	rows.Close()
	if e = rows.Err(); e != nil || count != 30 {
		t.Fatal("bounded query", count, e)
	}
	// With all live rows expired but retained, the same statement is empty.
	tx, e := conn.Begin(ctxFor(t))
	if e != nil {
		t.Fatal(e)
	}
	if _, e = tx.Exec(ctxFor(t), `WITH at AS MATERIALIZED (SELECT clock_timestamp()-interval '1 hour' AS value) UPDATE agenteam_account.invitations SET expires_at=at.value,created_at=at.value-interval '24 hours' FROM at`); e != nil {
		t.Fatal(e)
	}
	var emptyPlan []byte
	if e = tx.QueryRow(ctxFor(t), `EXPLAIN (ANALYZE, BUFFERS, FORMAT JSON) `+query, nil, nil, 26).Scan(&emptyPlan); e != nil {
		t.Fatal(e)
	}
	t.Logf("EXPLAIN after/no-live %s", emptyPlan)
	empty, e := tx.Query(ctxFor(t), query, nil, nil, 26)
	if e != nil {
		t.Fatal(e)
	}
	if empty.Next() {
		t.Fatal("expired scale rows visible")
	}
	empty.Close()
	if e = empty.Err(); e != nil {
		t.Fatal(e)
	}
	if e = tx.Rollback(ctxFor(t)); e != nil {
		t.Fatal(e)
	}
	t.Run("fresh-1-19", func(t *testing.T) {
		fresh := pgfixture.NewDatabase(t)
		migrate(t, fresh.Config(t, nil))
		var applied int
		if e := fresh.Connect(t).QueryRow(ctxFor(t), `SELECT count(*) FROM agenteam_meta.goose_db_version WHERE version_id>0 AND is_applied`).Scan(&applied); e != nil || applied != 19 {
			t.Fatal("fresh contiguous migrations", applied, e)
		}
	})
	t.Run("fault-rollback-and-same-bytes-recovery", func(t *testing.T) {
		broken := pgfixture.NewDatabase(t)
		cfg := broken.Config(t, nil)
		files := migrationFiles(t, "00018")
		base, e := postgres.NewSource(files, nil)
		if e != nil {
			t.Fatal(e)
		}
		migrate(t, cfg, base)
		conn := broken.Connect(t)
		files["00019_account_invitation_delivery_read.sql"] = &fstest.MapFile{Data: append(append([]byte{}, raw...), []byte("SELECT invitation_owned_migration_failure();\n")...)}
		bad, e := postgres.NewSource(files, nil)
		if e != nil {
			t.Fatal(e)
		}
		m, e := postgres.NewMigrator(cfg, bad)
		if e != nil {
			t.Fatal(e)
		}
		if result := m.Migrate(ctxFor(t)); result.Migrated {
			t.Fatal("faulting index DDL committed")
		}
		var absent bool
		var pending, applied int
		if e = conn.QueryRow(ctxFor(t), `SELECT to_regclass('agenteam_account.account_invitation_delivery_link') IS NULL,(SELECT count(*) FROM agenteam_meta.migration_journal WHERE version=19 AND state='pending'),(SELECT count(*) FROM agenteam_meta.goose_db_version WHERE version_id=19 AND is_applied)`).Scan(&absent, &pending, &applied); e != nil || !absent || pending != 1 || applied != 0 {
			t.Fatal("index/journal rollback", absent, pending, applied, e)
		}
		if _, e = conn.Exec(ctxFor(t), `CREATE FUNCTION public.invitation_owned_migration_failure() RETURNS void LANGUAGE SQL AS 'SELECT NULL::void'`); e != nil {
			t.Fatal(e)
		}
		migrate(t, cfg, bad) // Exact same pending bytes; only the owned failing dependency is repaired.
		if e = conn.QueryRow(ctxFor(t), `SELECT count(*) FROM agenteam_meta.goose_db_version WHERE version_id=19 AND is_applied`).Scan(&applied); e != nil || applied != 1 {
			t.Fatal("same-byte recovery journal", applied, e)
		}
	})

}

type invitationPlannedGate struct {
	*postgres.Store
	fired            atomic.Bool
	arrived, release chan struct{}
}

func (s *invitationPlannedGate) WithinTx(ctx context.Context, cause foundation.TransactionCause, fn func(context.Context, foundation.Tx) error) foundation.CommitResult {
	result := s.Store.WithinTx(ctx, cause, fn)
	if result.State() == foundation.Committed && cause.Details().Primary.Key() == "directory-early-plan" && s.fired.CompareAndSwap(false, true) {
		close(s.arrived)
		select {
		case <-s.release:
		case <-ctx.Done():
		}
	}
	return result
}
func TestAccountInvitationDeliveryReadSelectionAndBudget(t *testing.T) {
	db, store, _ := database(t)
	gate := &invitationPlannedGate{Store: store, arrived: make(chan struct{}), release: make(chan struct{})}
	accountKeys, _, _ := keys(t)
	authority, e := account.NewAuthority(gate, accountKeys)
	if e != nil {
		t.Fatal(e)
	}
	f := assembleB02(t, db, store, gate, authority, nil)
	admin := b02Admin(t, f)
	_, ring, _ := keys(t)
	facade, e := account.NewSystemHTTPFacade(f.service, ring)
	if e != nil {
		t.Fatal(e)
	}
	create := func(key, email string) (c.InvitationReceipt, error) {
		q, e := c.NewInvitationCreate(c.InvitationCreateFields{Actor: admin, Key: foundation.IdempotencyKey(key), Email: email})
		if e != nil {
			return c.InvitationReceipt{}, e
		}
		return f.service.CreateInvitation(ctxFor(t), q)
	}
	initial, e := create("directory-initial", "selection@example.test")
	if e != nil {
		t.Fatal(e)
	}
	other, e := create("directory-other", "other@example.test")
	if e != nil {
		t.Fatal(e)
	}

		t.Helper()
		v, e := facade.ListInvitations(ctxFor(t), admin, account.HTTPListRequest{Limit: 100})
		if e != nil {
			t.Fatal("list", e)
		}
		return v
	}
	find := func(inv c.InvitationID) account.HTTPInvitation {
		t.Helper()
		for _, v := range read().Items {
			if v.ID == inv {
				return v
			}
		}
		t.Fatal("invitation missing")
		return account.HTTPInvitation{}
	}
	check := func(job c.JobID) {
		t.Helper()
		v := find(initial.ID)
		if v.LatestDelivery.Status.JobID != job {
			t.Fatal("latest accepted task", v.LatestDelivery.Status.JobID.String(), job.String())
		}
	}
	check(initial.JobID)
	if v := find(initial.ID); v.LatestDelivery.Channel != nil || v.LatestDelivery.AttemptResult != nil || v.LatestDelivery.Status.Phase != "enqueue_pending" {
		t.Fatal("unenqueued projection")
	}

		t.Helper()
		if _, e := store.Exec(ctxFor(t), `UPDATE agenteam_account.invitations SET last_delivery_at=clock_timestamp()-interval '61 seconds' WHERE id=$1`, initial.ID.String()); e != nil {
			t.Fatal(e)
		}
	}
	cooldown()
	type accepted struct {
		receipt c.InvitationReceipt
		err     error
	}
	done := make(chan accepted, 1)
	go func() { v, e := create("directory-early-plan", "selection@example.test"); done <- accepted{v, e} }()
	released := false
	defer func() {
		if !released {
			close(gate.release)
		}
	}()
	await(t, gate.arrived)
	newer, e := create("directory-newer-plan", "selection@example.test")
	if e != nil {
		t.Fatal(e)
	}
	check(newer.JobID)
	cooldown()
	close(gate.release)
	released = true
	var late accepted
	select {
	case late = <-done:
	case <-time.After(5 * time.Second):
		t.Fatal("late acceptance did not join")
	}
	if late.err != nil {
		t.Fatal(late.err)
	}
	check(late.receipt.JobID)
	var earlyPlan, newerPlan, earlyAccepted, newerAccepted time.Time
	if e = store.QueryRow(ctxFor(t), `SELECT a.created_at,b.created_at,a.completed_at,b.completed_at FROM agenteam_account.commands a CROSS JOIN agenteam_account.commands b WHERE a.attempt_id=$1 AND b.attempt_id=$2`, late.receipt.JobID.String(), newer.JobID.String()).Scan(&earlyPlan, &newerPlan, &earlyAccepted, &newerAccepted); e != nil || !earlyPlan.Before(newerPlan) || !earlyAccepted.After(newerAccepted) {
		t.Fatal("controlled planned/accepted order absent", e)
	}
	t.Log("real early-planned/late-accepted transaction wins across roots")
	if _, e = f.service.ReconcileDeliveryIntents(ctxFor(t)); e != nil {
		t.Fatal(e)
	}
	port, runtime := newManualMail(t, f)
	cancelMailAttempt(t, port, runtime, initial.JobID, true)
	check(late.receipt.JobID) // Old completion must not become newest acceptance.
	current := initial.JobID
	for cycle := range 2 {
		status, e := f.service.GetMailJob(ctxFor(t), admin, current)
		if e != nil {
			t.Fatal(e)
		}
		q := c.MailJobRetry{Actor: admin, Key: foundation.IdempotencyKey(fmt.Sprintf("directory-retry-%d", cycle)), JobID: current, ExpectedVersion: status.Version}
		before := find(initial.ID).LatestDelivery.Status.JobID
		type observed struct {
			page account.HTTPInvitationList
			err  error
		}
		observations := make(chan observed, 8)
		ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
		go func() {
			for range 8 {
				p, e := facade.ListInvitations(ctx, admin, account.HTTPListRequest{Limit: 100})
				observations <- observed{p, e}
			}
		}()
		retry, e := f.service.RetryMailJob(ctxFor(t), q)
		if e != nil {
			cancel()
			t.Fatal(e)
		}
		for range 8 {
			select {
			case got := <-observations:
				if got.err != nil {
					cancel()
					t.Fatal(got.err)
				}
				for _, v := range got.page.Items {
					if v.ID == initial.ID && v.LatestDelivery.Status.JobID != before && v.LatestDelivery.Status.JobID != retry.JobID {
						t.Fatal("mixed concurrent retry snapshot")
					}
				}
			case <-ctx.Done():
				t.Fatal("concurrent reads did not join")
			}
		}
		cancel()
		replay, e := f.service.RetryMailJob(ctxFor(t), q)
		if e != nil || replay.JobID != retry.JobID {
			t.Fatal("same-key retry changed", e)
		}
		check(retry.JobID)
		if _, e = f.service.ReconcileDeliveryIntents(ctxFor(t)); e != nil {
			t.Fatal(e)
		}
		cancelMailAttempt(t, port, runtime, retry.JobID, true)
		current = retry.JobID
	}
	// Fixture ties prove the declared stable job ID order, not acceptance timing.
	tie := time.Now().UTC().Add(time.Second).Truncate(time.Microsecond)
	if _, e = store.Exec(ctxFor(t), `UPDATE agenteam_account.commands SET completed_at=$3 WHERE attempt_id IN ($1,$2)`, newer.JobID.String(), current.String(), tie); e != nil {
		t.Fatal(e)
	}
	want := current
	if newer.JobID.String() > current.String() {
		want = newer.JobID
	}
	check(want)
	if _, e = store.Exec(ctxFor(t), `UPDATE agenteam_account.commands SET completed_at=$2 WHERE attempt_id=$1`, current.String(), tie.Add(time.Second)); e != nil {
		t.Fatal(e)
	}
	check(current)
	t.Log("manual retry and retry-of-retry reads stayed on complete old/new facts; same-key and acceptance ties preserved")
	assertZero := func(ctx context.Context) {
		t.Helper()
		v, e := facade.ListInvitations(ctx, admin, account.HTTPListRequest{})
		if e == nil || v.Items != nil || v.NextCursor != "" {
			t.Fatal("failed read published candidates", e)
		}
	}
	if _, e = store.Exec(ctxFor(t), `UPDATE agenteam_account.commands SET completed_at='10000-01-01Z'::timestamptz WHERE attempt_id=$1`, current.String()); e != nil {
		t.Fatal(e)
	}
	assertZero(ctxFor(t))
	if _, e = store.Exec(ctxFor(t), `UPDATE agenteam_account.commands SET completed_at=$2 WHERE attempt_id=$1`, current.String(), tie.Add(time.Second)); e != nil {
		t.Fatal(e)
	}
	var exact string
	if e = store.QueryRow(ctxFor(t), `SELECT current_attempt_id::text FROM agenteam_account.mail_jobs WHERE id=$1`, current.String()).Scan(&exact); e != nil {
		t.Fatal(e)
	}
	var otherAttempt string
	if e = store.QueryRow(ctxFor(t), `SELECT current_attempt_id::text FROM agenteam_account.mail_jobs WHERE id=$1`, initial.JobID.String()).Scan(&otherAttempt); e != nil {
		t.Fatal(e)
	}
	for _, replacement := range []string{id[struct{}](t).String(), otherAttempt} {
		if _, e = store.Exec(ctxFor(t), `UPDATE agenteam_account.mail_jobs SET current_attempt_id=$2 WHERE id=$1`, current.String(), replacement); e != nil {
			t.Fatal(e)
		}
		assertZero(ctxFor(t))
	}
	if _, e = store.Exec(ctxFor(t), `UPDATE agenteam_account.mail_jobs SET current_attempt_id=$2 WHERE id=$1`, current.String(), exact); e != nil {
		t.Fatal(e)
	}
	if _, e = store.Exec(ctxFor(t), `UPDATE agenteam_account.mail_attempts SET fence=fence+100 WHERE id=$1`, exact); e != nil {
		t.Fatal(e)
	}
	assertZero(ctxFor(t))
	if _, e = store.Exec(ctxFor(t), `UPDATE agenteam_account.mail_attempts SET fence=fence-100 WHERE id=$1`, exact); e != nil {
		t.Fatal(e)
	}
	check(current)
	t.Log("invalid accepted time, missing/foreign current attempt and wrong fence each returned zero candidates")
	// Actual advisory wait uses the same global configuration key as production.
	lock, _ := foundation.SystemConfigLock("account-directory")
	conn := db.Connect(t)
	hold := func() pgx.Tx {
		t.Helper()
		tx, e := conn.Begin(ctxFor(t))
		if e != nil {
			t.Fatal(e)
		}
		if _, e = tx.Exec(ctxFor(t), `SELECT pg_advisory_xact_lock($1)`, lock.AdvisoryKey()); e != nil {
			t.Fatal(e)
		}
		t.Cleanup(func() { _ = tx.Rollback(context.Background()) })
		return tx
	}
	waitBlocked := func() {
		t.Helper()
		deadline := time.Now().Add(time.Second)
		for {
			var blocked bool
			if e = store.QueryRow(ctxFor(t), `SELECT EXISTS(SELECT 1 FROM pg_stat_activity WHERE datname=current_database() AND wait_event_type='Lock' AND wait_event='advisory' AND $1=ANY(pg_blocking_pids(pid)))`, int64(conn.PgConn().PID())).Scan(&blocked); e != nil {
				t.Fatal(e)
			}
			if blocked {
				return
			}
			if time.Now().After(deadline) {
				t.Fatal("owned reader never reached actual lock wait")
			}
			time.Sleep(5 * time.Millisecond)
		}
	}
	if db.Config(t, nil).LockTimeout() != time.Second {
		t.Fatal("owned database lock timeout changed")
	}
	for _, scenario := range []struct {
		name   string
		budget time.Duration
	}{{"earlier parent cancellation", 60 * time.Millisecond}, {"earlier database lock timeout", 5 * time.Second}} {
		tx := hold()
		ctx, cancel := context.WithTimeout(context.Background(), scenario.budget)
		t.Cleanup(cancel)
		type failedRead struct {
			page account.HTTPInvitationList
			err  error
		}
		done := make(chan failedRead, 1)
		joined := make(chan struct{})
		started := time.Now()
		go func() {
			v, e := facade.ListInvitations(ctx, admin, account.HTTPListRequest{})
			done <- failedRead{v, e}
			close(joined)
		}()
		t.Cleanup(func() {
			cancel()
			select {
			case <-joined:
			case <-time.After(time.Second):
				t.Error("cancelled lock read did not join")
			}
		})
		waitBlocked() // This exact connection owns the lock blocking the reader.
		var got failedRead
		select {
		case got = <-done:
		case <-time.After(3500 * time.Millisecond):
			t.Fatal("owned lock waiter did not actually join")
		}
		elapsed := time.Since(started)
		var fault *foundation.Fault
		if !errors.As(got.err, &fault) || fault.Code != foundation.InternalError || fault.CommitState != foundation.NotCommitted || postgres.CodeOf(got.err) != postgres.LockFailed || got.page.Items != nil || got.page.NextCursor != "" {
			t.Fatal("failed lock read published candidates or lost NotCommitted", got.err)
		}
		if scenario.budget < time.Second {
			if !errors.Is(got.err, context.DeadlineExceeded) || ctx.Err() != context.DeadlineExceeded || elapsed > scenario.budget+500*time.Millisecond {
				t.Fatal("earlier parent cancellation was extended or lost", got.err, elapsed)
			}
		} else {
			var pg *pgconn.PgError
			if !errors.As(got.err, &pg) || pg.Code != "55P03" || ctx.Err() != nil || elapsed > 3500*time.Millisecond {
				t.Fatal("earlier database timeout lost its original cause or live caller", got.err, elapsed)
			}
		}
		cancel()
		if e = tx.Rollback(ctxFor(t)); e != nil {
			t.Fatal(e)
		}
		t.Log(scenario.name, "actually joined after", elapsed, "with NotCommitted and zero candidates")
	}
	// Expiry cutoff is acquired after a witnessed lock wait, before the SELECT.
	if _, e = store.Exec(ctxFor(t), `WITH at AS MATERIALIZED (SELECT clock_timestamp()+interval '250 milliseconds' AS value) UPDATE agenteam_account.invitations SET expires_at=at.value,created_at=at.value-interval '24 hours' FROM at WHERE id=$1`, other.ID.String()); e != nil {
		t.Fatal(e)
	}
	tx := hold()
	results := make(chan struct {
		page account.HTTPInvitationList
		err  error
	}, 1)
	go func() {
		v, e := facade.ListInvitations(ctxFor(t), admin, account.HTTPListRequest{})
		results <- struct {
			page account.HTTPInvitationList
			err  error
		}{v, e}
	}()
	waitBlocked()
	deadline := time.Now().Add(time.Second)
	for {
		var expired bool
		if e = store.QueryRow(ctxFor(t), `SELECT clock_timestamp()>expires_at FROM agenteam_account.invitations WHERE id=$1`, other.ID.String()).Scan(&expired); e != nil {
			t.Fatal(e)
		}
		if expired {
			break
		}
		if time.Now().After(deadline) {
			t.Fatal("expiry fixture deadline")
		}
		time.Sleep(5 * time.Millisecond)
	}
	if e = tx.Rollback(ctxFor(t)); e != nil {
		t.Fatal(e)
	}
	result := <-results
	if result.err != nil {
		t.Fatal(result.err)
	}
	for _, v := range result.page.Items {
		if v.ID == other.ID {
			t.Fatal("lock wait reused stale expiry cutoff")
		}
	}
	var retained int
	if e = store.QueryRow(ctxFor(t), `SELECT count(*) FROM agenteam_account.invitations WHERE id=$1`, other.ID.String()).Scan(&retained); e != nil || retained != 1 {
		t.Fatal("read deleted expired record", e)
	}
	t.Log("expired during witnessed lock wait: omitted by read, retained until write cleanup")
	// The formal write service retires the expired link and creates a new ID for
	// the same email. Its old delivery history must not join the replacement.
	replacement, e := create("directory-same-email-new-id", "other@example.test")
	if e != nil || replacement.ID == other.ID {
		t.Fatal("expired same-email invitation did not get a new ID", e)
	}
	if v := find(replacement.ID); v.LatestDelivery.Status.JobID != replacement.JobID {
		t.Fatal("same email reused the retired ID's delivery")
	}
	for _, v := range read().Items {
		if v.ID == other.ID {
			t.Fatal("retired ID appeared beside its same-email replacement")
		}
	}
	var oldLink string
	if e = store.QueryRow(ctxFor(t), `SELECT link_id::text FROM agenteam_account.delivery_intents WHERE job_id=$1`, other.JobID.String()).Scan(&oldLink); e != nil || oldLink != other.ID.String() {
		t.Fatal("same-email replacement changed old delivery history", e)
	}
	t.Log("formal same-email creation used a new InvitationID and preserved the old intent link")
	tx = hold()
	go func() {
		v, e := facade.ListInvitations(ctxFor(t), admin, account.HTTPListRequest{})
		results <- struct {
			page account.HTTPInvitationList
			err  error
		}{v, e}
	}()
	waitBlocked()
	if _, e = store.Exec(ctxFor(t), `UPDATE agenteam_account.users SET role='user' WHERE id=$1`, admin.Details().UserID); e != nil {
		t.Fatal(e)
	}
	if e = tx.Rollback(ctxFor(t)); e != nil {
		t.Fatal(e)
	}
	result = <-results
	if !hasCode(result.err, foundation.Forbidden) || result.page.Items != nil || result.page.NextCursor != "" {
		t.Fatal("old authenticated actor bypassed current transaction authority", result.err)
	}
	if _, e = store.Exec(ctxFor(t), `UPDATE agenteam_account.users SET role='admin' WHERE id=$1`, admin.Details().UserID); e != nil {
		t.Fatal(e)
	}
	t.Log("current transaction authority denied the already authenticated actor after lock wait")
	// Real revoke/consume boundary remains owned by the existing write service.
	if e = f.service.RevokeInvitation(ctxFor(t), c.InvitationRevoke{Actor: admin, Key: "directory-revoke", ID: replacement.ID, ExpectedVersion: replacement.Version}); e != nil {
		t.Fatal(e)
	}
	for _, v := range read().Items {
		if v.ID == replacement.ID {
			t.Fatal("revoked row visible")
		}
	}
	t.Log("formal revoke removed the same-email replacement from the directory")
}
