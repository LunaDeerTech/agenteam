import ctypes, hashlib, json, os, pathlib, shutil, signal, subprocess, sys, tempfile, time

if ctypes.CDLL(None, use_errno=True).prctl(36, 1, 0, 0, 0) != 0:
    raise OSError(ctypes.get_errno(), 'PR_SET_CHILD_SUBREAPER')

root = pathlib.Path(__file__).parent
name, source = sys.argv[1:3]
run = root / 'runs' / name
run.mkdir(parents=True, exist_ok=False)
tmp = pathlib.Path(tempfile.mkdtemp(prefix='df-', dir='/tmp'))
env = dict(os.environ, AGENTEAM_DIALOG_WEB_ROOT=str(root/source/'web'),
           AGENTEAM_DIALOG_RUN_DIR=str(run), AGENTEAM_DIALOG_CHROMIUM='/usr/lib/chromium/chromium',
           TMPDIR=str(tmp), CI='1', NO_COLOR='1')
command = ['tests/account-captcha-web/node_modules/.bin/playwright', 'test', '--config',
           'tests/account-captcha-web/dialog-outside-focus.config.js', *sys.argv[3:]]

def processes():
    rows = {}
    for entry in pathlib.Path('/proc').iterdir():
        if not entry.name.isdigit(): continue
        try:
            stat = (entry/'stat').read_text().rsplit(')', 1)[1].split()
            rows[int(entry.name)] = {'pid': int(entry.name), 'state': stat[0], 'ppid': int(stat[1]),
                'session': int(stat[3]), 'starttime': int(stat[19]),
                'command': (entry/'cmdline').read_bytes().replace(b'\0', b' ').decode(errors='replace')}
        except (FileNotFoundError, ProcessLookupError, PermissionError): pass
    return rows

def ports():
    return subprocess.run(['ss', '-lntp'], capture_output=True, text=True, check=True).stdout

baseline = processes()
(run/'baseline.json').write_text(json.dumps({'processes': list(baseline.values()), 'ports': ports()}, indent=2))
(run/'runner.py.txt').write_text(pathlib.Path(__file__).read_text())
(run/'command.json').write_text(json.dumps({'argv': command, 'cwd': str(root/'candidate'),
    'env': {key:env[key] for key in ['AGENTEAM_DIALOG_WEB_ROOT','AGENTEAM_DIALOG_RUN_DIR','AGENTEAM_DIALOG_CHROMIUM','TMPDIR','CI','NO_COLOR']}}, indent=2))
started = time.monotonic()
tracked = {}
timed_out = False
with (run/'raw.log').open('wb') as raw:
    process = subprocess.Popen(command, cwd=root/'candidate', env=env, stdout=raw, stderr=subprocess.STDOUT, start_new_session=True)
    while process.poll() is None:
        snapshot = processes()
        owned = {process.pid} | set(tracked)
        for _ in range(10):
            added = {pid for pid,row in snapshot.items() if row['ppid'] in owned or row['session'] == process.pid or str(tmp) in row['command']}
            if added <= owned: break
            owned |= added
        for pid in owned:
            if pid in snapshot and (pid not in baseline or baseline[pid]['starttime'] != snapshot[pid]['starttime']):
                tracked[pid] = snapshot[pid]
        if time.monotonic()-started > 240:
            timed_out = True
            os.killpg(process.pid, signal.SIGTERM)
            break
        time.sleep(.08)
    try: code = process.wait(timeout=10)
    except subprocess.TimeoutExpired:
        os.killpg(process.pid, signal.SIGKILL)
        code = process.wait(timeout=10)

def remaining():
    snapshot = processes()
    return [row for pid,row in snapshot.items() if (pid in tracked and row['starttime'] == tracked[pid]['starttime']) or str(tmp) in row['command']]

reaped = []
for _ in range(40):
    while True:
        try: pid, status = os.waitpid(-1, os.WNOHANG)
        except ChildProcessError: break
        if not pid: break
        reaped.append({'pid': pid, 'status': status})
    if not remaining(): break
    time.sleep(.05)
first = remaining()
time.sleep(.25)
second = remaining()
tmp_contents = [str(p.relative_to(tmp)) for p in tmp.rglob('*')]
if not first and not second:
    shutil.rmtree(tmp)
result = {'exit': code, 'actual_wait_completed': True, 'timed_out': timed_out, 'elapsed_seconds': round(time.monotonic()-started, 3), 'reaped': reaped,
    'tracked_processes': list(tracked.values()), 'remaining_first': first, 'remaining_second': second,
    'ports_after': ports(), 'tmpdir': str(tmp), 'tmp_contents_before_cleanup': tmp_contents, 'tmp_contents': [str(p.relative_to(tmp)) for p in tmp.rglob('*')] if tmp.exists() else [],
    'raw_sha256': hashlib.sha256((run/'raw.log').read_bytes()).hexdigest()}
(run/'terminal.json').write_text(json.dumps(result, indent=2))
print(json.dumps({key: result[key] for key in ['exit','timed_out','elapsed_seconds','remaining_first','remaining_second','tmp_contents','raw_sha256']}, indent=2))
print((run/'raw.log').read_text())
sys.exit(code if code else int(bool(first or second or result['tmp_contents'])))
